W18 spec: metadata cleaners show what was removed, item by item (website)
Author: Claude (project R&D lead). Branch: feature/W18 (from next). Reply in Chinese prose; code comments in English.

WHY (from user-pain research, ExifCleaner issues #317 and #302)
People do not trust a cleaner that just says "done". They want to see which fields were removed and proof that
the cleaned copy was checked again. Both our tools already re-scan the cleaned copy, but they only print one
summary sentence. W18 turns that into a per-item before/after result.

FILES (all exist; they are hand-written, served as-is from public/tools/assets/)
- public/tools/assets/exif.js        — pure JPEG EXIF reader: EXIFR.parse(ArrayBuffer) →
  { kind: 'jpeg'|'other', found, fields: [[label, value]...], gps: {lat,lon}|null, hasGPS }
- public/tools/assets/exif-app.js    — photo tool UI (English + Chinese strings in T, chosen by <html lang>)
- public/tools/assets/mr-app.js      — file tool UI; the WASM engine returns
  analyze_file(bytes).metadata.metadata_found = [{category, name, value}...] and
  process_file(bytes) = { cleaned: Uint8Array, verification: { metadata_found: [...] } }
- tests run with `node --test tests/*.test.js`; tests/exif.test.js shows the style (require the UMD module).
I attach the full current exif-app.js and the relevant part of mr-app.js below.

DELIVERABLES
1. New pure module public/tools/assets/cleandiff.js (UMD like exif.js: module.exports in Node, window.CLEANDIFF in
   the page). No DOM, no network.
   CLEANDIFF.compare(before, after) where both are arrays of {category, name, value} →
     { removed: [items from before not present after], left: [items from before still present after],
       added: [items present after but not before] }
   Match key: lower-cased trimmed category + "\u0000" + lower-cased trimmed name. Values are NOT compared (a
   cleaned field with a changed value still counts as left). Duplicate keys are counted (multiset): two "Author"
   before and one after → one removed, one left. Missing/garbage input (null, non-arrays, items without name)
   never throws; such items use name "" and still count.
   CLEANDIFF.fromExif(parsed) converts an EXIFR.parse result into that item shape:
   GPS → {category:'Location', name:'GPS', value:'lat, lon' or 'present'}; each field → category 'Device' if the
   label matches /camera|lens|software|owner/i, 'Dates' if /date/i, else 'Other'.
2. tests/cleandiff.test.js: node:test + assert. Cover: all removed; some left; duplicates; added items; null and
   garbage input; case/space-insensitive keys; fromExif on a GPS result and on a no-metadata result.

function show(rows) { tbody.textContent = ''; rows.forEach(function (r) { tbody.appendChild(r); }); }

function reset() {
  current = null;
  cleanBtn.hidden = true;
  cleanBtn.disabled = false;
  cleanBtn.textContent = BUTTON;
}

function cleanName(name) {
  name = asText(name);
  var m = /^(.*)(\.[^.\/\\]+)$/.exec(name);
  if (m && m[1]) return m[1] + '-clean' + m[2];
  return (name || 'file') + '-clean';
}

function download(bytes, name, type) {
  var blob = new Blob([bytes], { type: type || 'application/octet-stream' });
  var url = URL.createObjectURL(blob);
  var a = document.createElement('a');
  a.href = url;
  a.download = name;
  document.body.appendChild(a);
  a.click();
  a.remove();
  setTimeout(function () { URL.revokeObjectURL(url); }, 30000);
}

function groupsOf(meta) {
  var list = (meta && meta.metadata_found) || [];
  var groups = [], index = {};
  list.forEach(function (item) {
    var cat = asText(item && item.category) || T.other;
    var key = cat.toLowerCase();
    if (!Object.prototype.hasOwnProperty.call(index, key)) {
      index[key] = groups.length;
      groups.push({ name: cat, items: [] });
    }
    groups[index[key]].items.push(item || {});
  });
  return { groups: groups, total: list.length };
}

async function inspect(file) {
  var mine = ++seq;
  reset();
  if (!file) return;
  if (file.size > MAX_BYTES) {
    show([]);
    setStatus(T.tooBig, 'fail');
    return;
  }

  var bytes;
  try {
    var buf = await file.arrayBuffer();
    bytes = new Uint8Array(buf);
  } catch (err) {
    if (mine !== seq) return;
    show([]);
    setStatus(T.cantReadThat + errText(err), 'fail');
    return;
  }
  if (mine !== seq) return;

  try {
    await engine();
  } catch (err) {
    if (mine !== seq) return;
    show([]);
    setStatus(T.noEngine, 'fail');
    return;
  }
  if (mine !== seq) return;

  var type;
  try {
    type = detect_file_type(bytes);
  } catch (err) {
    if (mine !== seq) return;
    show([]);
    setStatus(T.cantRead + errText(err), 'fail');
    return;
  }
  if (mine !== seq) return;

  if (type === 'unknown') {
    show([]);
    setStatus(T.unsupported(SUPPORTED), 'fail');
    return;
  }
  if (type === 'office-legacy') {
    show([]);
    setStatus(T.legacy, 'fail');
    return;
  }

  var res;
  try {
    res = analyze_file(bytes);
  } catch (err) {
    if (mine !== seq) return;
    show([]);
    setStatus(T.cantRead + errText(err), 'fail');
    return;
  }
  if (mine !== seq) return;

  var found = groupsOf(res && res.metadata);
  var rows = [row(T.file, T.fileDesc, 'info', T.info, [asText(file.name), type.toUpperCase() + ', ' + fmtSize(file.size)])];
  if (found.total) {
    found.groups.forEach(function (g) {
      var lines = g.items.map(function (it) {
        return (asText(it.name) || T.value) + ': ' + truncate(it.value);
      });
      rows.push(row(g.name, '', 'warn', T.found, lines, T.foundWhy));
    });
    setStatus(T.foundSummary(found.total, type.toUpperCase()), 'warn');
  } else {
    rows.push(row(T.metadata, T.metadataDesc, 'pass', T.none, [T.noMeta]));
    setStatus(T.noneSummary(type.toUpperCase()), 'ok');
  }
  show(rows);
  current = { file: file, bytes: bytes, type: type };
  cleanBtn.hidden = false;
  cleanBtn.disabled = false;
  cleanBtn.textContent = BUTTON;
}

async function runClean() {
  if (!current) return;
  var job = current;
  cleanBtn.disabled = true;
  cleanBtn.textContent = T.working;

  var res;
  try {
    await engine();
    res = process_file(job.bytes);
  } catch (err) {
    setStatus(T.cantClean + errText(err), 'fail');
    cleanBtn.disabled = false;
    cleanBtn.textContent = BUTTON;
    return;
  }

  var cleaned = res && res.cleaned;
  if (!cleaned || typeof cleaned.length !== 'number' || !cleaned.length) {
    setStatus(T.noData, 'fail');
    cleanBtn.disabled = false;
    cleanBtn.textContent = BUTTON;
    return;
  }

  var mime = MIME[job.type] || job.file.type || 'application/octet-stream';
  try {
    download(cleaned, cleanName(job.file.name), mime);
  } catch (err) {
    setStatus(T.cantSave + errText(err), 'fail');
    cleanBtn.disabled = false;
    cleanBtn.textContent = BUTTON;
    return;
  }

  var left = (res.verification && res.verification.metadata_found) || [];
  if (!left.length) {
    setStatus(T.clean, 'ok');
  } else {
    var names = left.map(function (it) {
      var cat = asText(it && it.category);
      var nm = asText(it && it.name) || T.value.toLowerCase();
      return (cat ? cat + ' / ' : '') + nm;
    });
    setStatus(T.left(left.length, names.join(T.listSep)), 'warn');
  }
  cleanBtn.disabled = false;
  cleanBtn.textContent = BUTTON;
}
