// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian.dns

import java.util.concurrent.atomic.AtomicLong

class Stats {
    val queries = AtomicLong()
    val blocked = AtomicLong()
    val forwarded = AtomicLong()
    val answered = AtomicLong()
    val failed = AtomicLong()
    val malformed = AtomicLong()

    fun snapshot(): StatsSnapshot = StatsSnapshot(
        queries.get(), blocked.get(), forwarded.get(),
        answered.get(), failed.get(), malformed.get()
    )
}

data class StatsSnapshot(
    val queries: Long,
    val blocked: Long,
    val forwarded: Long,
    val answered: Long,
    val failed: Long,
    val malformed: Long
)

data class LogEntry(
    val timeMs: Long,
    val name: String,
    val qtype: Int,
    val blocked: Boolean,
    val uid: Int = -1,
    val paused: Boolean = false,
    /** G14: the blocked CNAME target when the answer, not the query name, was blocked. */
    val via: String? = null
)

sealed interface Action {
    data class Respond(val packet: ByteArray, val name: String) : Action
    class Forward(val query: DnsQuery, val payload: ByteArray) : Action
    object Drop : Action
}

class QueryHandler(
    @Volatile var blocklist: Blocklist,
    val mode: BlockMode = BlockMode.NULL_IP,
    private val uidOf: (DnsQuery) -> Int = { -1 },
    private val clock: () -> Long = System::currentTimeMillis
) {
    val stats = Stats()
    val tally = AppTally()

    // Pause is set from the service thread and read from every forwarding thread; @Volatile keeps the
    // read/write visible without a lock. The absolute timestamp makes "until" survive clock reads
    // (the field is not a countdown).
    @Volatile var pausedUntilMs: Long = 0

    private val logCapacity = 200
    private val logEntries = arrayOfNulls<LogEntry>(logCapacity)
    private var logWritePos = 0
    private var logCount = 0
    private val logLock = Any()

    fun pause(durationMs: Long) {
        if (durationMs <= 0L) {
            resume()
            return
        }
        pausedUntilMs = clock() + durationMs
    }

    fun resume() {
        pausedUntilMs = 0L
    }

    fun isPaused(): Boolean = clock() < pausedUntilMs

    fun handle(packet: ByteArray, length: Int): Action {
        stats.queries.incrementAndGet()
        val q = DnsPacket.parseQuery(packet, length)
        if (q == null) {
            stats.malformed.incrementAndGet()
            return Action.Drop
        }
        val uid = try { uidOf(q) } catch (_: Throwable) { -1 }
        val hitBlocklist = blocklist.isBlocked(q.qname)
        val paused = isPaused()
        if (paused) {
            // Paused: every well-formed query is forwarded. A query that would have been blocked is
            // left out of the tally entirely (its "blocked" bookkeeping would otherwise keep rising
            // while blocking is suspended); ordinary allowed queries still count toward queries.
            if (!hitBlocklist) tally.record(uid, q.qname, false)
            stats.forwarded.incrementAndGet()
            val payload = DnsWire.extractQueryPayload(packet, q)
            appendLog(q.qname, q.qtype, blocked = false, uid = uid, paused = true)
            return Action.Forward(q, payload)
        }
        tally.record(uid, q.qname, hitBlocklist)
        if (hitBlocklist) {
            stats.blocked.incrementAndGet()
            val resp = DnsPacket.buildBlockedResponse(q, mode)
            appendLog(q.qname, q.qtype, blocked = true, uid = uid, paused = false)
            return Action.Respond(resp, q.qname)
        }
        stats.forwarded.incrementAndGet()
        val payload = DnsWire.extractQueryPayload(packet, q)
        appendLog(q.qname, q.qtype, blocked = false, uid = uid, paused = false)
        return Action.Forward(q, payload)
    }

    fun onUpstreamAnswer(q: DnsQuery, answer: ByteArray, len: Int): ByteArray? {
        val resp = DnsPacket.buildResponseFromUpstream(q, answer, len)
        if (resp == null) {
            stats.failed.incrementAndGet()
            return null
        }
        cloakedTarget(q, answer, len)?.let { target ->
            stats.blocked.incrementAndGet()
            appendLog(q.qname, q.qtype, blocked = true, uid = -1, paused = false, via = target)
            return DnsPacket.buildBlockedResponse(q, mode)
        }
        stats.answered.incrementAndGet()
        return resp
    }

    /**
     * G14, CNAME uncloaking: the first CNAME target in the answer that the current rules block, or
     * null. Not applied while paused, nor when the user explicitly allowed the query name itself
     * (their allow wins, as everywhere else). A user allow on the target makes match() say "allow",
     * so that target passes too.
     */
    private fun cloakedTarget(q: DnsQuery, answer: ByteArray, len: Int): String? {
        if (isPaused()) return null
        val bl = blocklist
        if (bl.match(q.qname)?.source == Blocklist.RuleSource.USER_ALLOW) return null
        for (t in DnsAnswer.cnameTargets(answer, len)) {
            if (bl.match(t)?.blocks == true) return t
        }
        return null
    }

    fun onUpstreamFailed(q: DnsQuery) {
        stats.failed.incrementAndGet()
    }

    fun snapshot(): StatsSnapshot = stats.snapshot()

    fun recent(): List<LogEntry> {
        synchronized(logLock) {
            val result = ArrayList<LogEntry>(logCount)
            for (i in 1..logCount) {
                val idx = (logWritePos - i + logCapacity) % logCapacity
                result.add(logEntries[idx]!!)
            }
            return result
        }
    }

    fun clearLog() {
        synchronized(logLock) {
            for (i in logEntries.indices) logEntries[i] = null
            logWritePos = 0
            logCount = 0
        }
        tally.clear()
    }

    private fun appendLog(name: String, qtype: Int, blocked: Boolean, uid: Int, paused: Boolean, via: String? = null) {
        val entry = LogEntry(clock(), name, qtype, blocked, uid, paused, via)
        synchronized(logLock) {
            logEntries[logWritePos] = entry
            logWritePos = (logWritePos + 1) % logCapacity
            if (logCount < logCapacity) logCount++
        }
    }
}

// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian.dns

/**
 * Domain block/allow matcher.
 *
 * Four rule sets are kept separately so that precedence is explicit and stable:
 *
 *   user allow  >  user block  >  builtin allow (@@)  >  builtin block
 *
 * The first three are consulted in that order, walking the query name and each of its parent
 * domains one label at a time. The first hit decides the answer. This means:
 *
 *  - a user allow always wins, even over a user block of the same name or a builtin block;
 *  - a user block beats a builtin @@ allow, which is the point of letting users blacklist a
 *    domain whose builtin whitelisting they disagree with;
 *  - builtin @@ still overrides builtin block, as before.
 *
 * The bundled lists are immutable per process; the user lists are applied with plusAllowed /
 * plusBlocked, both of which return a new instance and leave the receiver untouched.
 *
 * matched by suffix: "example.com" covers "a.b.example.com" but not "badexample.com".
 */
class Blocklist private constructor(
    private val builtinBlocked: HashSet<String>,
    private val builtinAllowed: HashSet<String>,
    private val userBlocked: HashSet<String>,
    private val userAllowed: HashSet<String>
) {
    /** Number of domains in the bundled block list (not counting user blocks). */
    val blockedCount: Int get() = builtinBlocked.size

    /** Number of builtin @@ allow rules (not counting user allows). */
    val allowedCount: Int get() = builtinAllowed.size

    /** Which of the four rule sets a [Match] came from. */
    enum class RuleSource { USER_ALLOW, USER_BLOCK, BUILTIN_ALLOW, BUILTIN_BLOCK }

    /**
     * The rule that decided a name: [rule] is the stored entry (the name itself or one of its
     * parent domains) and [source] the set it lives in.
     */
    data class Match(val rule: String, val source: RuleSource) {
        val blocks: Boolean get() = source == RuleSource.USER_BLOCK || source == RuleSource.BUILTIN_BLOCK
    }

    /**
     * Returns the first rule that applies to [name], using the precedence and suffix walk described
     * on the class, or null when no rule applies (or the name is not a valid hostname).
     * [isBlocked] is defined in terms of this function so the two can never disagree.
     */
    fun match(name: String): Match? {
        try {
            val n = normalizeHostname(name) ?: return null
            walk(n, userAllowed)?.let { return Match(it, RuleSource.USER_ALLOW) }
            walk(n, userBlocked)?.let { return Match(it, RuleSource.USER_BLOCK) }
            walk(n, builtinAllowed)?.let { return Match(it, RuleSource.BUILTIN_ALLOW) }
            walk(n, builtinBlocked)?.let { return Match(it, RuleSource.BUILTIN_BLOCK) }
            return null
        } catch (t: Throwable) {
            return null
        }
    }

    /** Called for every DNS query; allocates at most one Match. */
    fun isBlocked(name: String): Boolean = match(name)?.blocks ?: false

    /** [n] or the nearest parent domain of [n] contained in [set], else null. */
    private fun walk(n: String, set: HashSet<String>): String? {
        if (set.isEmpty()) return null
        var cur = n
        while (true) {
            if (set.contains(cur)) return cur
            val dot = cur.indexOf('.')
            if (dot < 0) return null
            cur = cur.substring(dot + 1)
        }
    }

    /**
     * Returns a new Blocklist with [extra] added to the user allow set. Invalid entries are ignored.
     * User allow is the highest-priority set, so a user allow overrides both user blocks and the
     * builtin lists.
     */
    fun plusAllowed(extra: Collection<String>): Blocklist {
        val newAllowed = HashSet<String>(userAllowed)
        for (s in extra) {
            val n = normalizeOrNull(s) ?: continue
            newAllowed.add(n)
        }
        return Blocklist(HashSet(builtinBlocked), HashSet(builtinAllowed), HashSet(userBlocked), newAllowed)
    }

    /**
     * Returns a new Blocklist with [extra] added to the user block set. Invalid entries are ignored.
     * User block is consulted after user allow but before the builtin lists, so it overrides
     * builtin @@ rules while still losing to an explicit user allow of the same name.
     */
    fun plusBlocked(extra: Collection<String>): Blocklist {
        val newBlocked = HashSet<String>(userBlocked)
        for (s in extra) {
            val n = normalizeOrNull(s) ?: continue
            newBlocked.add(n)
        }
        return Blocklist(HashSet(builtinBlocked), HashSet(builtinAllowed), newBlocked, HashSet(userAllowed))
    }

    companion object {
        private val WS = Regex("\\s+")

        private val IGNORED = setOf(
            "localhost", "localhost.localdomain", "broadcasthost", "local",
            "0.0.0.0", "127.0.0.1", "ip6-localhost", "ip6-loopback"
        )

        fun normalizeOrNull(s: String): String? = normalizeHostname(s)

        fun parse(text: String): Blocklist {
            val builtinBlocked = HashSet<String>(1024)
            val builtinAllowed = HashSet<String>(64)

            for (rawLine in text.lineSequence()) {
                var line = rawLine.trim()
                if (line.isEmpty()) continue
                if (line.startsWith("#") || line.startsWith("!")) continue

                val hash = line.indexOf('#')
                if (hash >= 0) line = line.substring(0, hash).trim()
                if (line.isEmpty()) continue

                var ascii = true
                for (ch in line) {
                    if (ch.code >= 128) { ascii = false; break }
                }
                if (!ascii) continue

                if (line.startsWith("@@")) {
                    val domain = parseAdblock(line.substring(2))
                    if (domain != null) builtinAllowed.add(domain)
                    continue
                }

                if (line.startsWith("||")) {
                    val domain = parseAdblock(line)
                    if (domain != null) builtinBlocked.add(domain)
                    continue
                }

                val tokens = line.split(WS)
                if (tokens.size >= 2 && looksLikeIp(tokens[0])) {
                    for (k in 1 until tokens.size) {
                        val d = normalizeHostname(tokens[k]) ?: continue
                        if (d in IGNORED) continue
                        builtinBlocked.add(d)
                    }
                    continue
                }

                val d = normalizeHostname(line) ?: continue
                if (d in IGNORED) continue
                builtinBlocked.add(d)
            }

            return Blocklist(builtinBlocked, builtinAllowed, HashSet(16), HashSet(16))
        }

        private fun parseAdblock(s: String): String? {
            if (!s.startsWith("||")) return null
            var body = s.substring(2)
            if (!body.endsWith("^")) return null
            body = body.substring(0, body.length - 1)
            if (body.contains('*') || body.contains('/') || body.contains('$') || body.contains('|')) return null
            return normalizeHostname(body)
        }

        private fun looksLikeIp(s: String): Boolean {
            val parts = s.split('.')
            if (parts.size != 4) return false
            for (p in parts) {
                if (p.isEmpty()) return false
                for (ch in p) if (!ch.isDigit()) return false
                val v = p.toIntOrNull() ?: return false
                if (v < 0 || v > 255) return false
            }
            return true
        }

        private fun normalizeHostname(s: String): String? {
            val d = s.trim().lowercase().trimEnd('.')
            if (d.isEmpty()) return null
            if (d.length > 253) return null
            if (!d.contains('.')) return null
            if (d.contains("..")) return null
            for (label in d.split('.')) {
                if (label.isEmpty()) return null
                if (label.length > 63) return null
                if (label.startsWith('-') || label.endsWith('-')) return null
                for (ch in label) {
                    val ok = (ch in 'a'..'z') || (ch in '0'..'9') || ch == '-' || ch == '_'
                    if (!ok) return null
                }
            }
            return d
        }
    }
}

class GuardianVpnService : VpnService() {

    private var tun: ParcelFileDescriptor? = null
    private var tunIn: FileInputStream? = null
    private var tunOut: FileOutputStream? = null
    private val writeLock = Any()
    private var readThread: Thread? = null
    private var pool: ExecutorService? = null
    @Volatile private var running = false
    private var handler: QueryHandler? = null
    private var upstream: Upstream? = null

    // Pause expires on its own (QueryHandler.isPaused() returns false once the absolute timestamp
    // passes), so the notification needs a one-shot timer that rewrites it back to "DNS filtering
    // active" when the pause ends. Posted on the main looper; cancelled on resume or stop.
    private val mainHandler = Handler(Looper.getMainLooper())
    @Volatile private var notifRefresh: Runnable? = null

    // G12: durable per-day block counts live in StatsStore. The running blocked counter is a Long
    // in QueryHandler and resets on every service start; every STATS_FLUSH_MS we take the delta
    // since the last flush and record it. A process kill between flushes costs at most one interval
    // of counts, and a long session only writes on the flush tick, never once per query.
    @Volatile private var lastFlushedBlocked: Long = 0L
    @Volatile private var statsFlush: Runnable? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> {
                stopVpnInternal()
                stopSelf()
                return START_NOT_STICKY
            }
            ACTION_PAUSE -> {
                val h = handler
                if (h != null) {
                    h.pause(PAUSE_DURATION_MS)
                    updateNotification()
                    scheduleNotifRefresh(PAUSE_DURATION_MS)
                    try {
                        android.service.quicksettings.TileService.requestListeningState(
                            applicationContext,
                            android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                        )
                    } catch (_: Throwable) {}
                }
                return START_STICKY
            }
            ACTION_RESUME -> {
                val h = handler
                if (h != null) {
                    h.resume()
                    updateNotification()
                    cancelNotifRefresh()
                    try {
                        android.service.quicksettings.TileService.requestListeningState(
                            applicationContext,
                            android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                        )
                    } catch (_: Throwable) {}
                }
                return START_STICKY
            }
            else -> {
                // Besides our own Start, two ways to get here: the system starts us for Always-on
                // VPN (action "android.net.VpnService"), or START_STICKY restarts us with a null
                // intent. Consent may have been revoked since; without it establish() would fail,
                // so stop and do not ask to be restarted.
                if (VpnService.prepare(this) != null) {
                    stopVpnInternal()
                    stopSelf()
                    return START_NOT_STICKY
                }
                if (!running && !isLoading) {
                    // startForeground can be refused (e.g. systemExempted on API 34+ when this app
                    // is no longer the configured VPN after a sticky restart). Stop rather than run unpromoted.
                    val promoted = try { startForegroundSafe(); true } catch (_: RuntimeException) { false }
                    if (!promoted) {
                        stopSelf()
                        return START_NOT_STICKY
                    }
                    isLoading = true
                    val gen = ++startGeneration
                    Thread { startVpnBackground(gen) }.apply { name = "guardian-start" }.start()
                }
            }
        }
        return START_STICKY
    }

    // Bumped by every start and stop. A start thread that finds a newer generation after loading the
    // list gives up, so Stop pressed while the list is loading cannot be overtaken by a late establish().
    @Volatile private var startGeneration = 0

    private fun startVpnBackground(gen: Int) {
        try {
            val blocklist = loadBlocklist()
            val effective = applyUserRulesTo(blocklist, applicationContext)
            // No uid lookup: on Android 10+ every query reaching the VPN is sent by the system DNS
            // service, so getConnectionOwnerUid names that service, not the app (seen in the emulator flow).
            val h = QueryHandler(effective, BlockMode.NULL_IP)
            if (gen != startGeneration) return

            val builder = Builder()
            builder.setSession("Guardian")
            // IPv4 DNS hijack target. Only DNS-server traffic is routed into the VPN; everything else
            // stays on the underlying network.
            builder.addAddress("10.111.222.1", 32)
            builder.addDnsServer("10.111.222.2")
            builder.addRoute("10.111.222.2", 32)
            // IPv6 DNS hijack target, a unique local address (fc00::/7, not routable on the internet). Some networks only
            // hand out IPv6 DNS servers; without this, those queries bypass Guardian. If the platform
            // refuses the IPv6 address (old build, unusual ROM), fall back to IPv4-only rather than
            // failing the whole session.
            var ipv6Configured = false
            try {
                builder.addAddress("fd00:6a72:6764::1", 128)
                ipv6Configured = true
            } catch (_: Throwable) {
                ipv6Configured = false
            }
            if (ipv6Configured) {
                try {
                    builder.addDnsServer("fd00:6a72:6764::2")
                    builder.addRoute("fd00:6a72:6764::2", 128)
                } catch (_: Throwable) {
                    // Address already added; a partially configured IPv6 side is harmless: no route
                    // is installed, so no IPv6 DNS traffic will enter the VPN from this side.
                }
            }
            builder.setMtu(1500)
            builder.setBlocking(true)

            val pfd = try { builder.establish() } catch (_: Throwable) { null }
            if (gen != startGeneration) {
                try { pfd?.close() } catch (_: Throwable) {}
                return
            }
            if (pfd == null) {
                isLoading = false
                stopSelf()
                return
            }
            tun = pfd
            tunIn = FileInputStream(pfd.fileDescriptor)
            tunOut = FileOutputStream(pfd.fileDescriptor)

            val up = Upstream(applicationContext) { sock -> protect(sock) }
            up.start()
            upstream = up

            handler = h
            activeHandler = h
            pool = Executors.newFixedThreadPool(8)
            running = true
            isRunning = true
            isLoading = false
            instance = this
            readAlwaysOn()
            // Let the Quick Settings tile refresh from "Starting…" to "On" without waiting for a tap.
            // Fully qualified to avoid touching this file's imports.
            try {
                android.service.quicksettings.TileService.requestListeningState(
                    applicationContext,
                    android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                )
            } catch (_: Throwable) {}

            // G12: baseline the running counter, then start the periodic flush.
            lastFlushedBlocked = 0L
            scheduleStatsFlush()

            val t = Thread { readLoop(pfd) }
            t.name = "guardian-tun-read"
            readThread = t
            t.start()
        } catch (_: Throwable) {
            isLoading = false
            stopSelf()
        }
    }

    }

    private fun buildNotification(): Notification {

    private fun updateNotification() {

            nm.notify(NOTIF_ID, buildNotification())

    }

    private fun scheduleNotifRefresh(delayMs: Long) {
        cancelNotifRefresh()
        val r = Runnable {
            notifRefresh = null
            updateNotification()
        }
        notifRefresh = r
        mainHandler.postDelayed(r, delayMs)
    }

    private fun cancelNotifRefresh() {
        notifRefresh?.let { mainHandler.removeCallbacks(it) }
        notifRefresh = null
    }

    // G12: every STATS_FLUSH_MS write the delta since the last flush into StatsStore, then re-arm.
    // The Runnable re-schedules itself rather than using a fixed-rate executor so the interval
    // survives the main looper being busy. The first tick is a full interval after start.
    private fun scheduleStatsFlush() {
        cancelStatsFlush()
        val r = Runnable {
            statsFlush = null
            flushStatsNow()
            scheduleStatsFlush()
        }
        statsFlush = r
        mainHandler.postDelayed(r, STATS_FLUSH_MS)
    }

    private fun cancelStatsFlush() {
        statsFlush?.let { mainHandler.removeCallbacks(it) }
        statsFlush = null
    }

    // Snapshot the running counter, record the difference from the last flush, and remember the new
    // baseline. Reading the snapshot first, then moving the baseline, is safe: any block that
    // happens between the two steps is still counted in the next flush's delta, never double counted.
    private fun flushStatsNow() {
        val h = handler ?: return
        val cur = h.snapshot().blocked
        val prev = lastFlushedBlocked
        if (cur <= prev) return
        val delta = cur - prev
        lastFlushedBlocked = cur
        try {
            StatsStore.record(
                applicationContext,
                if (delta > Int.MAX_VALUE.toLong()) Int.MAX_VALUE else delta.toInt()
            )
        } catch (_: Throwable) {}
    }

    private fun startForegroundSafe() {
        val notif = buildNotification()
        if (Build.VERSION.SDK_INT >= 34) {
            startForeground(NOTIF_ID, notif, ServiceInfo.FOREGROUND_SERVICE_TYPE_SYSTEM_EXEMPTED)
        } else if (Build.VERSION.SDK_INT >= 29) {
            startForeground(NOTIF_ID, notif, 0)
        } else {
            startForeground(NOTIF_ID, notif)
        }
    }

    private fun readLoop(pfd: ParcelFileDescriptor) {
        val buf = ByteArray(32767)
        val input = tunIn ?: return
        while (running) {
            val n = try { input.read(buf) } catch (_: Throwable) { if (!running) break else continue }
            if (n <= 0) continue
            val copy = buf.copyOf(n)
            val exec = pool ?: break
            try { exec.execute { processPacket(copy) } } catch (_: Throwable) {}
        }
    }

    private fun processPacket(packet: ByteArray) {
        val h = handler ?: return
        when (val act = h.handle(packet, packet.size)) {
            is Action.Respond -> writeTun(act.packet)
            is Action.Forward -> {
                val exec = pool ?: return
                try {
                    exec.execute {
                        val ans = upstream?.sendQuery(act.payload)
                        if (ans == null) {
                            h.onUpstreamFailed(act.query)
                        } else {
                            val resp = h.onUpstreamAnswer(act.query, ans, ans.size)
                            if (resp != null) writeTun(resp)
                        }
                    }
                } catch (_: Throwable) {}
            }
            Action.Drop -> {}
        }
    }

    private fun writeTun(packet: ByteArray) {
        val out = tunOut ?: return
        synchronized(writeLock) {
            try { out.write(packet) } catch (_: Throwable) {}
        }
    }

    private fun stopVpnInternal() {
        cancelNotifRefresh()
        // G12: capture whatever was blocked in the last interval before the handler goes away.
        cancelStatsFlush()
        flushStatsNow()
        startGeneration++
        running = false
        isRunning = false
        isLoading = false
        instance = null
        alwaysOn = null
        lockdown = null
        try { tun?.close() } catch (_: Throwable) {}
        try { tunIn?.close() } catch (_: Throwable) {}
        try { tunOut?.close() } catch (_: Throwable) {}
        tun = null; tunIn = null; tunOut = null
        try { readThread?.join(1000) } catch (_: Throwable) {}
        readThread = null
        try { pool?.shutdownNow() } catch (_: Throwable) {}
        pool = null
        try { upstream?.stop() } catch (_: Throwable) {}
        upstream = null
        handler = null
        activeHandler = null

        // Let the Quick Settings tile flip back to "Off" without waiting for a tap.
        try {
            android.service.quicksettings.TileService.requestListeningState(
                applicationContext,
                android.content.ComponentName(applicationContext, GuardianTileService::class.java)
            )
        } catch (_: Throwable) {}
    }

    private fun loadBlocklist(): Blocklist = loadBlocklistStatic(applicationContext)

    companion object {
        const val ACTION_START = "org.secureinchat.guardian.action.START"
        const val ACTION_STOP = "org.secureinchat.guardian.action.STOP"
        const val ACTION_PAUSE = "org.secureinchat.guardian.action.PAUSE"
        const val ACTION_RESUME = "org.secureinchat.guardian.action.RESUME"
        const val PAUSE_DURATION_MS = 5 * 60 * 1000L
        private const val STATS_FLUSH_MS = 60_000L
        private const val CHANNEL_ID = "guardian_vpn"
        private const val NOTIF_ID = 1
        private val LIST_ASSETS = listOf("blocklist-hagezi-light.txt", "blocklist-extra.txt")
        private val rulesLock = Any()

        @Volatile var isRunning: Boolean = false
            private set

        @Volatile var isLoading: Boolean = false
            private set

        @Volatile var activeHandler: QueryHandler? = null
            private set

        // null = unknown (not running, or API < 29). The UI hides the row on null.
        @Volatile var alwaysOn: Boolean? = null
            private set
        @Volatile var lockdown: Boolean? = null
            private set
        @Volatile private var instance: GuardianVpnService? = null

        /** Re-reads the Always-on state, e.g. when the user comes back from system settings. */
        fun refreshAlwaysOn() {
            instance?.readAlwaysOn()
        }

        // The bundled lists never change while the app runs, so parse them once per process.
        @Volatile private var baseCache: Blocklist? = null

        private fun loadBlocklistStatic(ctx: Context): Blocklist {
            baseCache?.let { return it }
            return try {
                val text = LIST_ASSETS.joinToString("\n") { name ->
                    ctx.assets.open(name).use { it.bufferedReader().readText() }
                }
                Blocklist.parse(text).also { baseCache = it }
            } catch (_: Throwable) {
                Blocklist.parse("")
            }
        }

        // Bundled list + user blocks + user allows. Precedence inside Blocklist is
        // "user allow > user block > builtin allow > builtin block", so the order of the two calls
        // below only matters for building the object, not for matching.
        private fun applyUserRulesTo(base: Blocklist, ctx: Context): Blocklist {
            var eff = base
            val userBlocked = UserRulesStore.blocked(ctx)
            if (userBlocked.isNotEmpty()) eff = eff.plusBlocked(userBlocked)
            val userAllowed = UserRulesStore.allowed(ctx)
            if (userAllowed.isNotEmpty()) eff = eff.plusAllowed(userAllowed)
            return eff
        }

        fun applyUserRules(ctx: Context) {
            val h = activeHandler ?: return
            Thread {
                synchronized(rulesLock) {
                    try {
                        val base = loadBlocklistStatic(ctx)
                        h.blocklist = applyUserRulesTo(base, ctx)
                    } catch (_: Throwable) {}
                }
            }.apply { name = "guardian-rules" }.start()
        }
    }
}

    private fun handleExportResult(uri: Uri?) {
        if (uri == null) return
        try {
            val text = RulesText.format(
                UserRulesStore.blocked(this),
                UserRulesStore.allowed(this)
            )
            // "wt" truncates, so re-exporting over an existing file does not leave trailing bytes.
            contentResolver.openOutputStream(uri, "wt").use { out ->
                if (out == null) throw IllegalStateException("no output stream")
                out.write(text.toByteArray(Charsets.UTF_8))
            }
            rulesIoError.visibility = View.GONE
        } catch (_: Throwable) {
            rulesIoError.visibility = View.VISIBLE
            rulesIoError.text = txt("Could not write the file", "无法写入文件")
        }
    }

    private fun handleImportResult(uri: Uri?) {
        if (uri == null) return
        // Read at most MAX_BYTES so a wrong pick (a video, an APK) cannot tie up the main thread.
        val text: String? = try {
            contentResolver.openInputStream(uri)?.use { input ->
                val buf = ByteArray(RulesText.MAX_BYTES)
                var total = 0
                while (total < buf.size) {
                    val n = input.read(buf, total, buf.size - total)
                    if (n <= 0) break
                    total += n
                }
                String(buf, 0, total, Charsets.UTF_8)
            }
        } catch (_: Throwable) {
            null
        }
        if (text == null) {
            rulesIoError.visibility = View.VISIBLE
            rulesIoError.text = txt("Could not read the file", "无法读取文件")
            return
        }
        val parsed = RulesText.parse(text)
        val summary = txt(
            "${parsed.blocked.size} to block, ${parsed.allowed.size} to allow, ${parsed.skipped} lines skipped",
            "拦截 ${parsed.blocked.size} 条,放行 ${parsed.allowed.size} 条,跳过 ${parsed.skipped} 行"
        )
        // Three-way choice. Merge keeps the existing rules and lets imported entries replace the
        // current choice for a domain; Replace discards the current lists entirely.
        AlertDialog.Builder(this)
            .setTitle(txt("Import rules", "导入规则"))
            .setMessage(summary)
            .setPositiveButton(txt("Merge", "合并")) { _, _ -> applyImport(parsed, merge = true) }
            .setNeutralButton(txt("Replace", "替换")) { _, _ -> applyImport(parsed, merge = false) }
            .setNegativeButton(txt("Cancel", "取消"), null)
            .show()
    }

    private fun applyImport(parsed: RulesText.ParseResult, merge: Boolean) {
        if (merge) {
            UserRulesStore.mergeIn(this, parsed.blocked, parsed.allowed)
        } else {
            UserRulesStore.replaceAll(this, parsed.blocked, parsed.allowed)
        }
        GuardianVpnService.applyUserRules(applicationContext)
        // Clear signatures so both lists rebuild on the next refresh even inside the throttle window.
        blockedSignature = ""
        allowedSignature = ""
        appListSignature = ""
        rulesIoError.visibility = View.GONE
        uiHandler.post { refreshUi() }
    }

        var text = if (on) txt("Always-on: On", "始终开启:已开启") else txt("Always-on: Off", "始终开启:未开启")

    private fun onToggleClicked() {
        if (GuardianVpnService.isLoading) return
        if (GuardianVpnService.isRunning) {
            // Plain startService: the stop path never calls startForeground, and a
            // startForegroundService() call that is not followed by startForeground() within a
            // few seconds crashes with ForegroundServiceDidNotStartInTimeException.
            val i = Intent(this, GuardianVpnService::class.java).setAction(GuardianVpnService.ACTION_STOP)
            startService(i)
        } else {
            val prep = VpnService.prepare(this)
            if (prep != null) {
                @Suppress("DEPRECATION")
                startActivityForResult(prep, REQ_VPN_PREPARE)
            } else {
                startVpnService()
            }
        }
    }

    private fun onPauseResumeClicked() {
        val h = GuardianVpnService.activeHandler ?: return
        val action = if (h.isPaused()) GuardianVpnService.ACTION_RESUME else GuardianVpnService.ACTION_PAUSE
        val i = Intent(this, GuardianVpnService::class.java).setAction(action)
        startService(i)
        // The service handles the action asynchronously; refresh soon so the button/status catch up.
        uiHandler.postDelayed({ refreshUi() }, 200)
    }

    // G12: cheap to read, so it is refreshed every tick alongside the in-session stats. The
    // signature check keeps a setText off the common path when the numbers have not changed.
    private fun renderHistory() {
        val today = StatsStore.today(this)
        val last7 = StatsStore.last7(this)
        val sig = "$today/$last7"
        if (sig == historySignature) return
        historySignature = sig
        setIfChanged(historyView, txt(

    private fun renderLog(entries: List<LogEntry>) {
        val sb = SpannableStringBuilder()
        val fmt = SimpleDateFormat("HH:mm:ss", Locale.getDefault())
        // Last 10 only: a glance at what is happening; the per-domain view below is what people act on.
        for (e in entries.take(10)) {
            val start = sb.length
            val tag = when {
                e.blocked -> txt("BLOCK", "拦截")
                e.paused -> txt("PAUSE", "暂停")
                else -> txt("ALLOW", "放行")
            }
            sb.append(fmt.format(Date(e.timeMs)))
                .append(' ').append(tag).append(' ')
                .append(e.name).append(" (type ").append(e.qtype.toString()).append(")")
            // G14: blocked because the answer's CNAME pointed at a listed tracker.
            e.via?.let { sb.append(txt(" → really $it", " → 实际指向 $it")) }
            sb.append('\n')
            if (e.blocked) {
                sb.setSpan(
                    ForegroundColorSpan(0xFFFF4444.toInt()),
                    start, sb.length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE
                )
            } else if (e.paused) {
                sb.setSpan(
                    ForegroundColorSpan(0xFFFFA500.toInt()),
                    start, sb.length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE
                )
            }
        }
        setIfChanged(logView, sb)
    }

    // "Blocked most often": the domains blocked in this session, each with an Allow button for false
    // positives. Not per app: on Android 10+ apps resolve names through the system DNS service, so every
    // query reaching the VPN comes from that service's uid (1051 on the emulator), not from the app.
    private fun renderAppList(h: QueryHandler?) {
        val top = h?.tally?.snapshot(1)?.firstOrNull()?.topBlocked ?: emptyList()
        // G11: the rule behind each row, worked out now from the live rules (the log does not store
        // it). It is part of the signature so a rule change (Allow, Block, import) rebuilds the rows
        // once the background rules thread has swapped the list, not only when counts change.
        val bl = h?.blocklist
        val rules = top.map { bl?.match(it.first) }
        val sig = top.joinToString(";") { "${it.first}:${it.second}" } + "|" +
            rules.joinToString(";") { m -> if (m == null) "-" else "${m.rule}:${m.source}" }
        if (sig == appListSignature) return
        // Counts change with every DNS query. Rebuild the list at most every 5 s unless a tap reset the
        // signature, so background traffic does not re-lay out the screen every second.
        val now = android.os.SystemClock.uptimeMillis()
        if (appListSignature.isNotEmpty() && now - appListBuiltAt < 5000) return
        appListBuiltAt = now
        appListSignature = sig
        appListContainer.removeAllViews()
        if (h == null) return
        val density = resources.displayMetrics.density
        val vpad = (4 * density).toInt()
        if (top.isEmpty()) {
            appListContainer.addView(TextView(this).apply {
                textSize = 12f
                text = txt("(nothing blocked yet)", "(还没有拦截)")
                setPadding(0, vpad, 0, vpad)
            })
            return
        }
        val secondary = secondaryTextColors()
        for ((i, d) in top.withIndex()) {
            val domRow = LinearLayout(this).apply {
                orientation = LinearLayout.HORIZONTAL
                setPadding(0, vpad, 0, vpad)
            }
            // Domain on the first line, the rule that blocked it in small grey text below.
            val texts = LinearLayout(this).apply {
                orientation = LinearLayout.VERTICAL
                layoutParams = LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f)
            }
            val dom = TextView(this).apply {
                textSize = 13f
                text = "${d.first} (${d.second})"
            }
            texts.addView(dom)
            texts.addView(TextView(this).apply {
                textSize = 11f
                if (secondary != null) setTextColor(secondary)
                text = ruleLine(rules[i])
            })
            val allowBtn = Button(this).apply {
                text = txt("Allow", "放行")
                textSize = 12f
                setOnClickListener {
                    UserRulesStore.add(this@MainActivity, d.first)
                    GuardianVpnService.applyUserRules(applicationContext)
                    appListSignature = ""
                    blockedSignature = ""
                    allowedSignature = ""
                    uiHandler.post { refreshUi() }
                }
            }
            domRow.addView(texts)
            domRow.addView(allowBtn)
            appListContainer.addView(domRow)
        }
    }

    /**
     * Text for the rule line under a "Blocked most often" row. The list counts past blocks, so the
     * current rules may no longer block the name (the user allowed it, or removed their own block);
     * say so rather than showing a rule that no longer applies.
     */
    private fun ruleLine(m: Blocklist.Match?): String = if (m == null) {
        txt("No longer blocked", "现在不再拦截")
    } else when (m.source) {
        Blocklist.RuleSource.BUILTIN_BLOCK -> txt("Rule: ${m.rule} (built-in list)", "规则:${m.rule}(内置名单)")
        Blocklist.RuleSource.USER_BLOCK -> txt("Rule: ${m.rule} (blocked by you)", "规则:${m.rule}(你拦截的)")
        Blocklist.RuleSource.USER_ALLOW -> txt("Now allowed by you: ${m.rule}", "现已被你放行:${m.rule}")
        Blocklist.RuleSource.BUILTIN_ALLOW -> txt("Now allowed by the built-in list: ${m.rule}", "现已被内置名单放行:${m.rule}")
    }

    /** The theme's secondary text colour, or null to keep the default colour. */
    private fun secondaryTextColors(): android.content.res.ColorStateList? {
        val a = obtainStyledAttributes(intArrayOf(android.R.attr.textColorSecondary))
        try {
            return a.getColorStateList(0)
        } finally {
            a.recycle()
        }
    }

    // "Blocked by you": the user's own block list. Mirror of the allow list below it; both are

    // The screen refreshes every second; setting identical text still re-lays out the view, which keeps
