任务 G4(Guardian):IPv6 上的 DNS 查询。

背景:G3b 已完成。CI 全绿:core 测试、app 编译和 lint、API 29/34 模拟器整套流程(包括 Allow/Remove)。我审查时改了两处你的代码:
- 停止/加载竞态,加了 startGeneration;
- 去掉了按 App 统计。模拟器实测所有 DNS 查询的 uid 都是 1051,因为 Android 10+ 的应用经由系统 DNS 服务解析,纯 DNS 的 VPN 看不出是哪个 App;界面改成「拦截最多的域名」。
附件里的文件以当前版本为准。

现状缺口:VPN 只声明了 IPv4 DNS 服务器(10.111.222.2)。在只有 IPv6 DNS 的网络上,或系统选用 IPv6 DNS 时,查询会绕过 Guardian。

硬性规则(同前):每个 .kt 文件第一行带 SPDX 头;不加依赖、不加权限;不出现公共 DNS 地址;日志只在内存。

一、core(DnsPacket.kt 及相关)
1. parseQuery 同时接受 IPv4 和 IPv6:
   - 第一个字节高 4 位 = 6 时按 IPv6 处理:固定 40 字节头;Next Header 必须是 17(UDP),有扩展头的包一律丢弃(返回 null);Payload Length 必须与实际长度一致;目的端口 53。
   - DnsQuery 的 srcIp/dstIp 为 16 字节。DnsQuery 增加 `val ipVersion: Int`(4 或 6),放在最后并给默认值 4,旧调用不变。ipHeaderLen 对 IPv6 为 40。
2. buildBlockedResponse 和 buildResponseFromUpstream 对 IPv6 查询构造 IPv6 + UDP 回包:
   - 交换源/目的地址和端口;Hop Limit 64;Payload Length 正确;
   - UDP 校验和按 IPv6 伪首部(RFC 8200 第 8.1 节)计算,结果为 0 时写 0xFFFF(IPv6 下 UDP 校验和不能为 0)。
   - NULL_IP 模式下 A 查询回 0.0.0.0、AAAA 回 ::,与 IPv4 路径一致。
3. Checksum 增加 `fun udp6(packet: ByteArray): Int` 和 `verify6`(或让 verify 自动识别版本)。
4. extractQueryPayload 对 IPv6 也正确。
5. 新测试文件 core/src/test/kotlin/org/secureinchat/guardian/dns/DnsIpv6Test.kt,写法照 DnsFilterTest(自己的 t()/eq(),main 打印 "DnsIpv6Test: N passed"):
   a. 合法 IPv6 A/AAAA 查询解析正确(地址、端口、qname、ipVersion 6);
   b. 有扩展头(Next Header 0 / 43 / 44 / 60)、Payload Length 不符、截断包、端口非 53:全部返回 null 且不抛异常;
   c. 拦截回包:IPv6 头各字段正确,UDP 校验和用独立实现验证通过,0 校验和规则;
   d. 上游回包包装:ID / 问题不匹配时返回 null(照 IPv4 的规则);
   e. 2 万个随机字节包(首字节强制为 0x60)不抛异常;
   f. 现有 IPv4 测试全部照旧通过。

// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian.dns

import java.io.ByteArrayOutputStream

enum class BlockMode { NXDOMAIN, NULL_IP }

data class DnsQuery(
    val srcIp: ByteArray,
    val dstIp: ByteArray,
    val srcPort: Int,
    val dstPort: Int,
    val ipHeaderLen: Int,
    val dnsId: Int,
    val flags: Int,
    val qname: String,
    val qtype: Int,
    val qclass: Int,
    val questionEnd: Int
)

object DnsPacket {

    fun parseQuery(packet: ByteArray, length: Int): DnsQuery? {
        try {
            if (length < 20) return null
            if (length > packet.size) return null

            val version = (packet[0].toInt() ushr 4) and 0xF
            if (version != 4) return null

            val ihl = packet[0].toInt() and 0xF
            if (ihl < 5) return null
            val ipHeaderLen = ihl * 4
            if (ipHeaderLen > length) return null

            val totalLen = ((packet[2].toInt() and 0xFF) shl 8) or (packet[3].toInt() and 0xFF)
            val packetLen = minOf(totalLen, length)
            if (packetLen < ipHeaderLen) return null

            if ((packet[9].toInt() and 0xFF) != 17) return null

            val flagsFrag = packet[6].toInt() and 0xFF
            val mf = (flagsFrag ushr 5) and 1
            val fragOffset = ((flagsFrag and 0x1F) shl 8) or (packet[7].toInt() and 0xFF)
            if (mf != 0 || fragOffset != 0) return null

            if (ipHeaderLen + 8 > packetLen) return null
            val srcPort = ((packet[ipHeaderLen].toInt() and 0xFF) shl 8) or (packet[ipHeaderLen + 1].toInt() and 0xFF)
            val dstPort = ((packet[ipHeaderLen + 2].toInt() and 0xFF) shl 8) or (packet[ipHeaderLen + 3].toInt() and 0xFF)
            if (dstPort != 53) return null

            val udpLen = ((packet[ipHeaderLen + 4].toInt() and 0xFF) shl 8) or (packet[ipHeaderLen + 5].toInt() and 0xFF)
            if (udpLen < 8) return null
            if (ipHeaderLen + udpLen > packetLen) return null

            val dnsOff = ipHeaderLen + 8
            val dnsLen = udpLen - 8
            if (dnsLen < 12) return null
            val dnsEnd = dnsOff + dnsLen

            val dnsId = ((packet[dnsOff].toInt() and 0xFF) shl 8) or (packet[dnsOff + 1].toInt() and 0xFF)
            val dnsFlags = ((packet[dnsOff + 2].toInt() and 0xFF) shl 8) or (packet[dnsOff + 3].toInt() and 0xFF)
            if (((dnsFlags ushr 15) and 1) != 0) return null
            if (((dnsFlags ushr 11) and 0xF) != 0) return null
            val qdcount = ((packet[dnsOff + 4].toInt() and 0xFF) shl 8) or (packet[dnsOff + 5].toInt() and 0xFF)
            if (qdcount != 1) return null

            var pos = dnsOff + 12
            val labels = ArrayList<String>(8)
            while (true) {
                if (pos >= dnsEnd) return null
                val b = packet[pos].toInt() and 0xFF
                if ((b and 0xC0) == 0xC0) return null
                if (b == 0) { pos++; break }
                if (b < 1 || b > 63) return null
                if (pos + 1 + b > dnsEnd) return null
                val sb = StringBuilder(b)
                for (i in 0 until b) {
                    val c = packet[pos + 1 + i].toInt() and 0xFF
                    if (c < 0x21 || c > 0x7E) return null
                    sb.append(c.toChar())
                }
                labels.add(sb.toString())
                pos += 1 + b
            }
            if (labels.isEmpty()) return null
            val qname = labels.joinToString(".").lowercase()
            if (qname.length > 253) return null

            if (pos + 4 > dnsEnd) return null
            val qtype = ((packet[pos].toInt() and 0xFF) shl 8) or (packet[pos + 1].toInt() and 0xFF)
            val qclass = ((packet[pos + 2].toInt() and 0xFF) shl 8) or (packet[pos + 3].toInt() and 0xFF)
            pos += 4
            val questionEnd = pos

            val srcIp = byteArrayOf(packet[12], packet[13], packet[14], packet[15])
            val dstIp = byteArrayOf(packet[16], packet[17], packet[18], packet[19])

            return DnsQuery(srcIp, dstIp, srcPort, dstPort, ipHeaderLen, dnsId, dnsFlags, qname, qtype, qclass, questionEnd)
        } catch (t: Throwable) {
            return null
        }
    }

    fun buildBlockedResponse(q: DnsQuery, mode: BlockMode): ByteArray {
        val dns = ByteArrayOutputStream(128)

        val rcode = if (mode == BlockMode.NXDOMAIN) 3 else 0
        var ancount = 0
        if (mode == BlockMode.NULL_IP && (q.qtype == 1 || q.qtype == 28)) ancount = 1

        var dnsFlags = 0
        dnsFlags = dnsFlags or (1 shl 15)               // QR = 1
        dnsFlags = dnsFlags or (q.flags and 0x0100)     // RD copied
        dnsFlags = dnsFlags or (1 shl 7)                // RA = 1
        dnsFlags = dnsFlags or rcode

        dns.write((q.dnsId ushr 8) and 0xFF)
        dns.write(q.dnsId and 0xFF)
        dns.write((dnsFlags ushr 8) and 0xFF)
        dns.write(dnsFlags and 0xFF)
        dns.write(0x00); dns.write(0x01)                 // QDCOUNT = 1
        dns.write((ancount ushr 8) and 0xFF)
        dns.write(ancount and 0xFF)
        dns.write(0x00); dns.write(0x00)                 // NSCOUNT
        dns.write(0x00); dns.write(0x00)                 // ARCOUNT

        for (label in q.qname.split('.')) {
            dns.write(label.length and 0xFF)
            dns.write(label.toByteArray(Charsets.US_ASCII))
        }
        dns.write(0x00)
        dns.write((q.qtype ushr 8) and 0xFF)
        dns.write(q.qtype and 0xFF)
        dns.write((q.qclass ushr 8) and 0xFF)
        dns.write(q.qclass and 0xFF)

        if (mode == BlockMode.NULL_IP && q.qtype == 1) {
            dns.write(0xC0); dns.write(0x0C)
            dns.write(0x00); dns.write(0x01)
            dns.write(0x00); dns.write(0x01)
            dns.write(0x00); dns.write(0x00); dns.write(0x00); dns.write(60)
            dns.write(0x00); dns.write(0x04)
            dns.write(0x00); dns.write(0x00); dns.write(0x00); dns.write(0x00)
        } else if (mode == BlockMode.NULL_IP && q.qtype == 28) {
            dns.write(0xC0); dns.write(0x0C)
            dns.write(0x00); dns.write(28)
            dns.write(0x00); dns.write(0x01)
            dns.write(0x00); dns.write(0x00); dns.write(0x00); dns.write(60)
            dns.write(0x00); dns.write(16)
            for (i in 0 until 16) dns.write(0x00)
        }

        return wrapUdp(q.dstIp, q.srcIp, q.dstPort, q.srcPort, dns.toByteArray())
    }

    fun buildResponseFromUpstream(q: DnsQuery, upstream: ByteArray, upstreamLen: Int): ByteArray? {
        try {
            if (upstreamLen < 12) return null
            if (upstreamLen > upstream.size) return null
            if (upstreamLen > 65507) return null

            val id = ((upstream[0].toInt() and 0xFF) shl 8) or (upstream[1].toInt() and 0xFF)
            if (id != q.dnsId) return null

            val flags = ((upstream[2].toInt() and 0xFF) shl 8) or (upstream[3].toInt() and 0xFF)
            if (((flags ushr 15) and 1) != 1) return null

            val qd = ((upstream[4].toInt() and 0xFF) shl 8) or (upstream[5].toInt() and 0xFF)
            if (qd != 1) return null

            var pos = 12
            val labels = ArrayList<String>(8)
            while (true) {
                if (pos >= upstreamLen) return null
                val b = upstream[pos].toInt() and 0xFF
                if ((b and 0xC0) == 0xC0) return null
                if (b == 0) { pos++; break }
                if (b < 1 || b > 63) return null
                if (pos + 1 + b > upstreamLen) return null
                val sb = StringBuilder(b)
                for (i in 0 until b) {
                    val c = upstream[pos + 1 + i].toInt() and 0xFF
                    if (c < 0x21 || c > 0x7E) return null
                    sb.append(c.toChar())
                }
                labels.add(sb.toString())
                pos += 1 + b
            }
            if (labels.isEmpty()) return null
            val name = labels.joinToString(".").lowercase()
            if (name != q.qname) return null

            if (pos + 4 > upstreamLen) return null
            val qtype = ((upstream[pos].toInt() and 0xFF) shl 8) or (upstream[pos + 1].toInt() and 0xFF)
            val qclass = ((upstream[pos + 2].toInt() and 0xFF) shl 8) or (upstream[pos + 3].toInt() and 0xFF)
            if (qtype != q.qtype) return null
            if (qclass != q.qclass) return null

            val payload = upstream.copyOfRange(0, upstreamLen)
            return wrapUdp(q.dstIp, q.srcIp, q.dstPort, q.srcPort, payload)
        } catch (t: Throwable) {
            return null
        }
    }

    internal fun wrapUdp(
        srcIp: ByteArray,
        dstIp: ByteArray,
        srcPort: Int,
        dstPort: Int,
        payload: ByteArray
    ): ByteArray {
        val udpLen = 8 + payload.size
        val udp = ByteArray(udpLen)
        udp[0] = ((srcPort ushr 8) and 0xFF).toByte()
        udp[1] = (srcPort and 0xFF).toByte()
        udp[2] = ((dstPort ushr 8) and 0xFF).toByte()
        udp[3] = (dstPort and 0xFF).toByte()
        udp[4] = ((udpLen ushr 8) and 0xFF).toByte()
        udp[5] = (udpLen and 0xFF).toByte()
        udp[6] = 0
        udp[7] = 0
        System.arraycopy(payload, 0, udp, 8, payload.size)

        val totalLen = 20 + udpLen
        val ip = ByteArray(totalLen)
        ip[0] = 0x45
        ip[1] = 0
        ip[2] = ((totalLen ushr 8) and 0xFF).toByte()
        ip[3] = (totalLen and 0xFF).toByte()
        ip[4] = 0
        ip[5] = 0
        ip[6] = 0x40
        ip[7] = 0
        ip[8] = 64
        ip[9] = 17
        ip[10] = 0
        ip[11] = 0
        System.arraycopy(srcIp, 0, ip, 12, 4)
        System.arraycopy(dstIp, 0, ip, 16, 4)

        val ipck = Checksum.ipv4Header(ip, 0, 20)
        ip[10] = ((ipck ushr 8) and 0xFF).toByte()
        ip[11] = (ipck and 0xFF).toByte()

        System.arraycopy(udp, 0, ip, 20, udpLen)

        val udpck = Checksum.udp(ip)
        ip[26] = ((udpck ushr 8) and 0xFF).toByte()
        ip[27] = (udpck and 0xFF).toByte()

        return ip
    }
}

object DnsWire {
    fun extractQueryPayload(packet: ByteArray, q: DnsQuery): ByteArray {
        val ihl = q.ipHeaderLen
        val udpLen = ((packet[ihl + 4].toInt() and 0xFF) shl 8) or (packet[ihl + 5].toInt() and 0xFF)
        val dnsLen = udpLen - 8
        val dnsOff = ihl + 8
        return packet.copyOfRange(dnsOff, dnsOff + dnsLen)
    }
}

object Checksum {

    fun ipv4Header(buf: ByteArray, off: Int, len: Int): Int {
        var sum = 0L
        var i = 0
        while (i + 1 < len) {
            sum += ((buf[off + i].toInt() and 0xFF) shl 8) or (buf[off + i + 1].toInt() and 0xFF)
            i += 2
        }
        if (i < len) {
            sum += (buf[off + i].toInt() and 0xFF) shl 8
        }
        while ((sum shr 16) != 0L) {
            sum = (sum and 0xFFFF) + (sum shr 16)
        }
        return (sum.inv() and 0xFFFF).toInt()
    }

    fun udp(packet: ByteArray): Int {
        val ihl = (packet[0].toInt() and 0xF) * 4
        val udpLen = ((packet[ihl + 4].toInt() and 0xFF) shl 8) or (packet[ihl + 5].toInt() and 0xFF)
        var sum = 0L

        sum += ((packet[12].toInt() and 0xFF) shl 8) or (packet[13].toInt() and 0xFF)
        sum += ((packet[14].toInt() and 0xFF) shl 8) or (packet[15].toInt() and 0xFF)
        sum += ((packet[16].toInt() and 0xFF) shl 8) or (packet[17].toInt() and 0xFF)
        sum += ((packet[18].toInt() and 0xFF) shl 8) or (packet[19].toInt() and 0xFF)
        sum += 17L
        sum += udpLen.toLong()

        var i = 0
        while (i + 1 < udpLen) {
            if (i != 6) {
                sum += ((packet[ihl + i].toInt() and 0xFF) shl 8) or (packet[ihl + i + 1].toInt() and 0xFF)
            }
            i += 2
        }
        if (i < udpLen) {
            sum += (packet[ihl + i].toInt() and 0xFF) shl 8
        }
        while ((sum shr 16) != 0L) {
            sum = (sum and 0xFFFF) + (sum shr 16)
        }
        val c = (sum.inv() and 0xFFFF).toInt()
        return if (c == 0) 0xFFFF else c
    }

    fun verify(packet: ByteArray): Boolean {
        if (packet.size < 20) return false
        val ihl = (packet[0].toInt() and 0xF) * 4
        if (ihl < 20 || ihl > packet.size) return false
        if (ipv4Header(packet, 0, ihl) != 0) return false
        if (packet.size < ihl + 8) return false
        val stored = ((packet[ihl + 6].toInt() and 0xFF) shl 8) or (packet[ihl + 7].toInt() and 0xFF)
        val computed = udp(packet)
        return stored == computed
    }
}

package org.secureinchat.guardian

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.content.pm.ServiceInfo
import android.net.VpnService
import android.os.Build
import android.os.ParcelFileDescriptor
import java.io.FileInputStream
import java.io.FileOutputStream
import java.util.concurrent.ExecutorService
import java.util.concurrent.Executors
import org.secureinchat.guardian.dns.Action
import org.secureinchat.guardian.dns.Blocklist
import org.secureinchat.guardian.dns.BlockMode
import org.secureinchat.guardian.dns.QueryHandler

class GuardianVpnService : VpnService() {

    private var tun: ParcelFileDescriptor? = null
    private var tunIn: FileInputStream? = null
    private var tunOut: FileOutputStream? = null
    private val writeLock = Any()
    private var readThread: Thread? = null
    private var pool: ExecutorService? = null
    @Volatile private var running = false
    private var handler: QueryHandler? = null
    private var upstream: Upstream? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> {
                stopVpnInternal()
                stopSelf()
                return START_NOT_STICKY
            }
            else -> {
                if (!running && !isLoading) {
                    // startForeground can be refused (e.g. systemExempted on API 34+ when this app
                    // is no longer the configured VPN after a sticky restart). Stop rather than run unpromoted.
                    val promoted = try { startForegroundSafe(); true } catch (_: RuntimeException) { false }
                    if (!promoted) {
                        stopSelf()
                        return START_NOT_STICKY
                    }
                    isLoading = true
                    val gen = ++startGeneration
                    Thread { startVpnBackground(gen) }.apply { name = "guardian-start" }.start()
                }
            }
        }
        return START_STICKY
    }

    // Bumped by every start and stop. A start thread that finds a newer generation after loading the
    // list gives up, so Stop pressed while the list is loading cannot be overtaken by a late establish().
    @Volatile private var startGeneration = 0

    private fun startVpnBackground(gen: Int) {
        try {
            val blocklist = loadBlocklist()
            val userAllowed = UserRulesStore.allowed(applicationContext)
            val effective = if (userAllowed.isEmpty()) blocklist else blocklist.plusAllowed(userAllowed)
            // No uid lookup: on Android 10+ every query reaching the VPN is sent by the system DNS
            // service, so getConnectionOwnerUid names that service, not the app (seen in the emulator flow).
            val h = QueryHandler(effective, BlockMode.NULL_IP)
            if (gen != startGeneration) return

            val builder = Builder()
            builder.setSession("Guardian")
            builder.addAddress("10.111.222.1", 32)
            builder.addDnsServer("10.111.222.2")
            builder.addRoute("10.111.222.2", 32)
            builder.setMtu(1500)
            builder.setBlocking(true)

            val pfd = try { builder.establish() } catch (_: Throwable) { null }
            if (gen != startGeneration) {
                try { pfd?.close() } catch (_: Throwable) {}
                return
            }
            if (pfd == null) {
                isLoading = false
                stopSelf()
                return
            }
            tun = pfd
            tunIn = FileInputStream(pfd.fileDescriptor)
            tunOut = FileOutputStream(pfd.fileDescriptor)

            val up = Upstream(applicationContext) { sock -> protect(sock) }
            up.start()
            upstream = up

            handler = h
            activeHandler = h
            pool = Executors.newFixedThreadPool(8)
            running = true
            isRunning = true
            isLoading = false

            val t = Thread { readLoop(pfd) }
            t.name = "guardian-tun-read"
            readThread = t
            t.start()
        } catch (_: Throwable) {
            isLoading = false
            stopSelf()
        }
    }

    override fun onRevoke() {
        stopVpnInternal()
        stopSelf()
        super.onRevoke()
    }

    override fun onDestroy() {
        stopVpnInternal()
        super.onDestroy()
    }

    private fun startForegroundSafe() {
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        if (nm.getNotificationChannel(CHANNEL_ID) == null) {
            nm.createNotificationChannel(
                NotificationChannel(CHANNEL_ID, "Guardian VPN", NotificationManager.IMPORTANCE_LOW)
            )
        }
        val stopIntent = Intent(this, GuardianVpnService::class.java).setAction(ACTION_STOP)
        val pi = PendingIntent.getService(
            this, 1, stopIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val notif: Notification = Notification.Builder(this, CHANNEL_ID)
            .setContentTitle("Guardian")
            .setContentText("DNS filtering active")
            .setSmallIcon(android.R.drawable.ic_lock_lock)
            .addAction(0, "Stop", pi)
            .setOngoing(true)
            .build()
        if (Build.VERSION.SDK_INT >= 34) {
            startForeground(NOTIF_ID, notif, ServiceInfo.FOREGROUND_SERVICE_TYPE_SYSTEM_EXEMPTED)
        } else if (Build.VERSION.SDK_INT >= 29) {
            startForeground(NOTIF_ID, notif, 0)
        } else {
            startForeground(NOTIF_ID, notif)
        }
    }

    private fun readLoop(pfd: ParcelFileDescriptor) {
        val buf = ByteArray(32767)
        val input = tunIn ?: return
        while (running) {
            val n = try { input.read(buf) } catch (_: Throwable) { if (!running) break else continue }
            if (n <= 0) continue
            val copy = buf.copyOf(n)
            val exec = pool ?: break
            try { exec.execute { processPacket(copy) } } catch (_: Throwable) {}
        }
    }

    private fun processPacket(packet: ByteArray) {
        val h = handler ?: return
        when (val act = h.handle(packet, packet.size)) {
            is Action.Respond -> writeTun(act.packet)
            is Action.Forward -> {
                val exec = pool ?: return
                try {
                    exec.execute {
                        val ans = upstream?.sendQuery(act.payload)
                        if (ans == null) {
                            h.onUpstreamFailed(act.query)
                        } else {
                            val resp = h.onUpstreamAnswer(act.query, ans, ans.size)
                            if (resp != null) writeTun(resp)
                        }
                    }
                } catch (_: Throwable) {}
            }
            Action.Drop -> {}
        }
    }

    private fun writeTun(packet: ByteArray) {
        val out = tunOut ?: return
        synchronized(writeLock) {
            try { out.write(packet) } catch (_: Throwable) {}
        }
    }

    private fun stopVpnInternal() {
        startGeneration++
        running = false
        isRunning = false
        isLoading = false
        try { tun?.close() } catch (_: Throwable) {}
        try { tunIn?.close() } catch (_: Throwable) {}
        try { tunOut?.close() } catch (_: Throwable) {}
        tun = null; tunIn = null; tunOut = null
        try { readThread?.join(1000) } catch (_: Throwable) {}
        readThread = null
        try { pool?.shutdownNow() } catch (_: Throwable) {}
        pool = null
        try { upstream?.stop() } catch (_: Throwable) {}
        upstream = null
        handler = null
        activeHandler = null
    }

    private fun loadBlocklist(): Blocklist = loadBlocklistStatic(applicationContext)

    companion object {
        const val ACTION_START = "org.secureinchat.guardian.action.START"
        const val ACTION_STOP = "org.secureinchat.guardian.action.STOP"
        private const val CHANNEL_ID = "guardian_vpn"
        private const val NOTIF_ID = 1
        private val LIST_ASSETS = listOf("blocklist-hagezi-light.txt", "blocklist-extra.txt")
        private val rulesLock = Any()

        @Volatile var isRunning: Boolean = false
            private set

        @Volatile var isLoading: Boolean = false
            private set

        @Volatile var activeHandler: QueryHandler? = null
            private set

        // The bundled lists never change while the app runs, so parse them once per process.
        @Volatile private var baseCache: Blocklist? = null

        private fun loadBlocklistStatic(ctx: Context): Blocklist {
            baseCache?.let { return it }
            return try {
                val text = LIST_ASSETS.joinToString("\n") { name ->
                    ctx.assets.open(name).use { it.bufferedReader().readText() }
                }
                Blocklist.parse(text).also { baseCache = it }
            } catch (_: Throwable) {
                Blocklist.parse("")
            }
        }

        fun applyUserRules(ctx: Context) {
            val h = activeHandler ?: return
            Thread {
                synchronized(rulesLock) {
                    try {
                        val base = loadBlocklistStatic(ctx)
                        val userAllowed = UserRulesStore.allowed(ctx)
                        val effective = if (userAllowed.isEmpty()) base else base.plusAllowed(userAllowed)
                        h.blocklist = effective
                    } catch (_: Throwable) {}
                }
            }.apply { name = "guardian-rules" }.start()
        }
    }
}
