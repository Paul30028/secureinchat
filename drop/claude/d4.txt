        function startRadar() {
            const resultArea = document.getElementById('radar-results');
            resultArea.classList.add('hidden');
            
            setTimeout(() => {
                // 真实获取浏览器特征
                document.getElementById('fp-screen').innerText = `${window.screen.width}x${window.screen.height}`;
                document.getElementById('fp-lang').innerText = navigator.language;
                document.getElementById('fp-tz').innerText = Intl.DateTimeFormat().resolvedOptions().timeZone;
                document.getElementById('fp-cpu').innerText = navigator.hardwareConcurrency || '未知';
                
                // 模拟生成一个基于特征的 Hash
                const rawFp = navigator.userAgent + window.screen.width + Intl.DateTimeFormat().resolvedOptions().timeZone;
                let hash = 0;
                for (let i = 0; i < rawFp.length; i++) {
                    hash = ((hash << 5) - hash) + rawFp.charCodeAt(i);
                    hash |= 0;
                }
                document.getElementById('fp-hash').innerText = Math.abs(hash).toString(16).padEnd(12, 'a').toUpperCase();
                
                resultArea.classList.remove('hidden');
            }, 2000);
        }

        // --- 3. 系统加固脚本生成 ---
        function genScript(os) {

Set-ItemProperty -Path "HKLM:\\SOFTWARE\\Policies\\Microsoft\\Windows\\DataCollection" -Name "AllowTelemetry" -Type DWord -Value 0

        // --- 5. 零宽盲水印 ---
        function embedWM() {
            const key = document.getElementById('wmKey').value;
            const text = document.getElementById('wmText').value;
            if(!key || !text) return alert('请输入内容');
            
            let binaryKey = '';
            for (let i = 0; i < key.length; i++) binaryKey += key.charCodeAt(i).toString(2).padStart(8, '0');
            let watermark = '';
            for (let i = 0; i < binaryKey.length; i++) watermark += binaryKey[i] === '0' ? '\u200B' : '\u200C';

            document.getElementById('wmResult').classList.remove('hidden');
            document.getElementById('wmResult').innerText = text + watermark;
            alert('水印已嵌入下方文本，请复制。');
        }

        function extractWM() {
            const text = document.getElementById('wmResult').innerText;
            if(!text) return;
            let binaryKey = '';
            for (let i = 0; i < text.length; i++) {
                if (text[i] === '\u200B') binaryKey += '0';
                if (text[i] === '\u200C') binaryKey += '1';
            }
            if(!binaryKey) return alert('未检测到水印');
            let key = '';
            for (let i = 0; i < binaryKey.length; i += 8) key += String.fromCharCode(parseInt(binaryKey.slice(i, i + 8), 2));
            alert('提取到作者水印：' + key);
        }

        // --- 6. 哈希校验 ---
        async function calcHash(file) {
            if(!file) return;
            const arrayBuffer = await file.arrayBuffer();
            const hashBuffer = await crypto.subtle.digest('SHA-256', arrayBuffer);
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
            
            document.getElementById('hashResult').classList.remove('hidden');
            document.getElementById('hashValue').innerText = hashHex;
        }

            const currentY = e.clientY - rect.top;

            scrCtx.drawImage(imgElement, 0, 0);
            scrCtx.fillStyle = '#0f172a';
            scrCtx.fillRect(startX, startY, currentX - startX, currentY - startY);
        });

        scrCanvas.addEventListener('mouseleave', () => { isDrawing = false; });

        // --- 5. 零宽字符盲水印 (真实实现) ---
        function embedWatermark() {
            const key = document.getElementById('wmKey').value;
            const text = document.getElementById('wmText').value;
            if(!key || !text) return alert('请输入标识和文本');

            // 将 key 转为二进制，并映射为 零宽字符 (U+200B 表示 0, U+200C 表示 1)
            let binaryKey = '';
            for (let i = 0; i < key.length; i++) {
                binaryKey += key.charCodeAt(i).toString(2).padStart(8, '0');
            }
            let watermark = '';
            for (let i = 0; i < binaryKey.length; i++) {
                watermark += binaryKey[i] === '0' ? '\u200B' : '\u200C';
            }

            const result = text + watermark;
            document.getElementById('wmResult').classList.remove('hidden');
            document.getElementById('wmResult').innerText = result;
            alert('水印已嵌入下方文本中，请直接复制。');
        }

        function extractWatermark() {
            const text = document.getElementById('wmResult').innerText;
            if(!text) return alert('没有可提取的文本');
            
            let binaryKey = '';
            for (let i = 0; i < text.length; i++) {
                if (text[i] === '\u200B') binaryKey += '0';
                if (text[i] === '\u200C') binaryKey += '1';
            }
            
            if(!binaryKey) return alert('未检测到水印');
            
            let key = '';
            for (let i = 0; i < binaryKey.length; i += 8) {
                const byte = binaryKey.slice(i, i + 8);
                key += String.fromCharCode(parseInt(byte, 2));
            }
            alert('提取到水印内容：' + key);
        }

        // --- 6. 文件哈希校验 (真实实现) ---
        async function calcHash(file) {
            if(!file) return;
            const arrayBuffer = await file.arrayBuffer();
            const hashBuffer = await crypto.subtle.digest('SHA-256', arrayBuffer);
            const hashArray = Array.from(new Uint8Array(hashBuffer));
            const hashHex = hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
            
            document.getElementById('hashResult').classList.remove('hidden');
            document.getElementById('hashValue').innerText = hashHex;
        }
