一、redact.js(逻辑)
1. redact(text, enabled, opts) 增加第三个可选参数 opts:
   - opts.terms:字符串数组,用户自己要隐藏的名字/公司/地址等。每项 trim 后长度 < 2 的忽略;去重(不区分大小写);长的优先匹配(避免 "Ann" 抢在 "Ann Lee" 前)。
   - 匹配不区分大小写。对含拉丁字母或数字的词,要求两侧不是字母/数字/下划线(Unicode 意义上:用 /[\p{L}\p{N}_]/u 判断前后各一个字符;如果运行环境不支持 \p 就退回 [A-Za-z0-9_]);对纯 CJK 词(中文、日文、韩文)不要求边界。
   - 这些命中用 rule id 'term',tag 'NAME',优先级高于所有现有规则(排在 RULES 处理之前,重叠时 term 赢)。同一个词(不区分大小写)用同一个编号。
   - enabled.term === false 时不替换 terms。
   - 返回值新增 mapping:对象,键是完整占位符(如 "[EMAIL_1]"),值是第一次出现时的原文(保留原大小写)。url_cred 规则的 mapping 值是被替换掉的那段(user:password),和 text 里被替换的内容一致。
2. 新函数 restore(text, mapping):
   - 把 text 里出现的占位符换回原文。识别这些写法(AI 经常改格式):[EMAIL_1]、[ EMAIL_1 ]、[email_1](大小写不敏感)、\[EMAIL\_1\](Markdown 转义)、【EMAIL_1】(全角方括号)。不识别不带括号的 EMAIL_1(太容易误伤)。

   - 原文里如果含有 "$" 等特殊字符也必须原样还原(不要用 String.replace 的替换字符串语法踩坑)。
3. 导出 { RULES, redact, restore, luhn, ibanOk, parseTerms }。parseTerms(str):按换行和英文/中文逗号、分号拆分,trim,去空,返回数组。

/* Text redaction rules. Pure functions, no network. Loaded by the page and by the Node tests. */
(function (root, factory) {
  if (typeof module === 'object' && module.exports) module.exports = factory(); else root.RD = factory();
})(typeof self !== 'undefined' ? self : this, function () {
  'use strict';

  function luhn(digits) {
    var sum = 0, alt = false;
    for (var i = digits.length - 1; i >= 0; i--) {
      var n = digits.charCodeAt(i) - 48;
      if (alt) { n *= 2; if (n > 9) n -= 9; }
      sum += n; alt = !alt;
    }
    return sum % 10 === 0;
  }
  function ibanOk(s) {
    var v = s.replace(/\s/g, '').toUpperCase();
    if (v.length < 15 || v.length > 34 || !/^[A-Z]{2}\d{2}[A-Z0-9]+$/.test(v)) return false;
    var r = v.slice(4) + v.slice(0, 4), rem = 0;
    for (var i = 0; i < r.length; i++) {
      var c = r.charCodeAt(i), d = c >= 65 ? String(c - 55) : r[i];
      for (var j = 0; j < d.length; j++) rem = (rem * 10 + (d.charCodeAt(j) - 48)) % 97;
    }
    return rem === 1;
  }
  function ipv4ok(s) {
    var p = s.split('.');
    return p.length === 4 && p.every(function (x) { return x.length > 0 && x.length <= 3 && +x <= 255 && (x === '0' || x[0] !== '0'); });
  }

  // Order matters: earlier rules win when matches overlap.
  var RULES = [
    { id: 'privkey', label: 'Private key block', tag: 'PRIVATE_KEY', re: /-----BEGIN [A-Z ]*PRIVATE KEY-----[\s\S]*?-----END [A-Z ]*PRIVATE KEY-----/g },
    { id: 'jwt', label: 'JSON Web Token', tag: 'JWT', re: /\beyJ[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}\.[A-Za-z0-9_-]{8,}\b/g },
    { id: 'apikey', label: 'API key or token', tag: 'API_KEY', re: /\b(?:AKIA[0-9A-Z]{16}|gh[pousr]_[A-Za-z0-9]{30,}|github_pat_[A-Za-z0-9_]{30,}|sk-[A-Za-z0-9_-]{20,}|xox[abprs]-[A-Za-z0-9-]{10,}|AIza[0-9A-Za-z_-]{35})\b/g },
    { id: 'url_cred', label: 'Password inside a URL', tag: 'URL_CREDENTIALS', re: /\b[a-z][a-z0-9+.-]*:\/\/[^\s\/:@]+:[^\s\/@]+@/gi, keep: function (m) { return m.slice(0, m.indexOf('://') + 3); }, suffix: '@' },
    { id: 'email', label: 'Email address', tag: 'EMAIL', re: /\b[A-Za-z0-9._%+-]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)*\.[A-Za-z]{2,}\b/g },
    { id: 'iban', label: 'IBAN', tag: 'IBAN', re: /\b[A-Z]{2}\d{2}(?: ?[A-Z0-9]{4}){2,7}(?: ?[A-Z0-9]{1,3})?\b/g, test: ibanOk },
    { id: 'card', label: 'Payment card number', tag: 'CARD', re: /\b(?:\d[ -]?){12,18}\d\b/g, test: function (m) { var d = m.replace(/[ -]/g, ''); return d.length >= 13 && d.length <= 19 && luhn(d); } },
    { id: 'ipv6', label: 'IPv6 address', tag: 'IPV6', re: /\b(?:[0-9A-Fa-f]{1,4}:){7}[0-9A-Fa-f]{1,4}\b|\b(?:[0-9A-Fa-f]{1,4}:){1,6}(?::[0-9A-Fa-f]{1,4}){1,6}\b|\b(?:[0-9A-Fa-f]{1,4}:){1,7}:(?![0-9A-Fa-f:])/g, test: function (m) { return /[0-9A-Fa-f]{2,}|:/.test(m) && m.indexOf(':') !== -1 && m.length > 4; } },
    { id: 'ipv4', label: 'IPv4 address', tag: 'IPV4', re: /\b\d{1,3}(?:\.\d{1,3}){3}\b/g, test: ipv4ok },
    { id: 'mac', label: 'MAC address', tag: 'MAC', re: /\b(?:[0-9A-Fa-f]{2}[:-]){5}[0-9A-Fa-f]{2}\b/g },
    { id: 'phone', label: 'Phone number', tag: 'PHONE', re: /(?:\+\d{1,3}[ .-]?)?(?:\(\d{2,4}\)[ .-]?|\d{2,4}[ .-])\d{3,4}[ .-]\d{3,4}\b|\+\d{8,15}\b/g, test: function (m) { var d = m.replace(/\D/g, ''); return d.length >= 8 && d.length <= 15; } },
  ];

  function redact(text, enabled) {
    text = String(text);
    var on = enabled || {};
    var hits = [];
    RULES.forEach(function (r) {
      if (on[r.id] === false) return;
      r.re.lastIndex = 0;
      var m;
      while ((m = r.re.exec(text)) !== null) {
        if (m[0].length === 0) { r.re.lastIndex++; continue; }
        if (r.test && !r.test(m[0])) continue;
        var start = m.index, end = m.index + m[0].length;
        if (r.id === 'url_cred') { start = m.index + m[0].indexOf('://') + 3; end = m.index + m[0].length - 1; }
        hits.push({ start: start, end: end, rule: r });
      }
    });
    hits.sort(function (a, b) { return a.start - b.start || b.end - a.end; });
    var picked = [], last = 0;
    hits.forEach(function (h) { if (h.start >= last) { picked.push(h); last = h.end; } });
    var counts = {}, out = '', pos = 0, seen = {};
    picked.forEach(function (h) {
      var original = text.slice(h.start, h.end);
      var key = h.rule.id + '|' + original.toLowerCase();
      var k = seen[h.rule.id] || (seen[h.rule.id] = { n: 0, map: {} });
      if (!(key in k.map)) k.map[key] = ++k.n;
      counts[h.rule.id] = (counts[h.rule.id] || 0) + 1;
      out += text.slice(pos, h.start) + '[' + h.rule.tag + '_' + k.map[key] + ']';
      pos = h.end;
    });
    out += text.slice(pos);
    return { text: out, counts: counts, total: picked.length };
  }

  return { RULES: RULES, redact: redact, luhn: luhn, ibanOk: ibanOk };
});

/* Text redactor page. All work happens in this page; nothing is sent anywhere or stored. */
(function () {
  'use strict';
  var R = window.RD;
  var src = document.getElementById('rd-in');
  var out = document.getElementById('rd-out');
  var tbody = document.getElementById('rows');
  var summary = document.getElementById('summary');
  var copyBtn = document.getElementById('rd-copy');
  var clearBtn = document.getElementById('rd-clear');
  var boxes = {};
  R.RULES.forEach(function (r) { boxes[r.id] = document.getElementById('rule-' + r.id); });

  function enabled() { var o = {}; R.RULES.forEach(function (r) { o[r.id] = boxes[r.id].checked; }); return o; }

  function row(rule, count) {
    var tr = document.createElement('tr'); tr.setAttribute('data-status', count ? 'warn' : 'info');
    var th = document.createElement('th'); th.scope = 'row'; th.setAttribute('data-label', 'Check');
    var n = document.createElement('span'); n.className = 'c-name'; n.textContent = rule.label; th.appendChild(n);
    var td = document.createElement('td'); td.setAttribute('data-label', 'Result');
    var v = document.createElement('div'); v.className = 'val mono'; v.textContent = count ? count + ' replaced with [' + rule.tag + '_n]' : 'None found'; td.appendChild(v);
    var ts = document.createElement('td'); ts.setAttribute('data-label', 'Status');
    var p = document.createElement('span'); p.className = 'pill'; p.textContent = count ? 'Redacted' : 'None'; ts.appendChild(p);
    tr.appendChild(th); tr.appendChild(td); tr.appendChild(ts);
    return tr;
  }

  function run() {
    var res = R.redact(src.value, enabled());
    out.value = res.text;
    tbody.textContent = '';
    R.RULES.forEach(function (r) { if (!boxes[r.id].checked) return; if (res.counts[r.id]) tbody.appendChild(row(r, res.counts[r.id])); });
    if (!src.value) {
      summary.textContent = 'Paste text to redact it.'; summary.setAttribute('data-tone', 'idle');
    } else if (!res.total) {
      summary.textContent = 'Nothing matched the selected rules. This does not mean the text is free of personal data.'; summary.setAttribute('data-tone', 'warn');
    } else {
      summary.textContent = res.total + ' item' + (res.total === 1 ? '' : 's') + ' replaced. Read the output before you share it: names and addresses are not detected.'; summary.setAttribute('data-tone', 'ok');
    }
    copyBtn.disabled = !out.value;
  }

  src.addEventListener('input', run);
  R.RULES.forEach(function (r) { boxes[r.id].addEventListener('change', run); });
  clearBtn.addEventListener('click', function () { src.value = ''; run(); src.focus(); });
  copyBtn.addEventListener('click', function () {
    var done = function () { copyBtn.textContent = 'Copied'; setTimeout(function () { copyBtn.textContent = 'Copy result'; }, 1500); };
    if (navigator.clipboard && navigator.clipboard.writeText) navigator.clipboard.writeText(out.value).then(done, function () { out.select(); });
    else { out.select(); }
  });
  run();
})();

    <h2 id="what-h">What it does</h2>
    <p>Paste text and the tool replaces emails, phone numbers, IP and MAC addresses, card numbers, IBANs (checked with the standard checksum), API keys, tokens, passwords inside URLs and private keys with labelled placeholders such as <span class="mono">[EMAIL_1]</span>. The same value always gets the same placeholder, so a log stays readable after redaction. This is useful before you paste a log, a config or an email into a bug report, a forum or an AI assistant.</p>
    <h2>What it does not do</h2>
    <p>It finds patterns, not meaning. Names, street addresses, company names and anything written in an unusual format are not detected. Always read the result before you share it. A card number is only replaced if it passes the standard checksum, so a made-up number may be left alone.</p>
    <h2>How your text is handled</h2>
    <p>The text is processed in this page. It is not uploaded, logged or stored, and the page loads no analytics. Your browser or an extension may still keep what you type, so use a private window for sensitive material. Related: <a href="/tools/photo-metadata-remover/">remove metadata from a photo</a>.</p>
  </section>
  <section class="wrap prose" aria-labelledby="faq-h">
    <h2 id="faq-h">Frequently asked questions</h2>
    <h3>Is my text sent to a server?</h3>
    <p>No. The matching runs in your browser. The page makes no network request when you paste or redact text.</p>
    <h3>Why did it miss a name or an address?</h3>
    <p>It matches patterns such as emails and numbers. A name has no fixed pattern, so it is not detected. Check the output yourself before you share it.</p>
    <h3>Why is the same placeholder used twice?</h3>
    <p>Equal values get the same placeholder, for example [IP_1] or [EMAIL_1], so you can still follow what happened in a log after redaction.</p>
    <h3>Can it find every API key?</h3>
    <p>No. It recognises common formats such as AWS, GitHub, Slack, Google and OpenAI-style keys, plus JSON Web Tokens and private key blocks. Other secrets, and keys in a custom format, can pass through. Rotate any secret that has been exposed.</p>
    <h3>Can it redact a file?</h3>
    <p>Not yet. Paste the text into the box. For photos, use the photo metadata remover.</p>
