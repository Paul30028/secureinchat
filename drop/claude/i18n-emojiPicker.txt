## 要做的事
1. 新文件 `apps/android-client/src/i18n/ns/emojiPicker.ts`：
   `import type { Messages } from "../index";`，导出 `export const zh: Messages`、`export const en: Messages`、`export const es: Messages`。
   三个对象的键完全一致；键一律以 `emojiPicker.` 开头（如 `emojiPicker.title`）；参数用 `{name}` 占位，三种语言占位符一致。
2. 改写 `apps/android-client/src/screens/EmojiPicker.tsx`：所有给人看的文字（包括 aria-label、placeholder、title、alt、错误提示、按钮文字）改成 `t("emojiPicker.xxx")`，`const t = useT();` 从 `"../i18n"` 导入。
   - 界面结构、样式、props、行为一律不变。不在组件外（模块顶层常量）调用 t；需要的话把常量挪进组件或改成键名。
   - 中文词条必须与原文**逐字相同**（现有测试按中文断言，必须原样通过）。
   - 日期、数字用 `Intl` 格式化时，locale 用 `useLocale()`（同样从 "../i18n" 导入）。
3. 英语自然简洁；西班牙语用中性拉美西语。语气平实准确，不夸大安全效果（不要写「完全安全」「无法破解」之类）。
4. 不加任何依赖；TypeScript strict，不用 any；保留现有注释。
5. 不要输出测试文件以外的其他文件；不要修改现有测试。

## 翻译层 API（apps/android-client/src/i18n/index.tsx，只读）
- `useT(): (key: string, params?: Record<string, string | number>) => string`
- `useLocale(): "zh" | "en" | "es"`
- 没有 Provider 时默认中文。

## 现有源码：apps/android-client/src/screens/EmojiPicker.tsx
import { colors, touchTarget } from "@secureinchat/ui";

/**
 * 表情面板。
 *
 * 不引入表情库：常用的就那么几十个，一个数组够了，而完整的 emoji 库要几百 KB，
 * 对一个装在手机上的小应用不值。
 *
 * 表情就是普通字符，跟着消息一起加密发送，中继看到的还是密文。
 */

const EMOJI_ROWS: { label: string; items: string[] }[] = [
  {
    label: "常用",
    items: ["😊", "😄", "🥹", "😭", "😅", "🤝", "🙏", "❤️", "👍", "👌", "🎉", "✨", "🌿", "☀️"],
  },
  {
    label: "心情",
    items: ["😌", "🤗", "😇", "🥰", "😴", "🤔", "😮", "😢", "😰", "😤", "🫶", "💪", "🕊️", "🌈"],
  },
  {
    label: "日常",
    items: ["🍚", "☕", "🎵", "📖", "🏠", "🚗", "⏰", "📅", "☔", "❄️", "🌙", "⭐", "🔥", "🎂"],
  },
];

export interface EmojiPickerProps {
  onPick: (emoji: string) => void;
}

export function EmojiPicker({ onPick }: EmojiPickerProps) {
  return (
    <div
      role="listbox"
      aria-label="选择表情"
      style={{
        maxHeight: 200,
        overflowY: "auto",
        padding: "8px 10px",
        borderTop: `0.5px solid ${colors.sageMint}`,
      }}
    >
      {EMOJI_ROWS.map((row) => (
        <div key={row.label} style={{ marginBottom: 8 }}>
          <div style={{ fontSize: 11, color: "#9A9A94", marginBottom: 4 }}>{row.label}</div>
          <div style={{ display: "flex", flexWrap: "wrap", gap: 2 }}>
            {row.items.map((emoji) => (
              <button
                key={emoji}
                role="option"
                aria-selected={false}
                aria-label={`表情 ${emoji}`}
                onClick={() => onPick(emoji)}
                style={{
                  minWidth: touchTarget.minDp,
                  minHeight: touchTarget.minDp,
                  fontSize: 22,
                  background: "transparent",
                  border: "none",
                  boxShadow: "none",
                  cursor: "pointer",
                  lineHeight: 1,
                }}
              >
                {emoji}
              </button>
            ))}
          </div>
        </div>
      ))}
    </div>
  );
}
