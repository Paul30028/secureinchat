// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian.dns

class Blocklist private constructor(
    private val blocked: HashSet<String>,
    private val allowed: HashSet<String>
) {
    val blockedCount: Int get() = blocked.size
    val allowedCount: Int get() = allowed.size

    fun isBlocked(name: String): Boolean {
        try {
            val n = normalizeHostname(name) ?: return false
            var cur = n
            while (true) {
                if (allowed.contains(cur)) return false
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            cur = n
            while (true) {
                if (blocked.contains(cur)) return true
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            return false
        } catch (t: Throwable) {
            return false
        }
    }

    fun plusAllowed(extra: Collection<String>): Blocklist {
        val newAllowed = HashSet<String>(allowed)
        for (s in extra) {
            val n = normalizeOrNull(s) ?: continue
            newAllowed.add(n)
        }
        return Blocklist(HashSet(blocked), newAllowed)
    }

    companion object {
        private val WS = Regex("\\s+")

        private val IGNORED = setOf(
            "localhost", "localhost.localdomain", "broadcasthost", "local",
            "0.0.0.0", "127.0.0.1", "ip6-localhost", "ip6-loopback"
        )

        fun normalizeOrNull(s: String): String? = normalizeHostname(s)

        fun parse(text: String): Blocklist {
            val blocked = HashSet<String>(1024)
            val allowed = HashSet<String>(64)

            for (rawLine in text.lineSequence()) {
                var line = rawLine.trim()
                if (line.isEmpty()) continue
                if (line.startsWith("#") || line.startsWith("!")) continue

                val hash = line.indexOf('#')
                if (hash >= 0) line = line.substring(0, hash).trim()
                if (line.isEmpty()) continue

                var ascii = true
                for (ch in line) {
                    if (ch.code >= 128) { ascii = false; break }
                }
                if (!ascii) continue

                if (line.startsWith("@@")) {
                    val domain = parseAdblock(line.substring(2))
                    if (domain != null) allowed.add(domain)
                    continue
                }

                if (line.startsWith("||")) {
                    val domain = parseAdblock(line)
                    if (domain != null) blocked.add(domain)
                    continue
                }

                val tokens = line.split(WS)
                if (tokens.size >= 2 && looksLikeIp(tokens[0])) {
                    for (k in 1 until tokens.size) {
                        val d = normalizeHostname(tokens[k]) ?: continue
                        if (d in IGNORED) continue
                        blocked.add(d)
                    }
                    continue
                }

                val d = normalizeHostname(line) ?: continue
                if (d in IGNORED) continue
                blocked.add(d)
            }

            return Blocklist(blocked, allowed)
        }

        private fun parseAdblock(s: String): String? {
            if (!s.startsWith("||")) return null
            var body = s.substring(2)
            if (!body.endsWith("^")) return null
            body = body.substring(0, body.length - 1)
            if (body.contains('*') || body.contains('/') || body.contains('$') || body.contains('|')) return null
            return normalizeHostname(body)
        }

        private fun looksLikeIp(s: String): Boolean {
            val parts = s.split('.')
            if (parts.size != 4) return false
            for (p in parts) {
                if (p.isEmpty()) return false
                for (ch in p) if (!ch.isDigit()) return false
                val v = p.toIntOrNull() ?: return false
                if (v < 0 || v > 255) return false
            }
            return true
        }

        private fun normalizeHostname(s: String): String? {
            val d = s.trim().lowercase().trimEnd('.')
            if (d.isEmpty()) return null
            if (d.length > 253) return null
            if (!d.contains('.')) return null
            if (d.contains("..")) return null
            for (label in d.split('.')) {
                if (label.isEmpty()) return null
                if (label.length > 63) return null
                if (label.startsWith('-') || label.endsWith('-')) return null
                for (ch in label) {
                    val ok = (ch in 'a'..'z') || (ch in '0'..'9') || ch == '-' || ch == '_'
                    if (!ok) return null
                }
            }
            return d
        }
    }
}

// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian

import android.content.Context
import org.secureinchat.guardian.dns.Blocklist

object UserRulesStore {
    private const val PREFS = "user_rules"
    private const val KEY = "allowed"

    fun allowed(ctx: Context): Set<String> {
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val raw = prefs.getStringSet(KEY, emptySet()) ?: emptySet()
        return HashSet(raw)
    }

    fun add(ctx: Context, domain: String): Boolean {
        val n = Blocklist.normalizeOrNull(domain) ?: return false
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val cur = HashSet(prefs.getStringSet(KEY, emptySet()) ?: emptySet())
        if (!cur.add(n)) return true
        prefs.edit().putStringSet(KEY, cur).apply()
        return true
    }

    fun remove(ctx: Context, domain: String) {
        val n = Blocklist.normalizeOrNull(domain) ?: domain.trim().lowercase().trimEnd('.')
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val cur = HashSet(prefs.getStringSet(KEY, emptySet()) ?: emptySet())
        if (cur.remove(n)) prefs.edit().putStringSet(KEY, cur).apply()
    }
}

package org.secureinchat.guardian

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager

import android.content.Intent
import android.content.pm.ServiceInfo

import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

import java.util.concurrent.Executors
import org.secureinchat.guardian.dns.Action
import org.secureinchat.guardian.dns.Blocklist
import org.secureinchat.guardian.dns.BlockMode
import org.secureinchat.guardian.dns.QueryHandler

class GuardianVpnService : VpnService() {

    private var tun: ParcelFileDescriptor? = null
    private var tunIn: FileInputStream? = null
    private var tunOut: FileOutputStream? = null
    private val writeLock = Any()
    private var readThread: Thread? = null
    private var pool: ExecutorService? = null
    @Volatile private var running = false
    private var handler: QueryHandler? = null
    private var upstream: Upstream? = null

    // Pause expires on its own (QueryHandler.isPaused() returns false once the absolute timestamp
    // passes), so the notification needs a one-shot timer that rewrites it back to "DNS filtering
    // active" when the pause ends. Posted on the main looper; cancelled on resume or stop.
    private val mainHandler = Handler(Looper.getMainLooper())
    @Volatile private var notifRefresh: Runnable? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> {
                stopVpnInternal()
                stopSelf()
                return START_NOT_STICKY
            }
            ACTION_PAUSE -> {
                val h = handler
                if (h != null) {
                    h.pause(PAUSE_DURATION_MS)
                    updateNotification()
                    scheduleNotifRefresh(PAUSE_DURATION_MS)
                    try {
                        android.service.quicksettings.TileService.requestListeningState(
                            applicationContext,
                            android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                        )
                    } catch (_: Throwable) {}
                }
                return START_STICKY
            }
            ACTION_RESUME -> {
                val h = handler
                if (h != null) {
                    h.resume()
                    updateNotification()
                    cancelNotifRefresh()
                    try {
                        android.service.quicksettings.TileService.requestListeningState(
                            applicationContext,
                            android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                        )
                    } catch (_: Throwable) {}
                }
                return START_STICKY
            }
            else -> {
                if (!running && !isLoading) {
                    // startForeground can be refused (e.g. systemExempted on API 34+ when this app
                    // is no longer the configured VPN after a sticky restart). Stop rather than run unpromoted.
                    val promoted = try { startForegroundSafe(); true } catch (_: RuntimeException) { false }
                    if (!promoted) {
                        stopSelf()
                        return START_NOT_STICKY
                    }
                    isLoading = true
                    val gen = ++startGeneration
                    Thread { startVpnBackground(gen) }.apply { name = "guardian-start" }.start()
                }
            }
        }
        return START_STICKY
    }

    // Bumped by every start and stop. A start thread that finds a newer generation after loading the
    // list gives up, so Stop pressed while the list is loading cannot be overtaken by a late establish().
    @Volatile private var startGeneration = 0

    private fun startVpnBackground(gen: Int) {
        try {
            val blocklist = loadBlocklist()
            val userAllowed = UserRulesStore.allowed(applicationContext)
            val effective = if (userAllowed.isEmpty()) blocklist else blocklist.plusAllowed(userAllowed)
            // No uid lookup: on Android 10+ every query reaching the VPN is sent by the system DNS
            // service, so getConnectionOwnerUid names that service, not the app (seen in the emulator flow).
            val h = QueryHandler(effective, BlockMode.NULL_IP)
            if (gen != startGeneration) return

            val builder = Builder()
            builder.setSession("Guardian")
            // IPv4 DNS hijack target. Only DNS-server traffic is routed into the VPN; everything else
            // stays on the underlying network.
            builder.addAddress("10.111.222.1", 32)
            builder.addDnsServer("10.111.222.2")
            builder.addRoute("10.111.222.2", 32)
            // IPv6 DNS hijack target, a unique local address (fc00::/7, not routable on the internet). Some networks only
            // hand out IPv6 DNS servers; without this, those queries bypass Guardian. If the platform
            // refuses the IPv6 address (old build, unusual ROM), fall back to IPv4-only rather than
            // failing the whole session.
            var ipv6Configured = false
            try {
                builder.addAddress("fd00:6a72:6764::1", 128)
                ipv6Configured = true
            } catch (_: Throwable) {
                ipv6Configured = false
            }
            if (ipv6Configured) {
                try {
                    builder.addDnsServer("fd00:6a72:6764::2")
                    builder.addRoute("fd00:6a72:6764::2", 128)
                } catch (_: Throwable) {
                    // Address already added; a partially configured IPv6 side is harmless: no route
                    // is installed, so no IPv6 DNS traffic will enter the VPN from this side.
                }
            }
            builder.setMtu(1500)
            builder.setBlocking(true)

            val pfd = try { builder.establish() } catch (_: Throwable) { null }
            if (gen != startGeneration) {
                try { pfd?.close() } catch (_: Throwable) {}
                return
            }
            if (pfd == null) {
                isLoading = false
                stopSelf()
                return
            }
            tun = pfd
            tunIn = FileInputStream(pfd.fileDescriptor)
            tunOut = FileOutputStream(pfd.fileDescriptor)

            val up = Upstream(applicationContext) { sock -> protect(sock) }
            up.start()
            upstream = up

            handler = h
            activeHandler = h
            pool = Executors.newFixedThreadPool(8)
            running = true
            isRunning = true
            isLoading = false
            // Let the Quick Settings tile refresh from "Starting…" to "On" without waiting for a tap.
            // Fully qualified to avoid touching this file's imports.
            try {
                android.service.quicksettings.TileService.requestListeningState(
                    applicationContext,
                    android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                )
            } catch (_: Throwable) {}

            val t = Thread { readLoop(pfd) }
            t.name = "guardian-tun-read"
            readThread = t
            t.start()
        } catch (_: Throwable) {
            isLoading = false
            stopSelf()
        }
    }

    override fun onRevoke() {
        stopVpnInternal()
        stopSelf()
        super.onRevoke()
    }

    override fun onDestroy() {
        stopVpnInternal()
        super.onDestroy()
    }

    private fun buildNotification(): Notification {
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        if (nm.getNotificationChannel(CHANNEL_ID) == null) {
            nm.createNotificationChannel(
                NotificationChannel(CHANNEL_ID, "Guardian VPN", NotificationManager.IMPORTANCE_LOW)
            )
        }
        val stopIntent = Intent(this, GuardianVpnService::class.java).setAction(ACTION_STOP)
        val stopPi = PendingIntent.getService(
            this, 1, stopIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val builder = Notification.Builder(this, CHANNEL_ID)
            .setContentTitle("Guardian")
            .setSmallIcon(android.R.drawable.ic_lock_lock)
            .setOngoing(true)
        val h = handler
        if (h != null && h.isPaused()) {
            val fmt = SimpleDateFormat("HH:mm", Locale.getDefault())
            val until = fmt.format(Date(h.pausedUntilMs))
            builder.setContentText("Paused until $until")
            val resumeIntent = Intent(this, GuardianVpnService::class.java).setAction(ACTION_RESUME)
            val resumePi = PendingIntent.getService(
                this, 2, resumeIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
            )
            builder.addAction(0, "Resume", resumePi)
        } else {
            builder.setContentText("DNS filtering active")
        }
        builder.addAction(0, "Stop", stopPi)
        return builder.build()
    }

    private fun updateNotification() {
        try {
            val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
            nm.notify(NOTIF_ID, buildNotification())
        } catch (_: Throwable) {}
    }

    private fun scheduleNotifRefresh(delayMs: Long) {
        cancelNotifRefresh()
        val r = Runnable {
            notifRefresh = null
            updateNotification()
        }
        notifRefresh = r
        mainHandler.postDelayed(r, delayMs)
    }

    private fun cancelNotifRefresh() {
        notifRefresh?.let { mainHandler.removeCallbacks(it) }
        notifRefresh = null
    }

    private fun startForegroundSafe() {
        val notif = buildNotification()
        if (Build.VERSION.SDK_INT >= 34) {
            startForeground(NOTIF_ID, notif, ServiceInfo.FOREGROUND_SERVICE_TYPE_SYSTEM_EXEMPTED)
        } else if (Build.VERSION.SDK_INT >= 29) {
            startForeground(NOTIF_ID, notif, 0)
        } else {
            startForeground(NOTIF_ID, notif)
        }
    }

    private fun readLoop(pfd: ParcelFileDescriptor) {
        val buf = ByteArray(32767)
        val input = tunIn ?: return
        while (running) {
            val n = try { input.read(buf) } catch (_: Throwable) { if (!running) break else continue }
            if (n <= 0) continue
            val copy = buf.copyOf(n)
            val exec = pool ?: break
            try { exec.execute { processPacket(copy) } } catch (_: Throwable) {}
        }
    }

    private fun processPacket(packet: ByteArray) {
        val h = handler ?: return
        when (val act = h.handle(packet, packet.size)) {
            is Action.Respond -> writeTun(act.packet)
            is Action.Forward -> {
                val exec = pool ?: return
                try {
                    exec.execute {
                        val ans = upstream?.sendQuery(act.payload)
                        if (ans == null) {
                            h.onUpstreamFailed(act.query)
                        } else {
                            val resp = h.onUpstreamAnswer(act.query, ans, ans.size)
                            if (resp != null) writeTun(resp)
                        }
                    }
                } catch (_: Throwable) {}
            }
            Action.Drop -> {}
        }
    }

    private fun writeTun(packet: ByteArray) {
        val out = tunOut ?: return
        synchronized(writeLock) {
            try { out.write(packet) } catch (_: Throwable) {}
        }
    }

    private fun stopVpnInternal() {
        cancelNotifRefresh()
        startGeneration++
        running = false
        isRunning = false
        isLoading = false
        try { tun?.close() } catch (_: Throwable) {}
        try { tunIn?.close() } catch (_: Throwable) {}
        try { tunOut?.close() } catch (_: Throwable) {}
        tun = null; tunIn = null; tunOut = null
        try { readThread?.join(1000) } catch (_: Throwable) {}
        readThread = null
        try { pool?.shutdownNow() } catch (_: Throwable) {}
        pool = null
        try { upstream?.stop() } catch (_: Throwable) {}
        upstream = null
        handler = null
        activeHandler = null

        // Let the Quick Settings tile flip back to "Off" without waiting for a tap.
        try {
            android.service.quicksettings.TileService.requestListeningState(
                applicationContext,
                android.content.ComponentName(applicationContext, GuardianTileService::class.java)
            )
        } catch (_: Throwable) {}
    }

    private fun loadBlocklist(): Blocklist = loadBlocklistStatic(applicationContext)

    companion object {
        const val ACTION_START = "org.secureinchat.guardian.action.START"
        const val ACTION_STOP = "org.secureinchat.guardian.action.STOP"
        const val ACTION_PAUSE = "org.secureinchat.guardian.action.PAUSE"
        const val ACTION_RESUME = "org.secureinchat.guardian.action.RESUME"
        const val PAUSE_DURATION_MS = 5 * 60 * 1000L
        private const val CHANNEL_ID = "guardian_vpn"
        private const val NOTIF_ID = 1
        private val LIST_ASSETS = listOf("blocklist-hagezi-light.txt", "blocklist-extra.txt")
        private val rulesLock = Any()

        @Volatile var isRunning: Boolean = false
            private set

        @Volatile var isLoading: Boolean = false
            private set

        @Volatile var activeHandler: QueryHandler? = null
            private set

        // The bundled lists never change while the app runs, so parse them once per process.
        @Volatile private var baseCache: Blocklist? = null

        private fun loadBlocklistStatic(ctx: Context): Blocklist {
            baseCache?.let { return it }
            return try {
                val text = LIST_ASSETS.joinToString("\n") { name ->
                    ctx.assets.open(name).use { it.bufferedReader().readText() }
                }
                Blocklist.parse(text).also { baseCache = it }
            } catch (_: Throwable) {
                Blocklist.parse("")
            }
        }

        fun applyUserRules(ctx: Context) {
            val h = activeHandler ?: return
            Thread {
                synchronized(rulesLock) {
                    try {
                        val base = loadBlocklistStatic(ctx)
                        val userAllowed = UserRulesStore.allowed(ctx)
                        val effective = if (userAllowed.isEmpty()) base else base.plusAllowed(userAllowed)
                        h.blocklist = effective
                    } catch (_: Throwable) {}
                }
            }.apply { name = "guardian-rules" }.start()
        }
    }
}

    override fun onResume() { super.onResume(); uiHandler.post(refreshTick) }
    override fun onPause() { super.onPause(); uiHandler.removeCallbacks(refreshTick) }

    override fun onDestroy() {
        try { cm?.unregisterNetworkCallback(pdnsCallback) } catch (_: Throwable) {}
        super.onDestroy()
    }

    private fun registerPdnsCallback() {
        try {
            val c = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
            cm = c
            val req = NetworkRequest.Builder()
                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
                .addCapability(NetworkCapabilities.NET_CAPABILITY_NOT_VPN)
                .build()
            c.registerNetworkCallback(req, pdnsCallback)
        } catch (_: Throwable) {}
    }

    private fun requestNotificationPermissionIfNeeded() {
        if (Build.VERSION.SDK_INT >= 33) {
            if (checkSelfPermission(android.Manifest.permission.POST_NOTIFICATIONS)
                != PackageManager.PERMISSION_GRANTED) {
                requestPermissions(arrayOf(android.Manifest.permission.POST_NOTIFICATIONS), REQ_NOTIF)
            }
        }
    }

    private fun onToggleClicked() {
        if (GuardianVpnService.isLoading) return
        if (GuardianVpnService.isRunning) {
            // Plain startService: the stop path never calls startForeground, and a
            // startForegroundService() call that is not followed by startForeground() within a
            // few seconds crashes with ForegroundServiceDidNotStartInTimeException.
            val i = Intent(this, GuardianVpnService::class.java).setAction(GuardianVpnService.ACTION_STOP)
            startService(i)
        } else {
            val prep = VpnService.prepare(this)
            if (prep != null) {
                @Suppress("DEPRECATION")
                startActivityForResult(prep, REQ_VPN_PREPARE)
            } else {
                startVpnService()
            }
        }
    }

    private fun onPauseResumeClicked() {
        val h = GuardianVpnService.activeHandler ?: return
        val action = if (h.isPaused()) GuardianVpnService.ACTION_RESUME else GuardianVpnService.ACTION_PAUSE
        val i = Intent(this, GuardianVpnService::class.java).setAction(action)
        startService(i)
        // The service handles the action asynchronously; refresh soon so the button/status catch up.
        uiHandler.postDelayed({ refreshUi() }, 200)
    }

    @Deprecated("Deprecated in Activity, unavoidable without AndroidX")
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        @Suppress("DEPRECATION")
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode == REQ_VPN_PREPARE && resultCode == RESULT_OK) {
            startVpnService()
        }
    }

    private fun startVpnService() {
        val i = Intent(this, GuardianVpnService::class.java).setAction(GuardianVpnService.ACTION_START)
        startForegroundService(i)
    }

    private fun refreshUi() {
        val loading = GuardianVpnService.isLoading
        val running = GuardianVpnService.isRunning
        val h = GuardianVpnService.activeHandler
        val paused = running && h != null && h.isPaused()

        setIfChanged(statusView, when {
            loading -> txt("Loading list…", "正在加载名单…")
            paused -> {
                val fmt = SimpleDateFormat("HH:mm", Locale.getDefault())
                val until = fmt.format(Date(h!!.pausedUntilMs))
                txt("Status: Paused until $until", "状态：已暂停至 $until")
            }
            running -> txt("Status: Running", "状态：运行中")
            else -> txt("Status: Stopped", "状态：已停止")
        })

        toggleBtn.isEnabled = !loading
        setIfChanged(toggleBtn, if (running) txt("Stop", "停止") else txt("Start", "启动"))

        if (running && !loading) {
            if (pauseBtn.visibility != View.VISIBLE) pauseBtn.visibility = View.VISIBLE
            setIfChanged(
                pauseBtn,
                if (paused) txt("Resume", "恢复") else txt("Pause 5 min", "暂停 5 分钟")
            )
        } else if (pauseBtn.visibility != View.GONE) {
            pauseBtn.visibility = View.GONE
        }

        if (h == null) {
            setIfChanged(statsView, "")
            setIfChanged(rulesView, "")
            setIfChanged(logView, "")
            renderAppList(null)
            renderAllowedList()
        } else {
            val s = h.snapshot()
            setIfChanged(statsView, txt(
                "Queries: ${s.queries}  Blocked: ${s.blocked}  Forwarded: ${s.forwarded}\n" +
                "Answered: ${s.answered}  Failed: ${s.failed}  Malformed: ${s.malformed}",
                "查询: ${s.queries}  拦截: ${s.blocked}  转发: ${s.forwarded}\n" +
                "应答: ${s.answered}  失败: ${s.failed}  畸形: ${s.malformed}"
            ))
            val userAllowed = UserRulesStore.allowed(this).size
            setIfChanged(rulesView, txt(
                "Rules: ${h.blocklist.blockedCount} blocked, $userAllowed allowed by you",
                "规则:拦截 ${h.blocklist.blockedCount},你放行 $userAllowed"
            ))
            renderLog(h.recent())
            renderAppList(h)
            renderAllowedList()
        }

        if (pdnsOn.isNotEmpty()) {
            if (hintView.visibility != View.VISIBLE) hintView.visibility = View.VISIBLE
            setIfChanged(hintView, txt(
                "Private DNS is enabled. Some queries may bypass Guardian.",
                "私人 DNS 开启时部分查询可能不经过 Guardian"
            ))
        } else if (hintView.visibility != View.GONE) {
            hintView.visibility = View.GONE
        }
    }

    private fun renderLog(entries: List<LogEntry>) {
        val sb = SpannableStringBuilder()
        val fmt = SimpleDateFormat("HH:mm:ss", Locale.getDefault())
        // Last 10 only: a glance at what is happening; the per-domain view below is what people act on.
        for (e in entries.take(10)) {
            val start = sb.length
            val tag = when {
                e.blocked -> txt("BLOCK", "拦截")
                e.paused -> txt("PAUSE", "暂停")
                else -> txt("ALLOW", "放行")
            }
            sb.append(fmt.format(Date(e.timeMs)))
                .append(' ').append(tag).append(' ')
                .append(e.name).append(" (type ").append(e.qtype.toString()).append(")\n")
            if (e.blocked) {
                sb.setSpan(
                    ForegroundColorSpan(0xFFFF4444.toInt()),
                    start, sb.length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE
                )
            } else if (e.paused) {
                sb.setSpan(
                    ForegroundColorSpan(0xFFFFA500.toInt()),
                    start, sb.length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE
                )
            }
        }
        setIfChanged(logView, sb)
    }

    // "Blocked most often": the domains blocked in this session, each with an Allow button for false
    // positives. Not per app: on Android 10+ apps resolve names through the system DNS service, so every
    // query reaching the VPN comes from that service's uid (1051 on the emulator), not from the app.
    private fun renderAppList(h: QueryHandler?) {
        val top = h?.tally?.snapshot(1)?.firstOrNull()?.topBlocked ?: emptyList()
        val sig = top.joinToString(";") { "${it.first}:${it.second}" }
        if (sig == appListSignature) return
        // Counts change with every DNS query. Rebuild the list at most every 5 s unless a tap reset the
        // signature, so background traffic does not re-lay out the screen every second.
        val now = android.os.SystemClock.uptimeMillis()
        if (appListSignature.isNotEmpty() && now - appListBuiltAt < 5000) return
        appListBuiltAt = now
        appListSignature = sig
        appListContainer.removeAllViews()
        if (h == null) return
        val density = resources.displayMetrics.density
        val vpad = (4 * density).toInt()
        if (top.isEmpty()) {
            appListContainer.addView(TextView(this).apply {
                textSize = 12f
                text = txt("(nothing blocked yet)", "(还没有拦截)")
                setPadding(0, vpad, 0, vpad)
            })
            return
        }
        for (d in top) {
            val domRow = LinearLayout(this).apply {
                orientation = LinearLayout.HORIZONTAL
                setPadding(0, vpad, 0, vpad)
            }
            val dom = TextView(this).apply {
                textSize = 13f
                text = "${d.first} (${d.second})"
                layoutParams = LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f)
            }
            val allowBtn = Button(this).apply {
                text = txt("Allow", "放行")
                textSize = 12f
                setOnClickListener {
                    UserRulesStore.add(this@MainActivity, d.first)
                    GuardianVpnService.applyUserRules(applicationContext)
                    appListSignature = ""
                    allowedSignature = ""
                    uiHandler.post { refreshUi() }
                }
            }
            domRow.addView(dom)
            domRow.addView(allowBtn)
            appListContainer.addView(domRow)
        }
    }

    private fun renderAllowedList() {
        val allowed = UserRulesStore.allowed(this).sorted()
        val sig = allowed.joinToString(",")
        if (sig == allowedSignature) return
        allowedSignature = sig
        allowedListContainer.removeAllViews()
        val density = resources.displayMetrics.density
        val vpad = (4 * density).toInt()
        if (allowed.isEmpty()) {
            allowedListContainer.addView(TextView(this).apply {
                textSize = 12f
                text = txt("(none)", "(无)")
                setPadding(0, vpad, 0, vpad)
            })
            return
        }
        for (d in allowed) {
            val row = LinearLayout(this).apply {
                orientation = LinearLayout.HORIZONTAL
                setPadding(0, vpad, 0, vpad)
            }
            val dom = TextView(this).apply {
                textSize = 13f
                text = d
                layoutParams = LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f)
            }
            val removeBtn = Button(this).apply {
                text = txt("Remove", "移除")
                textSize = 12f
                setOnClickListener {
                    UserRulesStore.remove(this@MainActivity, d)
                    GuardianVpnService.applyUserRules(applicationContext)
                    appListSignature = ""
                    allowedSignature = ""
                    uiHandler.post { refreshUi() }
                }
            }
            row.addView(dom)
            row.addView(removeBtn)
            allowedListContainer.addView(row)
        }
    }

    // The screen refreshes every second; setting identical text still re-lays out the view, which keeps
    // the window from ever being idle (wasted work, and UI tests cannot read the screen).
    private fun setIfChanged(view: TextView, value: CharSequence) {
        if (view.text.toString() != value.toString()) view.text = value
    }

    private fun txt(en: String, zh: String): String =
        if (Locale.getDefault().language == "zh") zh else en

    companion object {
        private const val REQ_VPN_PREPARE = 100
        private const val REQ_NOTIF = 101
    }
