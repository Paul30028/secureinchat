## 要做的事
1. 新文件 `apps/android-client/src/i18n/ns/joinGroup.ts`：
   `import type { Messages } from "../index";`，导出 `export const zh: Messages`、`export const en: Messages`、`export const es: Messages`。
   三个对象的键完全一致；键一律以 `joinGroup.` 开头（如 `joinGroup.title`）；参数用 `{name}` 占位，三种语言占位符一致。
2. 改写 `apps/android-client/src/screens/JoinGroupScreen.tsx`：所有给人看的文字（包括 aria-label、placeholder、title、alt、错误提示、按钮文字）改成 `t("joinGroup.xxx")`，`const t = useT();` 从 `"../i18n"` 导入。
   - 界面结构、样式、props、行为一律不变。不在组件外（模块顶层常量）调用 t；需要的话把常量挪进组件或改成键名。
   - 中文词条必须与原文**逐字相同**（现有测试按中文断言，必须原样通过）。
   - 日期、数字用 `Intl` 格式化时，locale 用 `useLocale()`（同样从 "../i18n" 导入）。
3. 英语自然简洁；西班牙语用中性拉美西语。语气平实准确，不夸大安全效果（不要写「完全安全」「无法破解」之类）。
4. 不加任何依赖；TypeScript strict，不用 any；保留现有注释。
5. 不要输出测试文件以外的其他文件；不要修改现有测试。

## 翻译层 API（apps/android-client/src/i18n/index.tsx，只读）
- `useT(): (key: string, params?: Record<string, string | number>) => string`
- `useLocale(): "zh" | "en" | "es"`

## 现有源码：apps/android-client/src/screens/JoinGroupScreen.tsx
import { useState } from "react";
import { colors, touchTarget, Button, Avatar } from "@secureinchat/ui";

export interface JoinGroupScreenProps {
  onSubmitInviteCode: (raw: string) => void;
  onCreateGroup: () => void;
  onBack: () => void;
  onScanQr: () => void;
}

/** 启动页。对应设计稿"01 启动"+"02 公告与进入"的简化合并——公告卡片、
 *  今日经文这些内容型模块不在这次范围内，先把"输入邀请码进群"这条主干打通。 */
export function JoinGroupScreen({ onSubmitInviteCode, onCreateGroup, onBack, onScanQr }: JoinGroupScreenProps) {
  const [code, setCode] = useState("");

  return (
    <div
      style={{
        minHeight: "100vh",
        background: colors.ivory,
        display: "flex",
        flexDirection: "column",
        alignItems: "center",
        justifyContent: "center",
        gap: 16,
        padding: 24,
      }}
    >
      <button
        onClick={onBack}
        aria-label="返回"
        style={{
          alignSelf: "flex-start",
          background: "transparent",
          border: "none",
          boxShadow: "none",
          color: colors.deepInkGreen,
          fontSize: 14,
          cursor: "pointer",
          minHeight: touchTarget.minDp,
        }}
      >
        ← 返回
      </button>
      <Avatar name="邀群密聊" size="large" />
      <span style={{ fontSize: 20, fontWeight: 500, color: colors.deepInkGreen }}>邀群密聊</span>
      <span style={{ fontSize: 12, color: "#8A8A82" }}>安全 · 隐私 · 专属</span>

      <div style={{ width: "100%", maxWidth: 320, marginTop: 24, display: "flex", flexDirection: "column", gap: 12 }}>
        <input
          type="text"
          value={code}
          onChange={(e) => setCode(e.target.value)}
          placeholder="输入邀请码"
          aria-label="邀请码输入框"
          style={{
            minHeight: touchTarget.minDp,
            border: `0.5px solid ${colors.sageMint}`,
            borderRadius: 14,
            padding: "0 16px",
            fontSize: 14,
            background: colors.ivory,
            color: colors.textPrimary,
            boxShadow: "none",
          }}
        />
        {/* 扫码是主路径——小团体多半是当面加入，扫一下比手动粘贴一长串快得多 */}
        <Button variant="primary" onClick={onScanQr} style={{ width: "100%" }}>
          扫描二维码加入
        </Button>

        <Button
          variant="secondary"
          onClick={() => onSubmitInviteCode(code)}
          disabled={code.trim().length === 0}
          style={{ width: "100%" }}
        >
          用邀请码加入
        </Button>
        <Button variant="secondary" onClick={onCreateGroup} style={{ width: "100%" }}>
          创建群聊
        </Button>

      </div>
    </div>
  );
}
