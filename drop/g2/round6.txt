// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian

import android.content.Context
import android.net.ConnectivityManager
import android.net.LinkProperties
import android.net.Network
import android.net.NetworkCapabilities
import android.net.NetworkRequest
import java.net.DatagramPacket
import java.net.DatagramSocket
import java.net.InetAddress
import java.util.concurrent.ConcurrentHashMap

class Upstream(
    context: Context,
    private val protect: (DatagramSocket) -> Boolean
) {
    private val cm = context.applicationContext
        .getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager

    private val networkServers = ConcurrentHashMap<Network, List<InetAddress>>()
    @Volatile private var started = false

    private val callback = object : ConnectivityManager.NetworkCallback() {
        override fun onAvailable(network: Network) { refresh(network) }
        override fun onLinkPropertiesChanged(network: Network, lp: LinkProperties) { refresh(network) }
        override fun onLost(network: Network) { networkServers.remove(network) }
    }

    fun start() {
        if (started) return
        started = true
        try {
            val req = NetworkRequest.Builder()
                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
                .addCapability(NetworkCapabilities.NET_CAPABILITY_NOT_VPN)
                .build()
            cm.registerNetworkCallback(req, callback)
        } catch (_: Throwable) {
            started = false
        }
    }

    fun stop() {
        if (!started) return
        started = false
        try { cm.unregisterNetworkCallback(callback) } catch (_: Throwable) {}
        networkServers.clear()
    }

    fun dnsServers(): List<InetAddress> {
        val out = LinkedHashSet<InetAddress>()
        for (servers in networkServers.values) out.addAll(servers)
        return out.toList()
    }

    fun sendQuery(payload: ByteArray): ByteArray? {
        if (networkServers.isEmpty()) return null
        for (server in dnsServers()) {
            val r = queryOne(server, payload) ?: continue
            return r
        }
        return null
    }

    private fun refresh(network: Network) {
        try {
            val lp: LinkProperties = cm.getLinkProperties(network) ?: return
            val list = lp.dnsServers ?: emptyList()
            if (list.isEmpty()) networkServers.remove(network)
            else networkServers[network] = ArrayList(list)
        } catch (_: Throwable) {}
    }

    private fun queryOne(server: InetAddress, payload: ByteArray): ByteArray? {
        var sock: DatagramSocket? = null
        try {
            sock = DatagramSocket()
            if (!protect(sock)) return null
            sock.soTimeout = 4000
            sock.send(DatagramPacket(payload, payload.size, server, 53))
            val buf = ByteArray(4096)
            val recv = DatagramPacket(buf, buf.size)
            sock.receive(recv)
            return buf.copyOf(recv.length)
        } catch (_: Throwable) {
            return null
        } finally {
            try { sock?.close() } catch (_: Throwable) {}
        }
    }
}

// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.content.Context
import android.content.Intent
import android.content.pm.ServiceInfo
import android.net.VpnService
import android.os.Build
import android.os.ParcelFileDescriptor
import java.io.FileInputStream
import java.io.FileOutputStream
import java.util.concurrent.ExecutorService
import java.util.concurrent.Executors
import org.secureinchat.guardian.dns.Action
import org.secureinchat.guardian.dns.Blocklist
import org.secureinchat.guardian.dns.QueryHandler

class GuardianVpnService : VpnService() {

    private var tun: ParcelFileDescriptor? = null
    private var tunIn: FileInputStream? = null
    private var tunOut: FileOutputStream? = null
    private val writeLock = Any()
    private var readThread: Thread? = null
    private var pool: ExecutorService? = null
    @Volatile private var running = false
    private var handler: QueryHandler? = null
    private var upstream: Upstream? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> {
                stopVpnInternal()
                stopSelf()
                return START_NOT_STICKY
            }
            else -> {
                if (!running) {
                    startForegroundSafe()
                    startVpnInternal()
                }
            }
        }
        return START_STICKY
    }

    override fun onRevoke() {
        stopVpnInternal()
        stopSelf()
        super.onRevoke()
    }

    override fun onDestroy() {
        stopVpnInternal()
        super.onDestroy()
    }

    private fun startForegroundSafe() {
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        if (nm.getNotificationChannel(CHANNEL_ID) == null) {
            nm.createNotificationChannel(
                NotificationChannel(CHANNEL_ID, "Guardian VPN", NotificationManager.IMPORTANCE_LOW)
            )
        }
        val stopIntent = Intent(this, GuardianVpnService::class.java).setAction(ACTION_STOP)
        val pi = PendingIntent.getService(
            this, 1, stopIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val notif: Notification = Notification.Builder(this, CHANNEL_ID)
            .setContentTitle("Guardian")
            .setContentText("DNS filtering active")
            .setSmallIcon(android.R.drawable.ic_lock_lock)
            .addAction(0, "Stop", pi)
            .setOngoing(true)
            .build()
        if (Build.VERSION.SDK_INT >= 34) {
            startForeground(NOTIF_ID, notif, ServiceInfo.FOREGROUND_SERVICE_TYPE_SYSTEM_EXEMPTED)
        } else if (Build.VERSION.SDK_INT >= 29) {
            startForeground(NOTIF_ID, notif, 0)
        } else {
            startForeground(NOTIF_ID, notif)
        }
    }

    private fun startVpnInternal() {
        val blocklist = loadBlocklist()
        val h = QueryHandler(blocklist)
        handler = h
        val builder = Builder()
        builder.setSession("Guardian")
        builder.addAddress("10.111.222.1", 32)
        builder.addDnsServer("10.111.222.2")
        builder.addRoute("10.111.222.2", 32)
        builder.setMtu(1500)
        builder.setBlocking(true)

        val pfd = try { builder.establish() } catch (_: Throwable) { null }
        if (pfd == null) {
            stopVpnInternal()
            stopSelf()
            return
        }
        tun = pfd
        tunIn = FileInputStream(pfd.fileDescriptor)
        tunOut = FileOutputStream(pfd.fileDescriptor)

        val up = Upstream(applicationContext) { sock -> protect(sock) }
        up.start()
        upstream = up

        running = true
        isRunning = true
        activeHandler = h
        pool = Executors.newFixedThreadPool(8)
        val t = Thread { readLoop(pfd) }
        t.name = "guardian-tun-read"
        readThread = t
        t.start()
    }

    private fun readLoop(pfd: ParcelFileDescriptor) {
        val buf = ByteArray(32767)
        val input = tunIn ?: return
        while (running) {
            val n = try { input.read(buf) } catch (_: Throwable) { if (!running) break else continue }
            if (n <= 0) continue
            val copy = buf.copyOf(n)
            val exec = pool ?: break
            try { exec.execute { processPacket(copy) } } catch (_: Throwable) {}
        }
    }

    private fun processPacket(packet: ByteArray) {
        val h = handler ?: return
        when (val act = h.handle(packet, packet.size)) {
            is Action.Respond -> writeTun(act.packet)
            is Action.Forward -> {
                val exec = pool ?: return
                try {
                    exec.execute {
                        val ans = upstream?.sendQuery(act.payload)
                        if (ans == null) {
                            h.onUpstreamFailed(act.query)
                        } else {
                            val resp = h.onUpstreamAnswer(act.query, ans, ans.size)
                            if (resp != null) writeTun(resp)
                        }
                    }
                } catch (_: Throwable) {}
            }
            Action.Drop -> {}
        }
    }

    private fun writeTun(packet: ByteArray) {
        val out = tunOut ?: return
        synchronized(writeLock) {
            try { out.write(packet) } catch (_: Throwable) {}
        }
    }

    private fun stopVpnInternal() {
        running = false
        isRunning = false
        try { tun?.close() } catch (_: Throwable) {}
        try { tunIn?.close() } catch (_: Throwable) {}
        try { tunOut?.close() } catch (_: Throwable) {}
        tun = null; tunIn = null; tunOut = null
        try { readThread?.join(1000) } catch (_: Throwable) {}
        readThread = null
        try { pool?.shutdownNow() } catch (_: Throwable) {}
        pool = null
        try { upstream?.stop() } catch (_: Throwable) {}
        upstream = null
        handler = null
        activeHandler = null
    }

    private fun loadBlocklist(): Blocklist {
        return try {
            assets.open("blocklist.txt").use { input ->
                Blocklist.parse(input.bufferedReader().readText())
            }
        } catch (_: Throwable) {
            Blocklist.parse("")
        }
    }

    companion object {
        const val ACTION_START = "org.secureinchat.guardian.action.START"
        const val ACTION_STOP = "org.secureinchat.guardian.action.STOP"
        private const val CHANNEL_ID = "guardian_vpn"
        private const val NOTIF_ID = 1

        @Volatile var isRunning: Boolean = false
            private set

        @Volatile var activeHandler: QueryHandler? = null
            private set
    }
}

// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian

import android.app.Activity
import android.content.Context
import android.content.Intent
import android.content.pm.PackageManager
import android.graphics.Typeface
import android.net.ConnectivityManager
import android.net.LinkProperties
import android.net.Network
import android.net.NetworkCapabilities
import android.net.NetworkRequest
import android.net.VpnService
import android.os.Build
import android.os.Bundle
import android.os.Handler
import android.os.Looper
import android.text.SpannableStringBuilder
import android.text.Spanned
import android.text.style.ForegroundColorSpan
import android.view.View
import android.widget.Button
import android.widget.LinearLayout
import android.widget.ScrollView
import android.widget.TextView
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale
import java.util.concurrent.ConcurrentHashMap
import org.secureinchat.guardian.dns.LogEntry

class MainActivity : Activity() {

    private lateinit var statusView: TextView
    private lateinit var statsView: TextView
    private lateinit var logView: TextView
    private lateinit var hintView: TextView
    private lateinit var toggleBtn: Button

    private val uiHandler = Handler(Looper.getMainLooper())
    private val refreshTick = object : Runnable {
        override fun run() {
            refreshUi()
            uiHandler.postDelayed(this, 1000)
        }
    }

    private val pdnsOn = ConcurrentHashMap.newKeySet<Network>()
    private var cm: ConnectivityManager? = null
    private val pdnsCallback = object : ConnectivityManager.NetworkCallback() {
        override fun onLinkPropertiesChanged(network: Network, lp: LinkProperties) {
            if (Build.VERSION.SDK_INT >= 28 && lp.isPrivateDnsActive) pdnsOn.add(network)
            else pdnsOn.remove(network)
        }
        override fun onLost(network: Network) { pdnsOn.remove(network) }
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        buildUi()
        requestNotificationPermissionIfNeeded()
        registerPdnsCallback()
    }

    override fun onResume() { super.onResume(); uiHandler.post(refreshTick) }
    override fun onPause() { super.onPause(); uiHandler.removeCallbacks(refreshTick) }

    override fun onDestroy() {
        try { cm?.unregisterNetworkCallback(pdnsCallback) } catch (_: Throwable) {}
        super.onDestroy()
    }

    private fun registerPdnsCallback() {
        try {
            val c = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
            cm = c
            val req = NetworkRequest.Builder()
                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
                .addCapability(NetworkCapabilities.NET_CAPABILITY_NOT_VPN)
                .build()
            c.registerNetworkCallback(req, pdnsCallback)
        } catch (_: Throwable) {}
    }

    private fun requestNotificationPermissionIfNeeded() {
        if (Build.VERSION.SDK_INT >= 33) {
            if (checkSelfPermission(android.Manifest.permission.POST_NOTIFICATIONS)
                != PackageManager.PERMISSION_GRANTED) {
                requestPermissions(arrayOf(android.Manifest.permission.POST_NOTIFICATIONS), REQ_NOTIF)
            }
        }
    }

    private fun buildUi() {
        val density = resources.displayMetrics.density
        val pad = (16 * density).toInt()

        val root = LinearLayout(this)
        root.orientation = LinearLayout.VERTICAL
        root.setPadding(pad, pad, pad, pad)

        root.addView(TextView(this).apply {
            text = "Guardian"
            textSize = 22f
        })

        statusView = TextView(this).apply { textSize = 16f }
        root.addView(statusView)

        toggleBtn = Button(this).apply {
            setOnClickListener { onToggleClicked() }
        }
        root.addView(toggleBtn)

        statsView = TextView(this).apply {
            textSize = 14f
            typeface = Typeface.MONOSPACE
            setPadding(0, pad / 2, 0, 0)
        }
        root.addView(statsView)

        hintView = TextView(this).apply {
            textSize = 13f
            setTextColor(0xFFFF8800.toInt())
            visibility = View.GONE
            setPadding(0, pad / 2, 0, 0)
        }
        root.addView(hintView)

        val privacy = TextView(this).apply {
            textSize = 12f
            text = txt(
                "Only DNS queries go through Guardian. Allowed queries are sent to your current network's DNS servers. No logs are saved.",
                "只有 DNS 查询经过 Guardian。放行的查询发给你当前网络的 DNS 服务器。记录不保存。"
            )
            setPadding(0, pad / 2, 0, 0)
        }
        root.addView(privacy)

        logView = TextView(this).apply {
            textSize = 12f
            typeface = Typeface.MONOSPACE
            setPadding(0, pad, 0, 0)
        }
        root.addView(logView)

        setContentView(ScrollView(this).apply { addView(root) })
    }

    private fun onToggleClicked() {
        if (GuardianVpnService.isRunning) {
            val i = Intent(this, GuardianVpnService::class.java).setAction(GuardianVpnService.ACTION_STOP)
            startForegroundService(i)
        } else {
            val prep = VpnService.prepare(this)
            if (prep != null) {
                @Suppress("DEPRECATION")
                startActivityForResult(prep, REQ_VPN_PREPARE)
            } else {
                startVpnService()
            }
        }
    }

    @Deprecated("Deprecated in Activity, unavoidable without AndroidX")
    override fun onActivityResult(requestCode: Int, resultCode: Int, data: Intent?) {
        @Suppress("DEPRECATION")
        super.onActivityResult(requestCode, resultCode, data)
        if (requestCode == REQ_VPN_PREPARE && resultCode == RESULT_OK) {
            startVpnService()
        }
    }

    private fun startVpnService() {
        val i = Intent(this, GuardianVpnService::class.java).setAction(GuardianVpnService.ACTION_START)
        startForegroundService(i)
    }

    private fun refreshUi() {
        val running = GuardianVpnService.isRunning
        statusView.text = if (running) {
            txt("Status: Running", "状态：运行中")
        } else {
            txt("Status: Stopped", "状态：已停止")
        }
        toggleBtn.text = if (running) {
            txt("Stop", "停止")
        } else {
            txt("Start", "启动")
        }
        val h = GuardianVpnService.activeHandler
        if (h == null) {
            statsView.text = ""
            logView.text = ""
        } else {
            val s = h.snapshot()
            statsView.text = txt(
                "Queries: ${s.queries}  Blocked: ${s.blocked}  Forwarded: ${s.forwarded}\n" +
                "Answered: ${s.answered}  Failed: ${s.failed}  Malformed: ${s.malformed}",
                "查询: ${s.queries}  拦截: ${s.blocked}  转发: ${s.forwarded}\n" +
                "应答: ${s.answered}  失败: ${s.failed}  畸形: ${s.malformed}"
            )
            renderLog(h.recent())
        }
        if (pdnsOn.isNotEmpty()) {
            hintView.visibility = View.VISIBLE
            hintView.text = txt(
                "Private DNS is enabled. Some queries may bypass Guardian.",
                "私人 DNS 开启时部分查询可能不经过 Guardian"
            )
        } else {
            hintView.visibility = View.GONE
        }
    }

    private fun renderLog(entries: List<LogEntry>) {
        val sb = SpannableStringBuilder()
        val fmt = SimpleDateFormat("HH:mm:ss", Locale.getDefault())
        for (e in entries.take(50)) {
            val start = sb.length
            val tag = if (e.blocked) txt("BLOCK", "拦截") else txt("ALLOW", "放行")
            sb.append(fmt.format(Date(e.timeMs)))
                .append(' ').append(tag).append(' ')
                .append(e.name).append(" (type ").append(e.qtype.toString()).append(")\n")
            if (e.blocked) {
                sb.setSpan(
                    ForegroundColorSpan(0xFFFF4444.toInt()),
                    start, sb.length, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE
                )
            }
        }
        logView.text = sb
    }

    private fun txt(en: String, zh: String): String =
        if (Locale.getDefault().language == "zh") zh else en

    companion object {
        private const val REQ_VPN_PREPARE = 100
        private const val REQ_NOTIF = 101
    }
}
