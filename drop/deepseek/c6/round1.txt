@secureinchat/chat-core 已依赖 @secureinchat/protocol（见 package.json），buildInviteForGroup 直接用 buildSic2Invite，不需要加依赖。

parseInviteAuto 解析出的字段名沿用 buildSic2Invite 的入参名（keyMaterialB64Url、adminPublicKeyRawB64Url）。如果 protocol 里实际字段名不同，测试要跟着改——这是我对未提供源码的 protocol 包的假设。

import { buildSic2Invite, type ParsedInvite } from "@secureinchat/protocol";
import { randomUUID } from "@secureinchat/crypto-core";

/**
 * 群密钥轮换（epoch rotation）。
 *
 * 这是"移出成员"真正生效的机制。中继按设计不知道谁是成员、也不保存消息，
 * 所以没有"把某人踢出房间"这种服务端操作——即使有，被踢的人手上的群密钥
 * 依然能解密他之前收到的、以及任何仍用旧密钥加密的消息。
 *
 * 真正的做法是换钥匙：生成全新的密钥材料，epoch 加一，把新邀请码发给留下的人。
 * 之后的消息用新密钥加密，拿着旧密钥的人再也读不到。
 *
 * ⚠️ 两个必须说清楚的限制：
 * 1. 轮换**不能**让对方读不到他已经收到的历史消息——那些明文早就在他设备上了。
 *    轮换保护的是"以后的消息"。
 * 2. 谁拿到新邀请码谁就能进来。所以轮换之后要把新码只发给该留下的人，
 *    发错了等于没轮换。
 */

export interface RotationResult {
  /** 新的密钥材料（base64url） */
  keyMaterialB64Url: string;
  /** 新的 epoch，比原来大 1 */
  epoch: number;
  /** 发给留下的成员的新邀请码 */
  inviteCode: string;
}

export interface RotateGroupInput {
  groupId: string;
  groupName: string;
  /** 当前 epoch，新的会是它加一 */
  currentEpoch: number;
  /** 管理员公钥保持不变——轮换换的是群密钥，不是管理员身份 */
  adminPublicKeyRawB64Url?: string | undefined;
  serverJoinCode: string;
}

function randomKeyMaterial(): string {
  const bytes = crypto.getRandomValues(new Uint8Array(32));
  let binary = "";
  for (const b of bytes) binary += String.fromCharCode(b);
  return btoa(binary).replace(/\+/g, "-").replace(/\//g, "_").replace(/=+$/, "");
}

export function rotateGroupKey(input: RotateGroupInput): RotationResult {
  const keyMaterialB64Url = randomKeyMaterial();
  const epoch = input.currentEpoch + 1;

  const inviteCode = buildSic2Invite({
    serverJoinCode: input.serverJoinCode,
    groupId: input.groupId,
    groupName: input.groupName,
    keyMaterialB64Url,
    epoch,
    ...(input.adminPublicKeyRawB64Url
      ? { adminPublicKeyRawB64Url: input.adminPublicKeyRawB64Url }
      : {}),
  });

  return { keyMaterialB64Url, epoch, inviteCode };
}

/**
 * 收到一个同群但 epoch 更高的邀请时，应该采纳它（有人轮换了密钥）。
 * epoch 相同或更低的要忽略——否则重放一个旧邀请码就能把大家降级回旧密钥，
 * 让被移除的人重新读得到消息。
 */
export function shouldAdoptInvite(current: { groupId: string; epoch: number }, incoming: ParsedInvite): boolean {
  if (incoming.groupId !== current.groupId) return false;
  return (incoming.epoch ?? 0) > current.epoch;
}

/**
 * 为已经加入的群重新生成邀请串。
 *
 * 邀请串没有有效期，所以不需要单独保存——每次打开都用当前 JoinedGroup 里的
 * 密钥材料和 epoch 现算。轮换过密钥的话，这里自然就是最新的，旧卡片随之失效。
 *
 * serverJoinCode 缺省时随机生成 8 位大写；中继目前不校验这个字段（见
 * CreateGroupScreen 的注释），所以换个新的也无妨。
 */
export function buildInviteForGroup(
  group: {
    groupId: string;
    groupName: string;
    keyMaterialB64Url: string;
    epoch: number;
    adminPublicKeyRawB64Url?: string | undefined;
  },
  serverJoinCode?: string
): string {
  const code = serverJoinCode ?? randomUUID().slice(0, 8).toUpperCase();
  return buildSic2Invite({
    serverJoinCode: code,
    groupId: group.groupId,
    groupName: group.groupName,
    keyMaterialB64Url: group.keyMaterialB64Url,
    epoch: group.epoch,
    ...(group.adminPublicKeyRawB64Url
      ? { adminPublicKeyRawB64Url: group.adminPublicKeyRawB64Url }
      : {}),
  });
}

import { describe, expect, it } from "vitest";
import { parseInviteAuto } from "@secureinchat/protocol";
import { buildInviteForGroup, rotateGroupKey, shouldAdoptInvite } from "../src/keyRotation";

const baseGroup = {
  groupId: "group-abc",
  groupName: "测试群",
  keyMaterialB64Url: "AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA",
  epoch: 0,
  adminPublicKeyRawB64Url: "PUBKEY_RAW_B64URL",
};

describe("buildInviteForGroup", () => {
  it("round-trips through parseInviteAuto", () => {
    const code = buildInviteForGroup(baseGroup);
    const parsed = parseInviteAuto(code);
    expect(parsed.groupId).toBe(baseGroup.groupId);
    expect(parsed.groupName).toBe(baseGroup.groupName);
    expect(parsed.epoch).toBe(baseGroup.epoch);
    expect(parsed.keyMaterialB64Url).toBe(baseGroup.keyMaterialB64Url);
    expect(parsed.adminPublicKeyRawB64Url).toBe(baseGroup.adminPublicKeyRawB64Url);
  });

  it("works without an admin public key", () => {
    const noAdmin = {
      groupId: baseGroup.groupId,
      groupName: baseGroup.groupName,
      keyMaterialB64Url: baseGroup.keyMaterialB64Url,
      epoch: baseGroup.epoch,
    };
    const code = buildInviteForGroup(noAdmin);
    const parsed = parseInviteAuto(code);
    expect(parsed.groupId).toBe(baseGroup.groupId);
    expect(parsed.groupName).toBe(baseGroup.groupName);
    expect(parsed.adminPublicKeyRawB64Url).toBeUndefined();
  });

  it("accepts an explicit serverJoinCode", () => {
    const code = buildInviteForGroup(baseGroup, "ABCD1234");
    const parsed = parseInviteAuto(code);
    expect(parsed.groupId).toBe(baseGroup.groupId);
  });

  it("after rotation, the invite has a higher epoch and should be adopted", () => {
    const rotated = rotateGroupKey({
      groupId: baseGroup.groupId,
      groupName: baseGroup.groupName,
      currentEpoch: baseGroup.epoch,
      adminPublicKeyRawB64Url: baseGroup.adminPublicKeyRawB64Url,
      serverJoinCode: "ABCD1234",
    });
    const code = buildInviteForGroup({
      groupId: baseGroup.groupId,
      groupName: baseGroup.groupName,
      keyMaterialB64Url: rotated.keyMaterialB64Url,
      epoch: rotated.epoch,
      adminPublicKeyRawB64Url: baseGroup.adminPublicKeyRawB64Url,
    });
    const parsed = parseInviteAuto(code);
    expect(parsed.epoch).toBe(baseGroup.epoch + 1);
    expect(parsed.keyMaterialB64Url).toBe(rotated.keyMaterialB64Url);
    expect(
      shouldAdoptInvite({ groupId: baseGroup.groupId, epoch: baseGroup.epoch }, parsed)
    ).toBe(true);
  });

真机验证点：
管理员设备上进入群设置，能看到「邀请新成员」这一行；普通成员设备上没有（持有该群管理员私钥才显示）。

点「邀请新成员」进入邀请卡片页：群名正确、二维码可被另一台设备扫码加入、复制按钮可用。

拿着管理员的另一台设备（无该群管理员私钥）进入群设置，不出现「邀请新成员」；只在有别的群管理员私钥时也不出现——确认用的是本群 loadAdminKey 而不是全局 holdsAdminKey。

管理员执行「更换群密钥」（ConnectedDevicesScreen 里的入口，实际名称按源码为准）后，再次打开邀请卡片：邀请串里的 epoch 比旧卡片高，旧卡片扫码加入的设备读不到之后的新消息，新卡片可以。

邀请卡片没有有效期提示（expiryLabel 未传），符合"无过期时间"的设计；确认页面没有误导性的倒计时或"X 天后失效"。

群设置页返回按钮：从聊天页 → 群设置 → 邀请卡片 → 返回，应回到群设置；从群设置再返回应回到聊天页。从列表长按 → 群设置 → 邀请卡片 → 返回，回群设置 → 再返回会回到聊天页（groupInvite 不记原 from，按说明默认 "chat"）。这个后退差异需要你确认是否可接受。

已加入群但从 JoinedGroup 里找不到密钥材料（人为构造：清掉 groups:joined 后手动进入 groupInvite，或破坏数据）时，页面显示错误文本、不出现空邀请卡片。

群名被改过之后（本机 rename），邀请卡片页标题和卡片里的群名显示的是改名后的当前名字（JoinedGroup 里的 groupName）。

点复制邀请码后，系统剪贴板里是完整 SIC2 串，另一台设备粘贴能解析入群。

buildInviteForGroup 不传 serverJoinCode 时，随机生成的 8 位大写码不影响解析（中继不校验）；传显式 serverJoinCode 时也照常工作。
