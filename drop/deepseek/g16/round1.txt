    // G12: wipe the durable block history. The in-session "Blocked: N" counter above keeps its
    // value; only the "Blocked today / last 7 days" row reads from StatsStore and will show 0.
    private fun onClearHistoryClicked() {
        StatsStore.clear(this)
        historySignature = ""
        uiHandler.post { refreshUi() }
    }

    // G16: return the first candidate intent the system can actually handle, or null. resolveActivity
    // covers both cases that make a settings screen unreachable: the API level is too low, or the
    // OEM removed that screen. Wrapped in try/catch because a malformed action string would make
    // resolveActivity throw on some old platforms.
    private fun firstResolvable(candidates: List<Intent>): Intent? {
        for (i in candidates) {
            try {
                if (i.resolveActivity(packageManager) != null) return i
            } catch (_: Throwable) {}
        }
        return null
    }

    // G16: one privacy row is a title, a one-sentence explanation, and either a button that opens
    // the first resolvable settings screen or (if none resolve) a plain hint telling the user where
    // to look in Settings. A row with an unreachable screen still shows its explanation; only the
    // button is hidden, so the checkup never lies about what it can do.
    private fun addPrivacyRow(
        parent: LinearLayout,
        pad: Int,
        titleEn: String,
        titleZh: String,
        descEn: String,
        descZh: String,
        candidates: List<Intent>,
        buttonEn: String,
        buttonZh: String,
        fallbackEn: String,
        fallbackZh: String
    ) {
        parent.addView(TextView(this).apply {
            textSize = 14f
            text = txt(titleEn, titleZh)
            setPadding(0, pad, 0, pad / 4)
        })
        parent.addView(TextView(this).apply {
            textSize = 12f
            text = txt(descEn, descZh)
        })
        val target = firstResolvable(candidates)
        if (target != null) {
            parent.addView(Button(this).apply {
                text = txt(buttonEn, buttonZh)
                textSize = 12f
                setOnClickListener {
                    try { startActivity(target) } catch (_: ActivityNotFoundException) {}
                }
            })
        } else {
            parent.addView(TextView(this).apply {
                textSize = 12f
                text = txt(fallbackEn, fallbackZh)
                setPadding(0, pad / 4, 0, 0)
            })
        }
    }

没有新增权限、没有新增依赖。All three buttons only call startActivity(Intent); no data is read or written. If GMS is absent on the emulator, row 1 falls back to Settings.ACTION_PRIVACY_SETTINGS; if that also does not exist (API < 31 AOSP image), the button is hidden and the "Search Settings for Ads" hint appears instead.

模拟器测试建议

三行标题存在:English locale 下断言界面上出现 "Delete your advertising ID"、"Precise location"、"Auto-revoke unused apps" 三段文字(可以用 uiautomator dump 或 UI Automator 的 textContains)。分区标题 "Privacy checkup" 也一并断言。

第 1 个按钮跳转:用文字 "Open ad settings" 找按钮(English locale)。Google APIs 镜像上有 GMS,com.google.android.gms.settings.ADS_PRIVACY 会解析成功,点击后前台会切到 Google Play services 的 com.google.android.gms 的 Ads 页面。断言:

adb shell dumpsys activity activities | grep -E 'mResumedActivity|topResumedActivity' 里的包名不是 org.secureinchat.guardian。

不硬编码具体包名。GMS 版本不同,前台 Activity 可能是 com.google.android.gms/.ads.settings.AdsSettingsActivity,也可能是别的;只断言"不是 Guardian"。

返回:adb shell input keyevent 4(返回键)。再查 dumpsys activity activities,前台 Activity 应是 org.secureinchat.guardian/.MainActivity。

回退路径(可选,针对 AOSP 无 GMS 镜像):同一个按钮在无 GMS 镜像上会走 Settings.ACTION_PRIVACY_SETTINGS,前台变成 com.android.settings;如果连这个也没有(API < 31 的 AOSP 镜像),按钮不出现、"Search Settings for \"Ads\"" 出现。建议测试用条件判断:如果 "Open ad settings" 存在就断言 "Search Settings for \"Ads\"" 不存在,反之亦然,避免在两种镜像上都要维护两套断言。

按钮 resolveActivity 为 null 时不应出现按钮:可以额外增加一条检查 —— 在已知没有 GMS 的 AOSP 镜像(API 29)上,断言界面上不存在 "Open ad settings" 按钮,但 "Delete your advertising ID" 标题仍在。

一个我没有把握、需要你在真机核实的点:com.google.android.gms.settings.ADS_PRIVACY 是 Google 未公开文档化的 intent,在部分 GMS 版本上可能被标为 exported=false 或需要额外权限,导致 resolveActivity 返回 null。代码在这种情况下会自动落到 Settings.ACTION_PRIVACY_SETTINGS,不会崩溃、也不会显示假按钮,但界面上用户看到的是隐私设置页而不是广告设置页。如果真机上发现这个行为,可以考虑是否需要在按下后弹一句提示告诉用户广告设置在哪里;我没有主动加,因为那会引入额外的 UI 分支,而你要求的是"打不开的不显示按钮"。
