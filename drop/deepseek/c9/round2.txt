function formatDateForFilename(now: Date): string {
  const y = now.getFullYear();
  const m = String(now.getMonth() + 1).padStart(2, "0");
  const d = String(now.getDate()).padStart(2, "0");
  return `${y}${m}${d}`;
}

function downloadText(filename: string, text: string): void {
  const blob = new Blob([text], { type: "application/json" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  // 延迟释放：某些浏览器在 click 后同步 revoke 会导致下载失败
  setTimeout(() => URL.revokeObjectURL(url), 1000);
}

/**
 * 加密备份与恢复。
 *
 * 备份里有群和聊天记录，用用户设的密码加密（PBKDF2 + AES-GCM）。
 * 不包含设备身份密钥（在系统 Keystore 里，不可导出）和管理员私钥
 * （走「管理员恢复码」这条路）。正在倒计时的阅后即焚消息也不备份——
 * 对方期望它到点消失。
 *
 * 导出：输两遍密码 → 生成加密文本 → 用 a[download] 触发浏览器下载。
 * 恢复：选文件 → 输密码 → 解密并预览（N 个群、M 条消息）→ 确认后合并。
 */
export function BackupScreen({ onBack, onExport, onPreviewRestore, onConfirmRestore }: BackupScreenProps) {
  const [mode, setMode] = useState<Mode>("export");

  // 导出状态
  const [exportPassword, setExportPassword] = useState("");
  const [exportConfirm, setExportConfirm] = useState("");
  const [exportError, setExportError] = useState<string | null>(null);
  const [exportBusy, setExportBusy] = useState(false);
  const [exportDone, setExportDone] = useState(false);

  // 恢复状态
  const [restoreText, setRestoreText] = useState<string | null>(null);
  const [restoreFileName, setRestoreFileName] = useState<string>("");
  const [restorePassword, setRestorePassword] = useState("");
  const [restoreError, setRestoreError] = useState<string | null>(null);
  const [restoreBusy, setRestoreBusy] = useState(false);
  const [preview, setPreview] = useState<RestorePreview | null>(null);
  const [restoreDone, setRestoreDone] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  async function handleExport() {
    setExportError(null);
    setExportDone(false);
    if (exportPassword.length < MIN_BACKUP_PASSWORD_LENGTH) {
      setExportError(`备份密码至少 ${MIN_BACKUP_PASSWORD_LENGTH} 个字符`);
      return;
    }
    if (exportPassword !== exportConfirm) {
      setExportError("两次输入的密码不一致");
      return;
    }
    setExportBusy(true);
    try {
      const text = await onExport(exportPassword);
      const filename = `secureinchat-backup-${formatDateForFilename(new Date())}.json`;
      downloadText(filename, text);
      setExportDone(true);
      setExportPassword("");
      setExportConfirm("");
    } catch (err) {
      setExportError(err instanceof Error ? err.message : "生成备份失败，请重试");
    } finally {
      setExportBusy(false);
    }
  }

  async function handleFileChosen(file: File) {
    setRestoreError(null);
    setPreview(null);
    setRestoreDone(false);
    try {
      const text = await file.text();
      setRestoreText(text);
      setRestoreFileName(file.name);
    } catch {
      setRestoreError("读取文件失败");
      setRestoreText(null);
      setRestoreFileName("");
    }
  }

  async function handlePreview() {
    if (!restoreText) return;
    setRestoreError(null);
    setPreview(null);
    setRestoreDone(false);
    setRestoreBusy(true);
    try {
      const result = await onPreviewRestore(restoreText, restorePassword);
      setPreview(result);
    } catch (err) {
      // 解密失败统一提示，不区分「密码错」和「文件被改」
      setRestoreError(err instanceof Error ? err.message : "密码不对，或文件已损坏");
    } finally {
      setRestoreBusy(false);
    }
  }

  async function handleConfirmRestore() {
    setRestoreError(null);
    setRestoreBusy(true);
    try {
      await onConfirmRestore();
      setRestoreDone(true);
      setPreview(null);
      setRestoreText(null);
      setRestorePassword("");
      setRestoreFileName("");
      if (fileInputRef.current) fileInputRef.current.value = "";
    } catch (err) {
      setRestoreError(err instanceof Error ? err.message : "恢复失败，请重试");
    } finally {
      setRestoreBusy(false);
    }
  }

  return (
    <div style={{ minHeight: "100%", background: colors.ivory, padding: 24, overflowY: "auto" }}>
      <button
        onClick={onBack}
        aria-label="返回"
        style={{
          background: "transparent",
          border: "none",
          boxShadow: "none",
          color: colors.deepInkGreen,
          fontSize: 14,
          marginBottom: 16,
          cursor: "pointer",
          minHeight: touchTarget.minDp,
        }}
      >
        ← 返回
      </button>

      <h1 style={{ fontSize: 17, fontWeight: 500, color: colors.textPrimary, margin: "0 0 8px" }}>备份与恢复</h1>
      <p style={{ fontSize: 12, color: "#8A8A82", lineHeight: 1.7, margin: "0 0 20px" }}>
        备份里有你的群和聊天记录，用你设的密码加密。忘记密码就无法恢复。
        设备密钥和管理员私钥不在备份里——管理员请另外保存好管理员恢复码。
        正在倒计时的阅后即焚消息不会被备份。
      </p>

      <div style={{ display: "flex", gap: 8, marginBottom: 20 }}>
        <Button
          variant={mode === "export" ? "primary" : "secondary"}
          onClick={() => setMode("export")}
          style={{ flex: 1 }}
        >
          导出备份
        </Button>
        <Button
          variant={mode === "restore" ? "primary" : "secondary"}
          onClick={() => setMode("restore")}
          style={{ flex: 1 }}
        >
          从备份恢复
        </Button>
      </div>

      {mode === "export" ? (
        <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
          <div style={{ fontSize: 14, color: colors.textPrimary }}>设置备份密码</div>
          <input
            type="password"
            value={exportPassword}
            onChange={(e) => {
              setExportPassword(e.target.value);
              setExportError(null);
              setExportDone(false);
            }}
            aria-label="备份密码输入框"
            placeholder={`至少 ${MIN_BACKUP_PASSWORD_LENGTH} 个字符`}
            style={inputStyle}
          />
          <input
            type="password"
            value={exportConfirm}
            onChange={(e) => {
              setExportConfirm(e.target.value);
              setExportError(null);
              setExportDone(false);
            }}
            aria-label="再次输入备份密码"
            placeholder="再输一次"
            style={inputStyle}
          />
          {exportError ? (
            <p role="alert" style={errorStyle}>
              {exportError}
            </p>
          ) : null}
          {exportDone ? (
            <p role="status" style={successStyle}>
              备份已下载。请把文件放到安全的地方——密码忘了就打不开了。
            </p>
          ) : null}
          <Button
            variant="primary"
            onClick={() => void handleExport()}
            disabled={exportBusy || exportPassword.length === 0 || exportConfirm.length === 0}
          >
            {exportBusy ? "生成中..." : "生成备份文件"}
          </Button>
        </div>
      ) : (
        <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
          <div style={{ fontSize: 14, color: colors.textPrimary }}>选择备份文件</div>
          <input
            ref={fileInputRef}
            type="file"
            accept=".json,application/json"
            onChange={(e) => {
              const file = e.target.files?.[0];
              if (file) void handleFileChosen(file);
            }}
            aria-label="选择备份文件"
            style={{
              minHeight: touchTarget.minDp,
              fontSize: 13,
              color: colors.textPrimary,
            }}
          />
          {restoreFileName ? (
            <p style={{ fontSize: 11, color: "#9A9A94", margin: 0 }}>已选择：{restoreFileName}</p>
          ) : null}

          <div style={{ fontSize: 14, color: colors.textPrimary, marginTop: 8 }}>输入备份密码</div>
          <input
            type="password"
            value={restorePassword}
            onChange={(e) => {
              setRestorePassword(e.target.value);
              setRestoreError(null);
              setPreview(null);
            }}
            aria-label="恢复密码输入框"
            placeholder="导出时设置的密码"
            style={inputStyle}
          />

          {restoreError ? (
            <p role="alert" style={errorStyle}>
              {restoreError}
            </p>
          ) : null}

          {preview ? (
            <div
              style={{
                border: `0.5px solid ${colors.sageMint}`,
                borderRadius: 12,
                padding: "12px 14px",
                background: `${colors.sageMint}22`,
              }}
            >
              <p style={{ fontSize: 13, color: colors.textPrimary, margin: 0 }}>
                将恢复 {preview.groups} 个群、{preview.messages} 条消息。
              </p>
              <p style={{ fontSize: 11, color: "#9A9A94", margin: "6px 0 0", lineHeight: 1.6 }}>
                合并到本机现有数据，不会覆盖已有的群和消息。恢复后会自动重连这些群。
              </p>
            </div>
          ) : null}

          {restoreDone ? (
            <p role="status" style={successStyle}>
              恢复完成。回到消息列表就能看到恢复的群。
            </p>
          ) : null}

          {!preview ? (
            <Button
              variant="primary"
              onClick={() => void handlePreview()}
              disabled={restoreBusy || !restoreText || restorePassword.length === 0}
            >
              {restoreBusy ? "解密中..." : "解密并预览"}
            </Button>
          ) : (
            <Button variant="primary" onClick={() => void handleConfirmRestore()} disabled={restoreBusy}>
              {restoreBusy ? "恢复中..." : "确认恢复"}
            </Button>
          )}
        </div>
      )}
    </div>
  );
}

关于 C8 的 PersistedMediaCache 怎么拿到恢复的媒体，明确说三层：

导出时：handleExportBackup 从 loadMessages(store, groupId) 读。C8 已经保证每次落盘都按 PersistedMediaCache 把已有媒体的 dataB64Url 补回去，所以读出来的 StoredMessage 自带媒体 base64，备份里自然包含媒体内容。

恢复落盘后：handleConfirmRestore 里每写完一个群，立刻 mediaCacheRef.current.seed(groupId, messages)。这是最关键的一步——PersistedMediaCache.buildStored 只认缓存里有的 id，不 seed 的话下一次任何保存（收到新消息、删除、到期、设置定时消失）都会把恢复的媒体 dataB64Url 写成 undefined，图片和语音就丢了。

恢复后自动重连：connectToGroup 内部会 loadMessages 并 mediaCacheRef.current.seed(groupId, keptStored)。这覆盖了「重启应用后第一次连上恢复的群」这条路径，和 C8 原本的逻辑一致，不需要改 connectToGroup。

真机验证点：

换手机后：新设备装应用 → 「我的」→「备份与恢复」→ 选旧手机导出的 .json → 输密码 → 预览显示「将恢复 N 个群、M 条消息」→ 确认后回到消息列表能看到所有群，点进去历史消息都在。

历史里的图片和语音能正常显示/播放，不是只有文件名；再发一条新消息后再重启应用，旧图片和语音仍在（验证 C8 的 seed 生效）。

恢复后自动重连恢复的群，连接状态显示正常；某个群连不上时不影响其他群。

恢复时把昵称也一并恢复（本机原本没设昵称的情况）；本机已有昵称时不覆盖。

备份文件里确实不含已过期和正在倒计时的阅后即焚消息：设置 5 分钟发一条，立即导出备份，恢复后这条消息不在。

输错密码时提示「密码不对，或文件已损坏」，不区分具体原因；手动改文件里的一个字符也一样。

恢复后原来的定时消失设置跟着回来（群设置里显示的还是备份时的选项）。

    const payload = await decryptBackup(text, password);
    pendingRestoreRef.current = payload;
    const messageCount = Object.values(payload.messagesByGroup).reduce(
      (sum, arr) => sum + arr.length,
      0
    );
    return { groups: payload.groups.length, messages: messageCount };
  }

  /**
   * 确认恢复：合并到本机现有数据，落盘，然后自动重连恢复的群。
   *
   * 关键一步：恢复的媒体要让 C8 的 PersistedMediaCache 知道。
   * 做法是 saveMessages 之后调用 mediaCacheRef.current.seed(groupId, messages)。
   * 不 seed 的话，下一次任何保存都会按「缓存里没有这条媒体」把 dataB64Url 写成
   * undefined，刚恢复的图片和语音就丢了。恢复的群自动重连时，connectToGroup
   * 里 loadMessages 也会再 seed 一次——两层保障。
   */
  async function handleConfirmRestore(): Promise<void> {
    const payload = pendingRestoreRef.current;
    if (!payload) throw new Error("没有待恢复的备份，请重新选择文件并解密");
    pendingRestoreRef.current = null;

    const store = await getDeviceStore();
    const existingGroups = await loadJoinedGroups(store);
    const existingMessagesByGroup: Record<string, StoredMessage[]> = {};
    for (const g of existingGroups) {
      existingMessagesByGroup[g.groupId] = await loadMessages(store, g.groupId);
    }
    const merged = mergeRestore(existingGroups, existingMessagesByGroup, payload);

    await saveJoinedGroups(store, merged.groups);
    for (const [groupId, messages] of Object.entries(merged.messagesByGroup)) {
      await saveMessages(store, groupId, messages);
      // 让 C8 的缓存知道恢复的媒体，后续保存才不会把它写丢
      mediaCacheRef.current.seed(groupId, messages);
    }

    // 昵称：只在备份里有、且本机还没设过时补上——不要覆盖用户当前的选择
    if (payload.nickname && !nicknameRef.current) {
      await saveNickname(payload.nickname);
      nicknameRef.current = payload.nickname;
      setNickname(payload.nickname);
    }

    // 各群的定时消失设置跟着 JoinedGroup 一起恢复
    setDisappearAfterByGroup(
      Object.fromEntries(merged.groups.map((g) => [g.groupId, g.disappearAfterMs ?? 0]))
    );

    // 自动重连恢复的群。恢复期间把 isAutoReconnectRef 设成 true，
    // 避免第一个群触发 shouldOpenChat 把用户从备份页拽到聊天页。
    const wasAutoReconnect = isAutoReconnectRef.current;
    isAutoReconnectRef.current = true;
    try {
      for (const g of merged.groups) {
        if (sessionsRef.current[g.groupId]) continue;
        try {
          await connectToGroup(
            {
              version: "SIC2",
              isCompatMode: false,
              serverJoinCode: "",
              groupId: g.groupId,
              keyMaterialB64Url: g.keyMaterialB64Url,
              epoch: g.epoch,
              ...(g.adminPublicKeyRawB64Url
                ? { adminPublicKeyRawB64Url: g.adminPublicKeyRawB64Url }
                : {}),
            },
            g.groupName
          );
        } catch {
          // 单个群连不上不影响其他群；用户可在消息列表里重试
        }
      }
    } finally {
      isAutoReconnectRef.current = wasAutoReconnect;
    }

import { describe, expect, it, vi } from "vitest";
import { render, screen, fireEvent, waitFor } from "@testing-library/react";

function renderScreen(overrides: {
  onExport?: (password: string) => Promise<string>;
  onPreviewRestore?: (text: string, password: string) => Promise<{ groups: number; messages: number }>;
  onConfirmRestore?: () => Promise<void>;
} = {}) {
  const onExport = overrides.onExport ?? vi.fn().mockResolvedValue("{\"format\":\"sic-backup\"}");

  const onConfirmRestore = overrides.onConfirmRestore ?? vi.fn().mockResolvedValue(undefined);

    <BackupScreen
      onBack={vi.fn()}
      onExport={onExport}
      onPreviewRestore={onPreviewRestore}
