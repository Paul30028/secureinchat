/**
 * 阅后即焚（消息定时消失）的公共逻辑。
 *
 * 计时从「本机收到/发出这条消息的时刻」开始，不用 sentAtMs——对方时钟不准
 * 会导致一收到就删，离线补发的消息也会一到就过期。
 *
 * 这个模块只做纯计算，不碰 UI、不碰存储，方便测试和复用。
 */

/** 可选的消失时长：关、5 分钟、1 小时、1 天、7 天。 */
export const DISAPPEAR_OPTIONS_MS = [
  0,
  5 * 60_000,
  3_600_000,
  86_400_000,
  7 * 86_400_000,
] as const;

/** 消息里允许携带的最短消失时长：1 分钟。 */
const MIN_EXPIRES_MS = 60_000;
/** 消息里允许携带的最长消失时长：7 天。 */
const MAX_EXPIRES_MS = 7 * 86_400_000;

/**
 * 把信封里带过来的 expiresInMs 校验成可用的值。
 * 必须是有限整数（Number.isInteger），且在 [60_000, 7 天] 范围内；
 * 否则当作没有（不删、不报错）。小数一律拒绝，不做四舍五入。
 * 老客户端发的消息没有这个字段，返回 undefined，照常显示。
 */
export function sanitizeExpiresInMs(v: unknown): number | undefined {
  if (typeof v !== "number" || !Number.isInteger(v)) return undefined;
  if (v < MIN_EXPIRES_MS || v > MAX_EXPIRES_MS) return undefined;
  return v;
}

/**
 * 本机收到/发出这条消息的时刻 + 时长 = 到期时间。没有时长就不设到期。
 */
export function expiresAtFor(
  receivedAtMs: number,
  expiresInMs: number | undefined
): number | undefined {
  if (expiresInMs === undefined) return undefined;
  return receivedAtMs + expiresInMs;
}

/**
 * 按当前时刻把消息分成「留着」和「已过期」两堆。
 * 到期时间缺失或不是有限数字的当作不过期（老消息、损坏数据）。
 */
export function partitionExpired<T extends { expiresAtMs?: number | undefined }>(
  items: T[],
  nowMs: number
): { keep: T[]; expired: T[] } {
  const keep: T[] = [];
  const expired: T[] = [];
  for (const item of items) {
    const at = item.expiresAtMs;
    if (typeof at === "number" && Number.isFinite(at) && at <= nowMs) {
      expired.push(item);
    } else {
      keep.push(item);
    }
  }
  return { keep, expired };
}

/**
 * 最早一条的到期时间，给定时器用。没有需要定时的消息时返回 undefined。
 * 调用方拿它减当前时间就是 setTimeout 的延迟；最少要等 1 秒，
 * 避免到期时间在过去（时钟回拨等）时立刻疯狂重排。
 */
export function nextExpiryMs<T extends { expiresAtMs?: number | undefined }>(
  items: T[]
): number | undefined {
  let earliest: number | undefined;
  for (const item of items) {
    const at = item.expiresAtMs;
    if (typeof at !== "number" || !Number.isFinite(at)) continue;
    if (earliest === undefined || at < earliest) earliest = at;
  }
  return earliest;
}

describe("sanitizeExpiresInMs", () => {
  it("accepts the boundary values", () => {
    expect(sanitizeExpiresInMs(60_000)).toBe(60_000);
    expect(sanitizeExpiresInMs(7 * 86_400_000)).toBe(7 * 86_400_000);
  });

  it("rejects below the minimum", () => {
    expect(sanitizeExpiresInMs(59_999)).toBeUndefined();
    expect(sanitizeExpiresInMs(0)).toBeUndefined();
    expect(sanitizeExpiresInMs(-1)).toBeUndefined();
  });

  it("rejects above the maximum", () => {
    expect(sanitizeExpiresInMs(7 * 86_400_000 + 1)).toBeUndefined();
  });

  it("rejects non-finite and non-number", () => {
    expect(sanitizeExpiresInMs(Number.NaN)).toBeUndefined();
    expect(sanitizeExpiresInMs(Number.POSITIVE_INFINITY)).toBeUndefined();
    expect(sanitizeExpiresInMs("300000")).toBeUndefined();
    expect(sanitizeExpiresInMs(null)).toBeUndefined();
    expect(sanitizeExpiresInMs(undefined)).toBeUndefined();
  });

  it("rejects non-integer numbers instead of rounding", () => {
    expect(sanitizeExpiresInMs(60_000.4)).toBeUndefined();
    expect(sanitizeExpiresInMs(300_000.6)).toBeUndefined();
  });
});

    expect(expiresAtFor(1000, undefined)).toBeUndefined();
  });

  it("adds the duration to the receive time", () => {

describe("partitionExpired", () => {

import {
  bytesToBase64Url,
  base64UrlToBytes,
  shouldPersistMediaBytes,
  type StoredMessage,
} from "@secureinchat/chat-core";
import type { DisplayMessage } from "./screens/ChatScreen";

/**
 * 内存里的消息（媒体用 objectUrl 渲染）和落盘的消息（媒体存 base64）之间的转换。
 *
 * objectUrl 是当次页面会话内的临时引用，重启后必然失效，所以不能直接存；
 * 反过来读出来的时候要重新 createObjectURL 才能渲染。
 */

export function toStored(msg: DisplayMessage, mediaBytes?: Uint8Array): StoredMessage {
  const base: StoredMessage = {
    id: msg.id,
    isOwn: msg.isOwn,
    senderLabel: msg.fromDeviceId,
    // 用消息自己的时间，不是落盘那一刻——否则重启后所有历史消息
    // 的时间都会变成最后一次保存的时间
    sentAtMs: msg.sentAtMs,
    text: msg.text,
    // 阅后即焚的到期时刻也要落盘：重启后重新加载时按当前时间过滤掉
    // 已经过期的消息，不能让它再冒出来。
    ...(msg.expiresAtMs !== undefined ? { expiresAtMs: msg.expiresAtMs } : {}),
  };
  if (!msg.media) return base;

  const persistBytes = mediaBytes && shouldPersistMediaBytes(msg.media.sizeBytes);
  return {
    ...base,
    media: {
      mediaKind: msg.media.mediaKind,
      fileName: msg.media.fileName,
      mimeType: msg.media.mimeType,
      sizeBytes: msg.media.sizeBytes,
      // 超过上限的媒体只留元信息：重启后会显示文件名和大小，但内容打不开。
      // 这比整条消息消失要好，用户至少知道当时收到过什么。
      dataB64Url: persistBytes ? bytesToBase64Url(mediaBytes) : undefined,
      ...(msg.media.durationMs !== undefined ? { durationMs: msg.media.durationMs } : {}),
    },
  };
}

export function fromStored(stored: StoredMessage): DisplayMessage {
  const base: DisplayMessage = {
    id: stored.id,
    isOwn: stored.isOwn,
    fromDeviceId: stored.senderLabel,
    sentAtMs: stored.sentAtMs,
    text: stored.text,
    ...(stored.expiresAtMs !== undefined ? { expiresAtMs: stored.expiresAtMs } : {}),
  };
  if (!stored.media) return base;

  let objectUrl = "";
  if (stored.media.dataB64Url) {
    const bytes = base64UrlToBytes(stored.media.dataB64Url);
    objectUrl = URL.createObjectURL(new Blob([bytes as BlobPart], { type: stored.media.mimeType }));
  }

  return {
    ...base,
    media: {
      mediaKind: stored.media.mediaKind,
      fileName: stored.media.fileName,
      mimeType: stored.media.mimeType,
      objectUrl,
      sizeBytes: stored.media.sizeBytes,
      ...(stored.media.durationMs !== undefined ? { durationMs: stored.media.durationMs } : {}),
    },
  };
}

/** 把时长转成界面上显示的中文标签。 */
export function disappearLabel(ms: number): string {
  if (ms === 300_000) return "5 分钟";

  if (ms === 604_800_000) return "7 天";

  // 每个群一个到期定时器。用 nextExpiryMs 算出下一条何时到期，
  // 到点再重排——不轮询。
  const expiryTimersRef = useRef<Map<string, ReturnType<typeof setTimeout>>>(new Map());

  /** 清掉某个群的到期定时器（离开群、卸载时用）。 */
  function clearExpiryTimer(groupId: string) {
    const timer = expiryTimersRef.current.get(groupId);
    if (timer !== undefined) {
      clearTimeout(timer);
      expiryTimersRef.current.delete(groupId);
    }
  }

  /** 按这个群当前最早的一条到期时间重排定时器。 */
  function scheduleExpiryFor(groupId: string) {
    clearExpiryTimer(groupId);
    const session = sessionsRef.current[groupId];
    if (!session) return;
    const next = nextExpiryMs(session.messages);
    if (next === undefined) return;
    // 最少 1 秒：到期时间已经在过去（时钟回拨等）时不要立刻疯狂重排
    const delay = Math.max(1000, next - Date.now());
    const timer = setTimeout(() => {
      void expireMessagesFor(groupId);
    }, delay);
    expiryTimersRef.current.set(groupId, timer);
  }

  /**
   * 把已经过期的消息从这个群里删掉、释放 objectUrl、落盘，然后为下一条重排。
   * 界面和加密存储里都要删——这是「阅后即焚」的实际执行点。
   */
  async function expireMessagesFor(groupId: string) {
    clearExpiryTimer(groupId);
    const session = sessionsRef.current[groupId];
    if (!session) return;
    const { keep, expired } = partitionExpired(session.messages, Date.now());
    if (expired.length === 0) {
      scheduleExpiryFor(groupId);
      return;
    }
    for (const m of expired) {
      if (m.media?.objectUrl) URL.revokeObjectURL(m.media.objectUrl);
    }
    setSessions((prev) => {
      const next = patchSession(prev, groupId, { messages: keep });
      sessionsRef.current = next;
      return next;
    });
    try {
      const store = await getDeviceStore();
      await saveMessages(store, groupId, keep.map((m) => toStored(m)));
    } catch {
      // 落盘失败不影响界面上的删除；下次写入会覆盖
    }
    scheduleExpiryFor(groupId);
  }

  // 组件卸载时把所有定时器清掉，避免在已卸载的组件上 setState
  useEffect(() => {
    return () => {
      for (const timer of expiryTimersRef.current.values()) clearTimeout(timer);
      expiryTimersRef.current.clear();
    };
  }, []);

  /**
   * 修改某个群的「消息定时消失」。
   * 保存到 JoinedGroup（本机、本群），并插一条本地系统提示——不发送给别人。
   */
  async function handleSetDisappearAfter(groupId: string, ms: number) {
    const store = await getDeviceStore();
    const groups = setDisappearAfter(await loadJoinedGroups(store), groupId, ms);
    await saveJoinedGroups(store, groups);
    setDisappearAfterByGroup((prev) => ({ ...prev, [groupId]: ms }));
    const session = sessionsRef.current[groupId];
    if (!session) return;
    await appendOwnMessage(groupId, {
      id: `sys-${nextMessageId++}`,
      isOwn: true,
      system: true,
      sentAtMs: Date.now(),
      text: ms > 0 ? `你开启了定时消失：${disappearLabel(ms)}` : "你关闭了定时消失",
    });
  }

 * 这个模块只做纯计算，不碰 UI、不碰存储，方便测试和复用。
 */

/** 可选的消失时长：关、5 分钟、1 小时、1 天、7 天。 */
export const DISAPPEAR_OPTIONS_MS = [
  0,
  5 * 60_000,
  3_600_000,
  86_400_000,
  7 * 86_400_000,
] as const;

/** 消息里允许携带的最短消失时长：1 分钟。 */
const MIN_EXPIRES_MS = 60_000;
/** 消息里允许携带的最长消失时长：7 天。 */
const MAX_EXPIRES_MS = 7 * 86_400_000;

/**
 * 把信封里带过来的 expiresInMs 校验成可用的值。
 * 必须是有限整数（Number.isInteger），且在 [60_000, 7 天] 范围内；
 * 否则当作没有（不删、不报错）。小数一律拒绝，不做四舍五入。
 * 老客户端发的消息没有这个字段，返回 undefined，照常显示。
 */
export function sanitizeExpiresInMs(v: unknown): number | undefined {
  if (typeof v !== "number" || !Number.isInteger(v)) return undefined;
  if (v < MIN_EXPIRES_MS || v > MAX_EXPIRES_MS) return undefined;
  return v;
}

/**
 * 本机收到/发出这条消息的时刻 + 时长 = 到期时间。没有时长就不设到期。
 */
export function expiresAtFor(
  receivedAtMs: number,
  expiresInMs: number | undefined
): number | undefined {
  if (expiresInMs === undefined) return undefined;
  return receivedAtMs + expiresInMs;
}

/**
 * 按当前时刻把消息分成「留着」和「已过期」两堆。
 * 到期时间缺失或不是有限数字的当作不过期（老消息、损坏数据）。
 */
export function partitionExpired<T extends { expiresAtMs?: number | undefined }>(
  items: T[],
  nowMs: number
): { keep: T[]; expired: T[] } {
  const keep: T[] = [];
  const expired: T[] = [];
  for (const item of items) {
    const at = item.expiresAtMs;
    if (typeof at === "number" && Number.isFinite(at) && at <= nowMs) {
      expired.push(item);
    } else {
      keep.push(item);
    }
  }
  return { keep, expired };
}

/**
 * 最早一条的到期时间，给定时器用。没有需要定时的消息时返回 undefined。
 * 调用方拿它减当前时间就是 setTimeout 的延迟；最少要等 1 秒，
 * 避免到期时间在过去（时钟回拨等）时立刻疯狂重排。
 */
export function nextExpiryMs<T extends { expiresAtMs?: number | undefined }>(
  items: T[]
): number | undefined {
  let earliest: number | undefined;
  for (const item of items) {
    const at = item.expiresAtMs;
    if (typeof at !== "number" || !Number.isFinite(at)) continue;
    if (earliest === undefined || at < earliest) earliest = at;
  }
  return earliest;
}

 *
 * objectUrl 是当次页面会话内的临时引用，重启后必然失效，所以不能直接存；
 * 反过来读出来的时候要重新 createObjectURL 才能渲染。
 */

export function toStored(msg: DisplayMessage, mediaBytes?: Uint8Array): StoredMessage {
  const base: StoredMessage = {
    id: msg.id,
    isOwn: msg.isOwn,
    senderLabel: msg.fromDeviceId,
    // 用消息自己的时间，不是落盘那一刻——否则重启后所有历史消息
    // 的时间都会变成最后一次保存的时间
    sentAtMs: msg.sentAtMs,
    text: msg.text,
    // 阅后即焚的到期时刻也要落盘：重启后重新加载时按当前时间过滤掉
    // 已经过期的消息，不能让它再冒出来。
    ...(msg.expiresAtMs !== undefined ? { expiresAtMs: msg.expiresAtMs } : {}),
  };
  if (!msg.media) return base;

  const persistBytes = mediaBytes && shouldPersistMediaBytes(msg.media.sizeBytes);
  return {
    ...base,
    media: {
      mediaKind: msg.media.mediaKind,
      fileName: msg.media.fileName,
      mimeType: msg.media.mimeType,
      sizeBytes: msg.media.sizeBytes,
      // 超过上限的媒体只留元信息：重启后会显示文件名和大小，但内容打不开。
      // 这比整条消息消失要好，用户至少知道当时收到过什么。
      dataB64Url: persistBytes ? bytesToBase64Url(mediaBytes) : undefined,
      ...(msg.media.durationMs !== undefined ? { durationMs: msg.media.durationMs } : {}),
    },
  };
}

export function fromStored(stored: StoredMessage): DisplayMessage {
  const base: DisplayMessage = {
    id: stored.id,
    isOwn: stored.isOwn,
    fromDeviceId: stored.senderLabel,
    sentAtMs: stored.sentAtMs,
    text: stored.text,
    ...(stored.expiresAtMs !== undefined ? { expiresAtMs: stored.expiresAtMs } : {}),
  };
  if (!stored.media) return base;

  let objectUrl = "";
  if (stored.media.dataB64Url) {
    const bytes = base64UrlToBytes(stored.media.dataB64Url);
    objectUrl = URL.createObjectURL(new Blob([bytes as BlobPart], { type: stored.media.mimeType }));
  }

  return {
    ...base,
    media: {
      mediaKind: stored.media.mediaKind,
      fileName: stored.media.fileName,
      mimeType: stored.media.mimeType,
      objectUrl,
      sizeBytes: stored.media.sizeBytes,
      ...(stored.media.durationMs !== undefined ? { durationMs: stored.media.durationMs } : {}),
    },
  };
}

  function clearExpiryTimer(groupId: string) {
    const timer = expiryTimersRef.current.get(groupId);
    if (timer !== undefined) {
      clearTimeout(timer);
      expiryTimersRef.current.delete(groupId);
    }

  function scheduleExpiryFor(groupId: string) {
    clearExpiryTimer(groupId);
    const session = sessionsRef.current[groupId];
    if (!session) return;
    const next = nextExpiryMs(session.messages);
    if (next === undefined) return;
    // 最少 1 秒：到期时间已经在过去（时钟回拨等）时不要立刻疯狂重排
    const delay = Math.max(1000, next - Date.now());

        { id: "plain" },

      ];
      // 刚发出：都还在

      expect(split.keep.map((m) => m.id)).toEqual(["plain", "burn"]);
      expect(split.expired).toEqual([]);

      // 推进 5 分钟
      vi.advanceTimersByTime(5 * 60_000);
      split = partitionExpired(messages, Date.now());
      expect(split.keep.map((m) => m.id)).toEqual(["plain"]);
      expect(split.expired.map((m) => m.id)).toEqual(["burn"]);
    } finally {
      vi.useRealTimers();
    }
