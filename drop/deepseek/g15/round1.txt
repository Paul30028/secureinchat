package org.secureinchat.guardian.dns

/**
 * One row of the Exodus tracker database (see app/src/main/assets/trackers-exodus.tsv).
 *
 * [networkRegex] is null when the source row has no network signature, or when its signature is not
 * a valid Java regex: in both cases the tracker has no network face and byDomain will never return
 * it. [codeSignatures] are class-name prefixes; the G17 scanner matches an installed app's DEX
 * class names against them, so they are kept verbatim (with or without a trailing dot).
 */
data class Tracker(
    val id: Int,
    val name: String,
    val categories: List<String>,
    val website: String,
    val networkRegex: Regex?,
    val codeSignatures: List<String>
)

/**
 * Lookup over the Exodus tracker set.
 *
 * Domain matching is "the domain contains a match for the tracker's network regex", which is what
 * Exodus/TrackerControl mean: a row for `doubleclick\.net` matches both `doubleclick.net` and
 * `stats.g.doubleclick.net` without the DB having to list every subdomain. When several rows match
 * one domain, the smallest id wins so the answer is stable across runs.
 *
 * The source data is licensed ODbL v1.0 / DbCL v1.0; see the attribution in MainActivity and the
 * header of the asset file.
 */
class TrackerDb private constructor(
    private val trackers: List<Tracker>
) {
    /** Trackers with a working network regex, sorted by id so the first hit is the smallest id. */
    private val networkTrackers: List<Tracker> =
        trackers.filter { it.networkRegex != null }.sortedBy { it.id }

    val size: Int get() = trackers.size
    val withNetwork: Int get() = networkTrackers.size

    /**
     * LRU of byDomain results, misses included. A domain is looked up for every visible log line
     * and every "Blocked most often" row on each UI refresh (1 Hz), so the cache turns repeated
     * lookups of the same handful of names into O(1). The regex scan itself is O(networkTrackers)
     * per miss; with 260 trackers and short hostnames that is tens of microseconds, fine for a
     * cache miss but not for a per-frame loop.
     */
    private val cache = object : LinkedHashMap<String, Tracker?>(CACHE_INITIAL, 0.75f, true) {
        override fun removeEldestEntry(eldest: MutableMap.MutableEntry<String, Tracker?>?): Boolean =
            size > CACHE_SIZE
    }

    /** First tracker whose network regex matches [name] (containsMatchIn, not full match), or null. */
    fun byDomain(name: String): Tracker? {
        val key = name.lowercase()
        synchronized(cache) {
            if (cache.containsKey(key)) return cache[key]
        }
        var found: Tracker? = null
        for (t in networkTrackers) {
            val r = t.networkRegex ?: continue
            if (r.containsMatchIn(key)) { found = t; break }
        }
        synchronized(cache) { cache[key] = found }
        return found
    }

    /**
     * Tracker whose DEX class-name signature is a prefix of [className], or null.
     *
     * Both dotted signatures ("com.krux.androidsdk") and non-dotted ones ("com.databerries") are
     * treated as plain prefixes, per the G17 spec. Not cached: the G17 scanner walks many class
     * names once per scan, so per-call cost is negligible and a per-process cache would just hold
     * garbage.
     */
    fun byClassName(className: String): Tracker? {
        for (t in trackers) {
            for (sig in t.codeSignatures) {
                if (sig.isEmpty()) continue
                if (className.startsWith(sig)) return t
            }
        }
        return null
    }

    companion object {
        private const val CACHE_SIZE = 512
        private const val CACHE_INITIAL = 128
        private const val EXPECTED_COLS = 6

        /**
         * Parses a TSV as produced by tools/update_trackers.py. Comment lines start with '#';
         * blank lines are ignored. A row with fewer than 6 tab-separated columns is dropped; a
         * row whose network regex does not compile is kept but gets a null regex (the tracker has
         * code signatures only). Rows are returned sorted by id so iteration is deterministic.
         */
        fun parse(tsv: String): TrackerDb {
            val list = ArrayList<Tracker>(512)
            for (raw in tsv.lineSequence()) {
                if (raw.isEmpty()) continue
                if (raw.startsWith("#")) continue
                val parts = raw.split('\t')
                if (parts.size < EXPECTED_COLS) continue
                val id = parts[0].trim().toIntOrNull() ?: continue
                val name = parts[1]
                if (name.isEmpty()) continue
                val catRaw = parts[2].trim()
                val cats = if (catRaw.isEmpty()) emptyList()
                else catRaw.split(',').map { it.trim() }.filter { it.isNotEmpty() }
                val website = parts[3]
                val regexRaw = parts[4]
                val rx = if (regexRaw.isEmpty()) null
                else try { Regex(regexRaw) } catch (_: Throwable) { null }
                val sigsRaw = parts[5]
                val sigs = if (sigsRaw.isEmpty()) emptyList()
                else sigsRaw.split('|').filter { it.isNotEmpty() }
                list.add(Tracker(id, name, cats, website, rx, sigs))
            }
            return TrackerDb(list.sortedBy { it.id })
        }
    }
}

package org.secureinchat.guardian.dns

import java.nio.file.Files
import java.nio.file.Paths

private var tn = 0
private fun t(name: String, block: () -> Unit) {
    try { block(); tn++ } catch (e: Throwable) {
        println("FAIL $name: ${e.message}")
        throw e
    }
}
private fun eq(a: Any?, b: Any?) { check(a == b) { "expected $a, got $b" } }

fun main() {

    t("parse normal rows") {
        val tsv = """
            # comment
            1	Teemo	Analytics	https://www.teemo.co	databerries\.com	com.databerries.|com.geolocstation.
            2	FidZup	Analytics	https://www.fidzup.com	fidzup	com.fidzup.
            3	Audience Studio (Krux)	Analytics,Profiling	https://www.salesforce.com	krxd\.net	com.krux.androidsdk
        """.trimIndent()
        val db = TrackerDb.parse(tsv)
        eq(db.size, 3)
        eq(db.withNetwork, 3)
        val teemo = db.byDomain("x.databerries.com")
        eq(teemo?.name, "Teemo")
        eq(teemo?.id, 1)
        eq(teemo?.categories, listOf("Analytics"))
    }

    t("empty categories kept as empty list") {
        val tsv = "1	Empty		http://x.com	x\.com	"
        val db = TrackerDb.parse(tsv)
        eq(db.size, 1)
        val tr = db.byDomain("x.com")
        eq(tr?.name, "Empty")
        eq(tr?.categories, emptyList<String>())
    }

    t("bad rows skipped, bad regex dropped to null") {
        val tsv = """
            1	Good	Analytics	http://good.com	good\.com	com.good.
            notanumber	Bad	X	http://bad.com	bad\.com	com.bad.
            2	OnlyThree	X
            3	NoRegexCompiles	Analytics	http://nre.com	*invalid(	com.nre.
            4	NoSigs	Analytics	http://ns.com	ns\.com	
        """.trimIndent()
        val db = TrackerDb.parse(tsv)
        // Rows kept: Good, NoRegexCompiles (tracker row valid, regex null), NoSigs (empty sigs).
        eq(db.size, 3)
        // Only Good and NoSigs have a usable network regex.
        eq(db.withNetwork, 2)
        eq(db.byDomain("good.com")?.name, "Good")
        eq(db.byDomain("nre.com"), null)  // bad regex means no network match
    }

    t("byDomain uses containsMatchIn, not full match") {
        val tsv = "1	InMobi	Advertisement	http://inmobi.com	inmobi\.com	com.inmobi."
        val db = TrackerDb.parse(tsv)
        eq(db.byDomain("w.inmobi.com")?.name, "InMobi")
        eq(db.byDomain("inmobi.com")?.name, "InMobi")
        eq(db.byDomain("example.org"), null)
    }

    t("multiple matches pick the smallest id") {
        val tsv = """
            10	LowPrio	Advertisement	http://x.com	doubleclick\.net	com.x.
            3	HighPrio	Advertisement	http://y.com	doubleclick\.net	com.y.
            7	Mid	Advertisement	http://z.com	doubleclick\.net	com.z.
        """.trimIndent()
        val db = TrackerDb.parse(tsv)
        eq(db.byDomain("stats.g.doubleclick.net")?.id, 3)
        // Same lookup, cached; must be identical.
        eq(db.byDomain("stats.g.doubleclick.net")?.id, 3)
    }

    t("misses are cached and stay null") {
        val tsv = "1	A	Analytics	http://a.com	a\.com	"
        val db = TrackerDb.parse(tsv)
        eq(db.byDomain("nomatch.org"), null)
        eq(db.byDomain("nomatch.org"), null)
        eq(db.byDomain("x.a.com")?.id, 1)
        eq(db.byDomain("x.a.com")?.id, 1)
    }

    t("byClassName prefix, with and without trailing dot") {
        val tsv = """
            1	Teemo	Analytics	http://teemo.co	databerries\.com	com.databerries.
            2	Krux	Analytics	http://krux.com	krxd\.net	com.krux.androidsdk
        """.trimIndent()
        val db = TrackerDb.parse(tsv)
        // Trailing-dot signature: matches only when the dot is present in the prefix.
        eq(db.byClassName("com.databerries.Foo")?.id, 1)
        eq(db.byClassName("com.databerries")?.id, null)
        eq(db.byClassName("com.databerriesx")?.id, null)
        // Non-dotted signature: matches any class name with that prefix, including "com.krux.androidsdkx".
        eq(db.byClassName("com.krux.androidsdk.Foo")?.id, 2)
        eq(db.byClassName("com.krux.androidsdkx")?.id, 2)
        eq(db.byClassName("org.other")?.id, null)
    }

    t("empty tsv gives empty db") {
        val db = TrackerDb.parse("")
        eq(db.size, 0)
        eq(db.withNetwork, 0)
        eq(db.byDomain("example.com"), null)
        eq(db.byClassName("com.example.Foo"), null)
    }

    t("real exodus tsv loads") {
        // CWD is guardian/ when CI runs; this path is relative to that.
        val path = Paths.get("app/src/main/assets/trackers-exodus.tsv")
        check(Files.exists(path)) {
            "trackers-exodus.tsv not found at $path (CWD=${Paths.get("").toAbsolutePath()})"
        }
        val text = Files.readString(path)
        val db = TrackerDb.parse(text)
        check(db.size >= 400) { "expected >= 400 trackers, got ${db.size}" }
        check(db.withNetwork >= 250) { "expected >= 250 with network, got ${db.withNetwork}" }
        // At least one of these two must map to some tracker. The spec explicitly says: if neither
        // does, report it, do not weaken the assertion. Per the TSV excerpt line 5 ("Google
        // DoubleClick" with doubleclick\.net in its regex) this should pass on doubleclick.net.
        val dc = db.byDomain("doubleclick.net")
        val bb = db.byDomain("bat.bing.com")
        check(dc != null || bb != null) {
            "neither doubleclick.net nor bat.bing.com maps to a tracker (dc=$dc, bb=$bb)"
        }
        println(
            "real tsv: size=${db.size} withNetwork=${db.withNetwork} " +
            "doubleclick.net->${dc?.name} bat.bing.com->${bb?.name}"
        )
    }

    println("TrackerDbTest: $tn passed")
}

package org.secureinchat.guardian.dns

/**
 * One row of the Exodus tracker database (see app/src/main/assets/trackers-exodus.tsv).
 *
 * [networkRegex] is null when the source row has no network signature, or when its signature is not
 * a valid Java regex: in both cases the tracker has no network face and byDomain will never return
 * it. [codeSignatures] are class-name prefixes; the G17 scanner matches an installed app's DEX
 * class names against them, so they are kept verbatim (with or without a trailing dot).
 */
data class Tracker(
    val id: Int,
    val name: String,
    val categories: List<String>,
    val website: String,
    val networkRegex: Regex?,
    val codeSignatures: List<String>
)

 * Lookup over the Exodus tracker set.
 *
 * Domain matching is "the domain contains a match for the tracker's network regex", which is what
 * Exodus/TrackerControl mean: a row for `doubleclick\.net` matches both `doubleclick.net` and
 * `stats.g.doubleclick.net` without the DB having to list every subdomain. When several rows match
 * one domain, the smallest id wins so the answer is stable across runs.
 *
 * The source data is licensed ODbL v1.0 / DbCL v1.0; see the attribution in MainActivity and the
 * header of the asset file.
 */
class TrackerDb private constructor(
    private val trackers: List<Tracker>
) {
    /** Trackers with a working network regex, sorted by id so the first hit is the smallest id. */
    private val networkTrackers: List<Tracker> =
        trackers.filter { it.networkRegex != null }.sortedBy { it.id }

    val size: Int get() = trackers.size
    val withNetwork: Int get() = networkTrackers.size

    /**
     * LRU of byDomain results, misses included. A domain is looked up for every visible log line
     * and every "Blocked most often" row on each UI refresh (1 Hz), so the cache turns repeated
     * lookups of the same handful of names into O(1). The regex scan itself is O(networkTrackers)
     * per miss; with 260 trackers and short hostnames that is tens of microseconds, fine for a
     * cache miss but not for a per-frame loop.
     */
    private val cache = object : LinkedHashMap<String, Tracker?>(CACHE_INITIAL, 0.75f, true) {
        override fun removeEldestEntry(eldest: MutableMap.MutableEntry<String, Tracker?>?): Boolean =
            size > CACHE_SIZE
    }

    /** First tracker whose network regex matches [name] (containsMatchIn, not full match), or null. */
    fun byDomain(name: String): Tracker? {
        val key = name.lowercase()
        synchronized(cache) {
            if (cache.containsKey(key)) return cache[key]
        }
        var found: Tracker? = null
        for (t in networkTrackers) {
            val r = t.networkRegex ?: continue
            if (r.containsMatchIn(key)) { found = t; break }
        }
        synchronized(cache) { cache[key] = found }
        return found
    }

    /**
     * Tracker whose DEX class-name signature is a prefix of [className], or null.
     *
     * Both dotted signatures ("com.krux.androidsdk") and non-dotted ones ("com.databerries") are
     * treated as plain prefixes, per the G17 spec. Not cached: the G17 scanner walks many class
     * names once per scan, so per-call cost is negligible and a per-process cache would just hold
     * garbage.
     */
    fun byClassName(className: String): Tracker? {
        for (t in trackers) {
            for (sig in t.codeSignatures) {
                if (sig.isEmpty()) continue
                if (className.startsWith(sig)) return t
            }
        }
        return null
    }

    companion object {
        private const val CACHE_SIZE = 512
        private const val CACHE_INITIAL = 128
        private const val EXPECTED_COLS = 6

        /**
         * Parses a TSV as produced by tools/update_trackers.py. Comment lines start with '#';
         * blank lines are ignored. A row with fewer than 6 tab-separated columns is dropped; a
         * row whose network regex does not compile is kept but gets a null regex (the tracker has
         * code signatures only). Rows are returned sorted by id so iteration is deterministic.
         */
        fun parse(tsv: String): TrackerDb {
            val list = ArrayList<Tracker>(512)
            for (raw in tsv.lineSequence()) {
                if (raw.isEmpty()) continue
                if (raw.startsWith("#")) continue
                val parts = raw.split('\t')
                if (parts.size < EXPECTED_COLS) continue
                val id = parts[0].trim().toIntOrNull() ?: continue
                val name = parts[1]
                if (name.isEmpty()) continue
                val catRaw = parts[2].trim()
                val cats = if (catRaw.isEmpty()) emptyList()
                else catRaw.split(',').map { it.trim() }.filter { it.isNotEmpty() }
                val website = parts[3]
                val regexRaw = parts[4]
                val rx = if (regexRaw.isEmpty()) null
                else try { Regex(regexRaw) } catch (_: Throwable) { null }
                val sigsRaw = parts[5]
                val sigs = if (sigsRaw.isEmpty()) emptyList()
                else sigsRaw.split('|').filter { it.isNotEmpty() }
                list.add(Tracker(id, name, cats, website, rx, sigs))
            }
            return TrackerDb(list.sortedBy { it.id })
        }
    }
