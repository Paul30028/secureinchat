// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian

import android.app.PendingIntent
import android.content.Intent
import android.net.VpnService
import android.os.Build
import android.os.Handler
import android.os.Looper
import android.service.quicksettings.Tile
import android.service.quicksettings.TileService
import java.util.Locale

class GuardianTileService : TileService() {

    private val uiHandler = Handler(Looper.getMainLooper())

    private fun txt(en: String, zh: String): String =
        if (Locale.getDefault().language == "zh") zh else en

    override fun onStartListening() {
        super.onStartListening()
        refreshTile()
    }

    private fun refreshTile() {
        val tile = qsTile ?: return
        val running = GuardianVpnService.isRunning
        val loading = GuardianVpnService.isLoading
        tile.state = if (running) Tile.STATE_ACTIVE else Tile.STATE_INACTIVE
        tile.label = "Guardian"
        if (Build.VERSION.SDK_INT >= 29) {
            tile.subtitle = when {
                loading -> txt("Starting…", "正在启动…")
                running -> txt("On", "已开启")
                else -> txt("Off", "已关闭")
            }
        }
        try { tile.updateTile() } catch (_: Throwable) {}
    }

    override fun onClick() {
        super.onClick()
        // While the list is loading, ignore clicks. GuardianVpnService.isLoading is written from the
        // service thread but is @Volatile, and the service is a singleton within the process, so we
        // can rely on it here without extra coordination.
        if (GuardianVpnService.isLoading) return

        if (GuardianVpnService.isRunning) {
            // Plain startService: the stop path never calls startForeground, and a
            // startForegroundService() call that is not followed by startForeground() within a
            // few seconds crashes with ForegroundServiceDidNotStartInTimeException.
            val i = Intent(this, GuardianVpnService::class.java)
                .setAction(GuardianVpnService.ACTION_STOP)
            startService(i)
        } else {
            val prep = VpnService.prepare(this)
            if (prep != null) {
                // Consent has not been granted yet. TileService cannot startActivityForResult, so
                // hand off to MainActivity, which will show the system consent dialog and start
                // the VPN on RESULT_OK.
                openMainActivityForConsent()
            } else {
                val i = Intent(this, GuardianVpnService::class.java)
                    .setAction(GuardianVpnService.ACTION_START)
                startForegroundService(i)
            }
        }
        // Service start/stop is asynchronous, so refresh the tile once more a second later.
        uiHandler.postDelayed({ refreshTile() }, 1000)
    }

    private fun openMainActivityForConsent() {
        val activityIntent = Intent(this, MainActivity::class.java)
            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        if (Build.VERSION.SDK_INT >= 34) {
            val pi = PendingIntent.getActivity(
                this, 0, activityIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
            )
            startActivityAndCollapse(pi)
        } else {
            @Suppress("DEPRECATION")
            startActivityAndCollapse(activityIntent)
        }
    }
}

关于 cmd statusbar click-tile org.secureinchat.guardian/.GuardianTileService 的确认与注意点：
这个命令会直接调用 tile 的 onClick()（等价于点按磁贴本体，不是展开箭头），与我们读取 GuardianVpnService.isRunning/isLoading 直接决定启动/停止的逻辑一致；已授权状态下第一次点击会走 startForegroundService(ACTION_START)，第二次点击会走 startService(ACTION_STOP)。功能上没问题。

TileService 只会在磁贴"被系统认为可见"（展开面板、或至少监听中）时才处于 listening 状态；cmd statusbar click-tile 在未拉起过面板时也能触发点击，但 qsTile 可能为 null，因此 refreshTile() 内有 ?: return 保护，不会崩。

磁贴启动后 1 秒再刷一次可能仍然偏早：72k 域名列表在多数设备上加载 <500ms，但极慢设备或冷启动情况下可能超过 1 秒，磁贴会先停在 "Starting…"；随后 GuardianVpnService 里的 requestListeningState 会在 isLoading = false 后主动触发 onStartListening()，把磁贴刷成 "On"，所以最终一致。

首次使用（尚未授权 VPN）时点击磁贴不会直接启动 VPN，而是打开 MainActivity 走 VpnService.prepare 系统授权对话框；授权返回后 MainActivity 会调用 startVpnService()，所以最终也会启动。也就是说"打开 MainActivity 让用户授权"这条路径下，授权成功后不需要用户再按 Start。MainActivity 不需要改。

一个已知限制：requestListeningState 在部分 ROM 上会被系统忽略（尤其 Android 13 定制系统），此时磁贴要等用户下次展开面板才会刷新；功能不受影响，只是视觉延迟。这是 AOSP 已知行为，不算 bug。

磁贴图标用 @android:drawable/ic_lock_lock，与通知图标一致，不需要新增 drawable 资源、不需要新增权限。

若把磁贴放在磁贴区之外、或用户尚未编辑磁贴面板加入它，cmd statusbar click-tile 仍能触发 onClick，但用户手动"下拉点击"是看不到它的——这是系统行为，不是实现问题。
