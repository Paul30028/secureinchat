// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian.dns

/**
 * Domain block/allow matcher.
 *
 * Four rule sets are kept separately so that precedence is explicit and stable:
 *
 *   user allow  >  user block  >  builtin allow (@@)  >  builtin block
 *
 * The first three are consulted in that order, walking the query name and each of its parent
 * domains one label at a time. The first hit decides the answer. This means:
 *
 *  - a user allow always wins, even over a user block of the same name or a builtin block;
 *  - a user block beats a builtin @@ allow, which is the point of letting users blacklist a
 *    domain whose builtin whitelisting they disagree with;
 *  - builtin @@ still overrides builtin block, as before.
 *
 * The bundled lists are immutable per process; the user lists are applied with plusAllowed /
 * plusBlocked, both of which return a new instance and leave the receiver untouched.
 *
 * matched by suffix: "example.com" covers "a.b.example.com" but not "badexample.com".
 */
class Blocklist private constructor(
    private val builtinBlocked: HashSet<String>,
    private val builtinAllowed: HashSet<String>,
    private val userBlocked: HashSet<String>,
    private val userAllowed: HashSet<String>
) {
    /** Number of domains in the bundled block list (not counting user blocks). */
    val blockedCount: Int get() = builtinBlocked.size

    /** Number of builtin @@ allow rules (not counting user allows). */
    val allowedCount: Int get() = builtinAllowed.size

    fun isBlocked(name: String): Boolean {
        try {
            val n = normalizeHostname(name) ?: return false
            var cur = n
            while (true) {
                if (userAllowed.contains(cur)) return false
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            cur = n
            while (true) {
                if (userBlocked.contains(cur)) return true
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            cur = n
            while (true) {
                if (builtinAllowed.contains(cur)) return false
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            cur = n
            while (true) {
                if (builtinBlocked.contains(cur)) return true
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            return false
        } catch (t: Throwable) {
            return false
        }
    }

    /**
     * Returns a new Blocklist with [extra] added to the user allow set. Invalid entries are ignored.
     * User allow is the highest-priority set, so a user allow overrides both user blocks and the
     * builtin lists.
     */
    fun plusAllowed(extra: Collection<String>): Blocklist {
        val newAllowed = HashSet<String>(userAllowed)
        for (s in extra) {
            val n = normalizeOrNull(s) ?: continue
            newAllowed.add(n)
        }
        return Blocklist(HashSet(builtinBlocked), HashSet(builtinAllowed), HashSet(userBlocked), newAllowed)
    }

    /**
     * Returns a new Blocklist with [extra] added to the user block set. Invalid entries are ignored.
     * User block is consulted after user allow but before the builtin lists, so it overrides
     * builtin @@ rules while still losing to an explicit user allow of the same name.
     */
    fun plusBlocked(extra: Collection<String>): Blocklist {
        val newBlocked = HashSet<String>(userBlocked)
        for (s in extra) {
            val n = normalizeOrNull(s) ?: continue
            newBlocked.add(n)
        }
        return Blocklist(HashSet(builtinBlocked), HashSet(builtinAllowed), newBlocked, HashSet(userAllowed))
    }

    companion object {
        private val WS = Regex("\\s+")

        private val IGNORED = setOf(
            "localhost", "localhost.localdomain", "broadcasthost", "local",
            "0.0.0.0", "127.0.0.1", "ip6-localhost", "ip6-loopback"
        )

        fun normalizeOrNull(s: String): String? = normalizeHostname(s)

        fun parse(text: String): Blocklist {
            val builtinBlocked = HashSet<String>(1024)
            val builtinAllowed = HashSet<String>(64)

            for (rawLine in text.lineSequence()) {
                var line = rawLine.trim()
                if (line.isEmpty()) continue
                if (line.startsWith("#") || line.startsWith("!")) continue

                val hash = line.indexOf('#')
                if (hash >= 0) line = line.substring(0, hash).trim()
                if (line.isEmpty()) continue

                var ascii = true
                for (ch in line) {
                    if (ch.code >= 128) { ascii = false; break }
                }
                if (!ascii) continue

                if (line.startsWith("@@")) {
                    val domain = parseAdblock(line.substring(2))
                    if (domain != null) builtinAllowed.add(domain)
                    continue
                }

                if (line.startsWith("||")) {
                    val domain = parseAdblock(line)
                    if (domain != null) builtinBlocked.add(domain)
                    continue
                }

                val tokens = line.split(WS)
                if (tokens.size >= 2 && looksLikeIp(tokens[0])) {
                    for (k in 1 until tokens.size) {
                        val d = normalizeHostname(tokens[k]) ?: continue
                        if (d in IGNORED) continue
                        builtinBlocked.add(d)
                    }
                    continue
                }

                val d = normalizeHostname(line) ?: continue
                if (d in IGNORED) continue
                builtinBlocked.add(d)
            }

            return Blocklist(builtinBlocked, builtinAllowed, HashSet(16), HashSet(16))
        }

        private fun parseAdblock(s: String): String? {
            if (!s.startsWith("||")) return null
            var body = s.substring(2)
            if (!body.endsWith("^")) return null
            body = body.substring(0, body.length - 1)
            if (body.contains('*') || body.contains('/') || body.contains('$') || body.contains('|')) return null
            return normalizeHostname(body)
        }

        private fun looksLikeIp(s: String): Boolean {
            val parts = s.split('.')
            if (parts.size != 4) return false
            for (p in parts) {
                if (p.isEmpty()) return false
                for (ch in p) if (!ch.isDigit()) return false
                val v = p.toIntOrNull() ?: return false
                if (v < 0 || v > 255) return false
            }
            return true
        }

        private fun normalizeHostname(s: String): String? {
            val d = s.trim().lowercase().trimEnd('.')
            if (d.isEmpty()) return null
            if (d.length > 253) return null
            if (!d.contains('.')) return null
            if (d.contains("..")) return null
            for (label in d.split('.')) {
                if (label.isEmpty()) return null
                if (label.length > 63) return null
                if (label.startsWith('-') || label.endsWith('-')) return null
                for (ch in label) {
                    val ok = (ch in 'a'..'z') || (ch in '0'..'9') || ch == '-' || ch == '_'
                    if (!ok) return null
                }
            }
            return d
        }
    }
}

// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian.dns

private var urn = 0
private fun urt(name: String, block: () -> Unit) {
    try { block(); urn++ } catch (e: Throwable) {
        println("FAIL $name: ${e.message}")
        throw e
    }
}
private fun ureq(a: Any?, b: Any?) { check(a == b) { "expected $a, got $b" } }

fun main() {

    urt("user block hits the domain and its subdomains, not siblings") {
        val base = Blocklist.parse("")
        val ext = base.plusBlocked(listOf("tracker.example.com"))
        check(ext.isBlocked("tracker.example.com")) { "exact" }
        check(ext.isBlocked("sub.tracker.example.com")) { "sub" }
        check(ext.isBlocked("a.b.tracker.example.com")) { "nested sub" }
        check(!ext.isBlocked("example.com")) { "parent" }
        check(!ext.isBlocked("badtracker.example.com")) { "sibling" }
        check(!ext.isBlocked("other.example.com")) { "other" }
        // base untouched
        check(!base.isBlocked("tracker.example.com")) { "base exact" }
        check(!base.isBlocked("sub.tracker.example.com")) { "base sub" }
    }

    urt("user allow wins over user block of the same name") {
        val base = Blocklist.parse("")
        val ext = base.plusBlocked(listOf("tracker.example.com"))
            .plusAllowed(listOf("tracker.example.com"))
        check(!ext.isBlocked("tracker.example.com")) { "exact allow" }
        check(!ext.isBlocked("sub.tracker.example.com")) { "sub allow" }
    }

    urt("order of plusBlocked / plusAllowed does not change the outcome") {
        val base = Blocklist.parse("")
        val a = base.plusBlocked(listOf("blocked.example.com"))
            .plusAllowed(listOf("safe.example.com"))
        val b = base.plusAllowed(listOf("safe.example.com"))
            .plusBlocked(listOf("blocked.example.com"))
        for (bl in listOf(a, b)) {
            check(bl.isBlocked("blocked.example.com")) { "blocked" }
            check(bl.isBlocked("x.blocked.example.com")) { "blocked sub" }
            check(!bl.isBlocked("safe.example.com")) { "safe" }
            check(!bl.isBlocked("y.safe.example.com")) { "safe sub" }
        }
    }

    urt("user block beats a builtin @@ allow") {
        // Builtin list contains both a block and an @@ allow for the same name. Without any user
        // rule, the builtin allow wins, as before.
        val text = "@@||internal.example.com^\ninternal.example.com"
        val base = Blocklist.parse(text)
        check(!base.isBlocked("internal.example.com")) { "builtin allow vs builtin block" }
        check(!base.isBlocked("sub.internal.example.com")) { "builtin allow sub" }

        // User block must override the builtin @@ rule.
        val ext = base.plusBlocked(listOf("internal.example.com"))
        check(ext.isBlocked("internal.example.com")) { "user block vs builtin allow" }
        check(ext.isBlocked("sub.internal.example.com")) { "user block vs builtin allow sub" }

        // base untouched
        check(!base.isBlocked("internal.example.com")) { "base unchanged" }
    }

    urt("user allow beats a builtin block") {
        val base = Blocklist.parse("tracker.example.com")
        check(base.isBlocked("tracker.example.com")) { "builtin block" }
        val ext = base.plusAllowed(listOf("tracker.example.com"))
        check(!ext.isBlocked("tracker.example.com")) { "user allow" }
        check(!ext.isBlocked("sub.tracker.example.com")) { "user allow sub" }
        // base untouched
        check(base.isBlocked("tracker.example.com")) { "base unchanged" }
    }

    urt("invalid inputs are ignored and leave the source untouched") {
        val base = Blocklist.parse("example.com")
        val bad = listOf(
            "", "com", "..", "a..b.com",
            "-x.com", "x-.com",
            "a".repeat(64) + ".com",
            "\u4e2d\u6587.com",
            " leading-space.com " // trims to a valid name, actually accepted
        )
        val extBlocked = base.plusBlocked(bad)
        // The entries that survive normalization are the trimmable one; everything else is dropped.
        check(extBlocked.isBlocked("leading-space.com")) { "trimmed accepted" }
        check(!extBlocked.isBlocked("com")) { "com not blocked" }
        check(extBlocked.isBlocked("example.com")) { "builtin still applies" }
        // base unchanged
        check(!base.isBlocked("leading-space.com")) { "base no user block" }
        check(base.isBlocked("example.com")) { "base builtin" }
        // Counts reflect the builtin list only.
        val ref = Blocklist.parse("example.com")
        ureq(extBlocked.blockedCount, ref.blockedCount)

        val extAllowed = base.plusAllowed(bad)
        check(extAllowed.isBlocked("example.com")) { "user allow of invalid ignored, builtin still active" }
        check(!extAllowed.isBlocked("leading-space.com")) { "user allow of trimmed name accepted" }
        check(!base.isBlocked("leading-space.com")) { "base no user allow" }
    }

    urt("blockedCount / allowedCount describe builtin lists, not user lists") {
        val base = Blocklist.parse("a.example.com\n@@||b.example.com^")
        ureq(base.blockedCount, 1)
        ureq(base.allowedCount, 1)
        val ext = base.plusBlocked(listOf("c.example.com"))
            .plusAllowed(listOf("d.example.com"))
        ureq(ext.blockedCount, 1)
        ureq(ext.allowedCount, 1)
    }

    println("UserRulesTest: $urn passed")
}
