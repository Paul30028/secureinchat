package org.secureinchat.guardian.dns

import java.util.concurrent.atomic.AtomicLong

class Stats {
    val queries = AtomicLong()
    val blocked = AtomicLong()
    val forwarded = AtomicLong()
    val answered = AtomicLong()
    val failed = AtomicLong()
    val malformed = AtomicLong()

    fun snapshot(): StatsSnapshot = StatsSnapshot(
        queries.get(), blocked.get(), forwarded.get(),
        answered.get(), failed.get(), malformed.get()
    )
}

data class StatsSnapshot(
    val queries: Long,
    val blocked: Long,
    val forwarded: Long,
    val answered: Long,
    val failed: Long,
    val malformed: Long
)

data class LogEntry(
    val timeMs: Long,
    val name: String,
    val qtype: Int,
    val blocked: Boolean,
    val uid: Int = -1
)

sealed interface Action {
    data class Respond(val packet: ByteArray, val name: String) : Action
    class Forward(val query: DnsQuery, val payload: ByteArray) : Action
    object Drop : Action
}

class QueryHandler(
    @Volatile var blocklist: Blocklist,
    val mode: BlockMode = BlockMode.NULL_IP,
    private val uidOf: (DnsQuery) -> Int = { -1 },
    private val clock: () -> Long = System::currentTimeMillis
) {
    val stats = Stats()
    val tally = AppTally()

    private val logCapacity = 200
    private val logEntries = arrayOfNulls<LogEntry>(logCapacity)
    private var logWritePos = 0
    private var logCount = 0
    private val logLock = Any()

    fun handle(packet: ByteArray, length: Int): Action {
        stats.queries.incrementAndGet()
        val q = DnsPacket.parseQuery(packet, length)
        if (q == null) {
            stats.malformed.incrementAndGet()
            return Action.Drop
        }
        val uid = try { uidOf(q) } catch (_: Throwable) { -1 }
        val blocked = blocklist.isBlocked(q.qname)
        tally.record(uid, q.qname, blocked)
        if (blocked) {
            stats.blocked.incrementAndGet()
            val resp = DnsPacket.buildBlockedResponse(q, mode)
            appendLog(q.qname, q.qtype, true, uid)
            return Action.Respond(resp, q.qname)
        }
        stats.forwarded.incrementAndGet()
        val payload = DnsWire.extractQueryPayload(packet, q)
        appendLog(q.qname, q.qtype, false, uid)
        return Action.Forward(q, payload)
    }

    fun onUpstreamAnswer(q: DnsQuery, answer: ByteArray, len: Int): ByteArray? {
        val resp = DnsPacket.buildResponseFromUpstream(q, answer, len)
        if (resp != null) {
            stats.answered.incrementAndGet()
        } else {
            stats.failed.incrementAndGet()
        }
        return resp
    }

    fun onUpstreamFailed(q: DnsQuery) {
        stats.failed.incrementAndGet()
    }

    fun snapshot(): StatsSnapshot = stats.snapshot()

    fun recent(): List<LogEntry> {
        synchronized(logLock) {
            val result = ArrayList<LogEntry>(logCount)
            for (i in 1..logCount) {
                val idx = (logWritePos - i + logCapacity) % logCapacity
                result.add(logEntries[idx]!!)
            }
            return result
        }
    }

    fun clearLog() {
        synchronized(logLock) {
            for (i in logEntries.indices) logEntries[i] = null
            logWritePos = 0
            logCount = 0
        }
        tally.clear()
    }

    private fun appendLog(name: String, qtype: Int, blocked: Boolean, uid: Int) {
        val entry = LogEntry(clock(), name, qtype, blocked, uid)
        synchronized(logLock) {
            logEntries[logWritePos] = entry
            logWritePos = (logWritePos + 1) % logCapacity
            if (logCount < logCapacity) logCount++
        }
    }
}

package org.secureinchat.guardian.dns

data class AppRow(
    val uid: Int,
    val queries: Long,
    val blocked: Long,
    val topBlocked: List<Pair<String, Int>>
)

class AppTally(
    val maxApps: Int = 300,
    val maxDomainsPerApp: Int = 100
) {
    private class AppData(val uid: Int) {
        var queries: Long = 0
        var blocked: Long = 0
        val domains = HashMap<String, Int>(16)
    }

    private val lock = Any()
    private val apps = HashMap<Int, AppData>(64)
    private val other = AppData(-2)

    fun record(uid: Int, name: String, blocked: Boolean) {
        synchronized(lock) {
            val app: AppData = if (uid == -2) {
                other
            } else {
                apps[uid] ?: run {
                    if (apps.size >= maxApps) other
                    else AppData(uid).also { apps[uid] = it }
                }
            }
            app.queries++
            if (blocked) {
                app.blocked++
                val existing = app.domains[name]
                if (existing != null) {
                    app.domains[name] = existing + 1
                } else if (app.domains.size < maxDomainsPerApp) {
                    app.domains[name] = 1
                }
            }
        }
    }

    fun snapshot(limit: Int = 30): List<AppRow> {
        synchronized(lock) {
            val all = ArrayList<AppData>(apps.size + 1)
            for (a in apps.values) if (a.queries > 0) all.add(a)
            if (other.queries > 0) all.add(other)
            all.sortWith(
                compareByDescending<AppData> { it.blocked }
                    .thenByDescending { it.queries }
                    .thenBy { it.uid }
            )
            val n = minOf(limit, all.size)
            val out = ArrayList<AppRow>(n)
            for (i in 0 until n) {
                val a = all[i]
                val top = a.domains.entries
                    .sortedWith(
                        compareByDescending<Map.Entry<String, Int>> { it.value }
                            .thenBy { it.key }
                    )
                    .take(20)
                    .map { it.key to it.value }
                out.add(AppRow(a.uid, a.queries, a.blocked, top))
            }
            return out
        }
    }

    fun clear() {
        synchronized(lock) {
            apps.clear()
            other.queries = 0
            other.blocked = 0
            other.domains.clear()
        }
    }
}

package org.secureinchat.guardian.dns

class Blocklist private constructor(
    private val blocked: HashSet<String>,
    private val allowed: HashSet<String>
) {
    val blockedCount: Int get() = blocked.size
    val allowedCount: Int get() = allowed.size

    fun isBlocked(name: String): Boolean {
        try {
            val n = normalizeHostname(name) ?: return false
            var cur = n
            while (true) {
                if (allowed.contains(cur)) return false
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            cur = n
            while (true) {
                if (blocked.contains(cur)) return true
                val dot = cur.indexOf('.')
                if (dot < 0) break
                cur = cur.substring(dot + 1)
            }
            return false
        } catch (t: Throwable) {
            return false
        }
    }

    fun plusAllowed(extra: Collection<String>): Blocklist {
        val newAllowed = HashSet<String>(allowed)
        for (s in extra) {
            val n = normalizeOrNull(s) ?: continue
            newAllowed.add(n)
        }
        return Blocklist(HashSet(blocked), newAllowed)
    }

    companion object {
        private val WS = Regex("\\s+")

        private val IGNORED = setOf(
            "localhost", "localhost.localdomain", "broadcasthost", "local",
            "0.0.0.0", "127.0.0.1", "ip6-localhost", "ip6-loopback"
        )

        fun normalizeOrNull(s: String): String? = normalizeHostname(s)

        fun parse(text: String): Blocklist {
            val blocked = HashSet<String>(1024)
            val allowed = HashSet<String>(64)

            for (rawLine in text.lineSequence()) {
                var line = rawLine.trim()
                if (line.isEmpty()) continue
                if (line.startsWith("#") || line.startsWith("!")) continue

                val hash = line.indexOf('#')
                if (hash >= 0) line = line.substring(0, hash).trim()
                if (line.isEmpty()) continue

                var ascii = true
                for (ch in line) {
                    if (ch.code >= 128) { ascii = false; break }
                }
                if (!ascii) continue

                if (line.startsWith("@@")) {
                    val domain = parseAdblock(line.substring(2))
                    if (domain != null) allowed.add(domain)
                    continue
                }

                if (line.startsWith("||")) {
                    val domain = parseAdblock(line)
                    if (domain != null) blocked.add(domain)
                    continue
                }

                val tokens = line.split(WS)
                if (tokens.size >= 2 && looksLikeIp(tokens[0])) {
                    for (k in 1 until tokens.size) {
                        val d = normalizeHostname(tokens[k]) ?: continue
                        if (d in IGNORED) continue
                        blocked.add(d)
                    }
                    continue
                }

                val d = normalizeHostname(line) ?: continue
                if (d in IGNORED) continue
                blocked.add(d)
            }

            return Blocklist(blocked, allowed)
        }

        private fun parseAdblock(s: String): String? {
            if (!s.startsWith("||")) return null
            var body = s.substring(2)
            if (!body.endsWith("^")) return null
            body = body.substring(0, body.length - 1)
            if (body.contains('*') || body.contains('/') || body.contains('$') || body.contains('|')) return null
            return normalizeHostname(body)
        }

        private fun looksLikeIp(s: String): Boolean {
            val parts = s.split('.')
            if (parts.size != 4) return false
            for (p in parts) {
                if (p.isEmpty()) return false
                for (ch in p) if (!ch.isDigit()) return false
                val v = p.toIntOrNull() ?: return false
                if (v < 0 || v > 255) return false
            }
            return true
        }

        private fun normalizeHostname(s: String): String? {
            val d = s.trim().lowercase().trimEnd('.')
            if (d.isEmpty()) return null
            if (d.length > 253) return null
            if (!d.contains('.')) return null
            if (d.contains("..")) return null
            for (label in d.split('.')) {
                if (label.isEmpty()) return null
                if (label.length > 63) return null
                if (label.startsWith('-') || label.endsWith('-')) return null
                for (ch in label) {
                    val ok = (ch in 'a'..'z') || (ch in '0'..'9') || ch == '-' || ch == '_'
                    if (!ok) return null
                }
            }
            return d
        }
    }
}

    udp[5] = (udpLen and 0xFF).toByte()

    ip[0] = 0x45
    ip[2] = ((totalLen ushr 8) and 0xFF).toByte()

private fun tBuildQuery(name: String, qtype: Int = 1): ByteArray {
    val dns = ByteArrayOutputStream()
    dns.write(0x12); dns.write(0x34)
    dns.write(0x01); dns.write(0x00)
    dns.write(0x00); dns.write(0x01)
    dns.write(0x00); dns.write(0x00)
    dns.write(0x00); dns.write(0x00)
    dns.write(0x00); dns.write(0x00)
    for (label in name.split('.')) {
        dns.write(label.length and 0xFF)
        dns.write(label.toByteArray(Charsets.US_ASCII))
    }
    dns.write(0x00)
    dns.write((qtype ushr 8) and 0xFF); dns.write(qtype and 0xFF)
    dns.write(0x00); dns.write(0x01)
    return tIpUdp(dns.toByteArray())
}

package org.secureinchat.guardian

import android.content.Context
import org.secureinchat.guardian.dns.Blocklist

object UserRulesStore {
    private const val PREFS = "user_rules"
    private const val KEY = "allowed"

    fun allowed(ctx: Context): Set<String> {
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val raw = prefs.getStringSet(KEY, emptySet()) ?: emptySet()
        return HashSet(raw)
    }

    fun add(ctx: Context, domain: String): Boolean {
        val n = Blocklist.normalizeOrNull(domain) ?: return false
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val cur = HashSet(prefs.getStringSet(KEY, emptySet()) ?: emptySet())
        if (!cur.add(n)) return true
        prefs.edit().putStringSet(KEY, cur).apply()
        return true
    }

    fun remove(ctx: Context, domain: String) {
        val n = Blocklist.normalizeOrNull(domain) ?: domain.trim().lowercase().trimEnd('.')
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val cur = HashSet(prefs.getStringSet(KEY, emptySet()) ?: emptySet())
        if (cur.remove(n)) prefs.edit().putStringSet(KEY, cur).apply()
    }
}

import android.content.Context

object AppLabels {
    private val cache = HashMap<Int, String>()
    private val lock = Any()

    fun label(ctx: Context, uid: Int): String {
        when (uid) {
            -1 -> return "Unknown app / 未知应用"
            -2 -> return "Other apps / 其他应用"
            1000 -> return "Android system / Android 系统"
        }
        synchronized(lock) { cache[uid]?.let { return it } }
        val label = try {
            val pm = ctx.packageManager
            val pkgs = pm.getPackagesForUid(uid)
            if (pkgs.isNullOrEmpty()) {
                "uid $uid"
            } else {
                val appInfo = pm.getApplicationInfo(pkgs[0], 0)
                pm.getApplicationLabel(appInfo).toString()
            }
        } catch (_: Throwable) {
            "uid $uid"
        }
        synchronized(lock) { cache[uid] = label }
        return label
    }
}

    private var readThread: Thread? = null

    private var upstream: Upstream? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> {
                stopVpnInternal()
                stopSelf()
                return START_NOT_STICKY
            }
            else -> {
                if (!running && !isLoading) {
                    val promoted = try { startForegroundSafe(); true } catch (_: RuntimeException) { false }
                    if (!promoted) {
                        stopSelf()
                        return START_NOT_STICKY
                    }
                    isLoading = true
                    Thread { startVpnBackground() }.apply { name = "guardian-start" }.start()
                }
            }
        }
        return START_STICKY
    }

    private fun startVpnBackground() {
        try {
            val blocklist = loadBlocklist()
            val userAllowed = UserRulesStore.allowed(applicationContext)
            val effective = if (userAllowed.isEmpty()) blocklist else blocklist.plusAllowed(userAllowed)
            val h = QueryHandler(effective, BlockMode.NULL_IP, { q -> uidOfQuery(q) })

            val builder = Builder()
            builder.setSession("Guardian")
            builder.addAddress("10.111.222.1", 32)
            builder.addDnsServer("10.111.222.2")
            builder.addRoute("10.111.222.2", 32)
            builder.setMtu(1500)
            builder.setBlocking(true)

            val pfd = try { builder.establish() } catch (_: Throwable) { null }
            if (pfd == null) {
                isLoading = false
                stopSelf()
                return
            }
            tun = pfd
            tunIn = FileInputStream(pfd.fileDescriptor)
            tunOut = FileOutputStream(pfd.fileDescriptor)

            activeHandler = h
            pool = Executors.newFixedThreadPool(8)

            isLoading = false

    }

    private fun uidOfQuery(q: DnsQuery): Int {
        if (Build.VERSION.SDK_INT < 29) return -1
        return try {
            val cm = getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
            val src = InetSocketAddress(InetAddress.getByAddress(q.srcIp), q.srcPort)
            val dst = InetSocketAddress(InetAddress.getByAddress(q.dstIp), q.dstPort)
            cm.getConnectionOwnerUid(OsConstants.IPPROTO_UDP, src, dst)
        } catch (_: Throwable) {
            -1
        }
    }

    private fun startForegroundSafe() {
        val nm = getSystemService(Context.NOTIFICATION_SERVICE) as NotificationManager
        if (nm.getNotificationChannel(CHANNEL_ID) == null) {
            nm.createNotificationChannel(
                NotificationChannel(CHANNEL_ID, "Guardian VPN", NotificationManager.IMPORTANCE_LOW)
            )
        }
        val stopIntent = Intent(this, GuardianVpnService::class.java).setAction(ACTION_STOP)
        val pi = PendingIntent.getService(
            this, 1, stopIntent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val notif: Notification = Notification.Builder(this, CHANNEL_ID)
            .setContentTitle("Guardian")
            .setContentText("DNS filtering active")
            .setSmallIcon(android.R.drawable.ic_lock_lock)
            .addAction(0, "Stop", pi)
            .setOngoing(true)
            .build()
        if (Build.VERSION.SDK_INT >= 34) {
            startForeground(NOTIF_ID, notif, ServiceInfo.FOREGROUND_SERVICE_TYPE_SYSTEM_EXEMPTED)
        } else if (Build.VERSION.SDK_INT >= 29) {
            startForeground(NOTIF_ID, notif, 0)
        } else {
            startForeground(NOTIF_ID, notif)
        }
    }

    private fun readLoop(pfd: ParcelFileDescriptor) {
        val buf = ByteArray(32767)
        val input = tunIn ?: return
        while (running) {
            val n = try { input.read(buf) } catch (_: Throwable) { if (!running) break else continue }
            if (n <= 0) continue
            val copy = buf.copyOf(n)
            val exec = pool ?: break
            try { exec.execute { processPacket(copy) } } catch (_: Throwable) {}
        }
    }

    private fun processPacket(packet: ByteArray) {
        val h = handler ?: return
        when (val act = h.handle(packet, packet.size)) {
            is Action.Respond -> writeTun(act.packet)
            is Action.Forward -> {
                val exec = pool ?: return
                try {
                    exec.execute {
                        val ans = upstream?.sendQuery(act.payload)
                        if (ans == null) {
                            h.onUpstreamFailed(act.query)
                        } else {
                            val resp = h.onUpstreamAnswer(act.query, ans, ans.size)
                            if (resp != null) writeTun(resp)
                        }
                    }
                } catch (_: Throwable) {}
            }
            Action.Drop -> {}
        }
    }

    private fun writeTun(packet: ByteArray) {
        val out = tunOut ?: return
        synchronized(writeLock) {
            try { out.write(packet) } catch (_: Throwable) {}
        }
    }

    private fun stopVpnInternal() {
        running = false
        isRunning = false
        isLoading = false
        try { tun?.close() } catch (_: Throwable) {}
        try { tunIn?.close() } catch (_: Throwable) {}
        try { tunOut?.close() } catch (_: Throwable) {}
        tun = null; tunIn = null; tunOut = null
        try { readThread?.join(1000) } catch (_: Throwable) {}
        readThread = null
        try { pool?.shutdownNow() } catch (_: Throwable) {}
        pool = null
        try { upstream?.stop() } catch (_: Throwable) {}
        upstream = null
        handler = null
        activeHandler = null
    }

    private fun loadBlocklist(): Blocklist = loadBlocklistStatic(applicationContext)

        const val ACTION_START = "org.secureinchat.guardian.action.START"
        const val ACTION_STOP = "org.secureinchat.guardian.action.STOP"
        private const val CHANNEL_ID = "guardian_vpn"
        private const val NOTIF_ID = 1
        private val LIST_ASSETS = listOf("blocklist-hagezi-light.txt", "blocklist-extra.txt")
        private val rulesLock = Any()

        @Volatile var isRunning: Boolean = false
            private set

        @Volatile var isLoading: Boolean = false
            private set

        @Volatile var activeHandler: QueryHandler? = null
            private set

        private fun loadBlocklistStatic(ctx: Context): Blocklist {
            return try {
                val text = LIST_ASSETS.joinToString("\n") { name ->
                    ctx.assets.open(name).use { it.bufferedReader().readText() }
                }
                Blocklist.parse(text)
            } catch (_: Throwable) {
                Blocklist.parse("")
            }
        }

        fun applyUserRules(ctx: Context) {
            val h = activeHandler ?: return
            Thread {
                synchronized(rulesLock) {
                    try {
                        val base = loadBlocklistStatic(ctx)
                        val userAllowed = UserRulesStore.allowed(ctx)
                        val effective = if (userAllowed.isEmpty()) base else base.plusAllowed(userAllowed)
                        h.blocklist = effective
                    } catch (_: Throwable) {}
                }
            }.apply { name = "guardian-rules" }.start()
        }
    }
