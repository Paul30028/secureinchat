// SPDX-License-Identifier: GPL-3.0-or-later
package org.secureinchat.guardian

import android.annotation.SuppressLint
import android.app.PendingIntent
import android.content.Intent
import android.net.VpnService
import android.os.Build
import android.os.Handler
import android.os.Looper
import android.service.quicksettings.Tile
import android.service.quicksettings.TileService
import java.util.Locale

class GuardianTileService : TileService() {

    private val uiHandler = Handler(Looper.getMainLooper())

    private fun txt(en: String, zh: String): String =
        if (Locale.getDefault().language == "zh") zh else en

    override fun onStartListening() {
        super.onStartListening()
        refreshTile()
    }

    private fun refreshTile() {
        val tile = qsTile ?: return
        val running = GuardianVpnService.isRunning
        val loading = GuardianVpnService.isLoading
        // Paused is a sub-state of "running": the VPN session stays up and the tile remains ACTIVE,
        // only the label changes. activeHandler is set while running, so this read is a plain null
        // check; isPaused() is clock-based and safe to call from the main thread.
        val paused = running && (GuardianVpnService.activeHandler?.isPaused() ?: false)
        tile.state = if (running) Tile.STATE_ACTIVE else Tile.STATE_INACTIVE
        tile.label = "Guardian"
        if (Build.VERSION.SDK_INT >= 29) {
            tile.subtitle = when {
                loading -> txt("Starting…", "正在启动…")
                paused -> txt("Paused", "已暂停")
                running -> txt("On", "已开启")
                else -> txt("Off", "已关闭")
            }
        }
        try { tile.updateTile() } catch (_: Throwable) {}
    }

    override fun onClick() {
        super.onClick()
        // While the list is loading, ignore clicks. GuardianVpnService.isLoading is written from the
        // service thread but is @Volatile, and the service is a singleton within the process, so we
        // can rely on it here without extra coordination.
        if (GuardianVpnService.isLoading) return

        if (GuardianVpnService.isRunning) {
            // Plain startService: the stop path never calls startForeground, and a
            // startForegroundService() call that is not followed by startForeground() within a
            // few seconds crashes with ForegroundServiceDidNotStartInTimeException.
            val i = Intent(this, GuardianVpnService::class.java)
                .setAction(GuardianVpnService.ACTION_STOP)
            startService(i)
        } else {
            val prep = VpnService.prepare(this)
            if (prep != null) {
                // Consent has not been granted yet. TileService cannot startActivityForResult, so open
                // MainActivity; the user taps Start there, which shows the system consent dialog.
                openMainActivityForConsent()
            } else {
                val i = Intent(this, GuardianVpnService::class.java)
                    .setAction(GuardianVpnService.ACTION_START)
                startForegroundService(i)
            }
        }
        // Service start/stop is asynchronous, so refresh the tile once more a second later.
        uiHandler.postDelayed({ refreshTile() }, 1000)
    }

    // The Intent overload is the only one before API 34; the PendingIntent overload is used from 34 on.
    @SuppressLint("StartActivityAndCollapseDeprecated")
    private fun openMainActivityForConsent() {
        val activityIntent = Intent(this, MainActivity::class.java)
            .addFlags(Intent.FLAG_ACTIVITY_NEW_TASK)
        if (Build.VERSION.SDK_INT >= 34) {
            val pi = PendingIntent.getActivity(
                this, 0, activityIntent,
                PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
            )
            startActivityAndCollapse(pi)
        } else {
            @Suppress("DEPRECATION")
            startActivityAndCollapse(activityIntent)
        }
    }
}

    xmlns:tools="http://schemas.android.com/tools">

    <uses-permission android:name="android.permission.INTERNET" />
    <uses-permission android:name="android.permission.ACCESS_NETWORK_STATE" />
    <uses-permission android:name="android.permission.FOREGROUND_SERVICE" />
    <uses-permission android:name="android.permission.FOREGROUND_SERVICE_SYSTEM_EXEMPTED" />
    <uses-permission android:name="android.permission.POST_NOTIFICATIONS" />


    <application
        android:label="Guardian"
        android:icon="@mipmap/ic_launcher"
        android:allowBackup="false"
        android:dataExtractionRules="@xml/data_extraction_rules"
        android:supportsRtl="true">

        <activity
            android:name=".MainActivity"
            android:exported="true"
            android:launchMode="singleTop">
            <intent-filter>
                <action android:name="android.intent.action.MAIN" />
                <category android:name="android.intent.category.LAUNCHER" />
            </intent-filter>
        </activity>

        <!-- systemExempted: the official FGS type page lists "VPN apps (configured using
             Settings > Network & Internet > VPN)" as allowed, alongside apps holding the exact-alarm
             permissions. Lint's ForegroundServicePermission check only knows the alarm route, so it is
             suppressed here; Guardian does not need alarms. The service starts only after
             VpnService.prepare() consent, and falls back to stopping if startForeground is refused.
             https://developer.android.com/develop/background-work/services/fgs/service-types -->
        <service
            tools:ignore="ForegroundServicePermission"
            android:name=".GuardianVpnService"
            android:exported="false"
            android:permission="android.permission.BIND_VPN_SERVICE"
            android:foregroundServiceType="systemExempted">
            <intent-filter>
                <action android:name="android.net.VpnService" />
            </intent-filter>
        </service>

        <!-- Quick Settings tile. BIND_QUICK_SETTINGS_TILE is a service-level protection, not a
             uses-permission: it tells the system only it may bind to this TileService. Exported
             is true because the system (SystemUI) is the caller. -->
        <service
            android:name=".GuardianTileService"
            android:exported="true"
            android:icon="@android:drawable/ic_lock_lock"
            android:label="Guardian"
            android:permission="android.permission.BIND_QUICK_SETTINGS_TILE">
            <intent-filter>
                <action android:name="android.service.quicksettings.action.QS_TILE" />
            </intent-filter>
            <meta-data
                android:name="android.service.quicksettings.ACTIVE_TILE"
                android:value="false" />
        </service>
    </application>
</manifest>

<!--
  Guardian stores no query history on disk: the only thing in SharedPreferences is the user's
  allow-list ("user_rules"), which is device-local policy and should not leave the phone. That
  said, allowBackup is already false in the manifest, so cloud-backup and device-transfer are
  already disabled at the OS level. This file exists so that targeting API 31+ does not trip the
  DataExtractionRules lint warning; it excludes every domain from both cloud-backup and
  device-transfer, making the intent explicit and robust if allowBackup is ever toggled on later.

  Domains listed here cover both credential-protected (normal) and device-protected storage.
-->
<data-extraction-rules>
    <cloud-backup>
        <exclude domain="root" path="." />
        <exclude domain="file" path="." />
        <exclude domain="database" path="." />
        <exclude domain="sharedpref" path="." />
        <exclude domain="external" path="." />
        <exclude domain="device_root" path="." />
        <exclude domain="device_file" path="." />
        <exclude domain="device_database" path="." />
        <exclude domain="device_sharedpref" path="." />

关于 GuardianVpnService.kt 的一个小改进（可选，但建议采纳）：
上一批已交付的 GuardianVpnService.kt 中，ACTION_PAUSE / ACTION_RESUME 分支更新了通知但没有调用 requestListeningState。磁贴打开时（onStartListening）会正确读取 activeHandler.isPaused() 并显示 "Paused"，所以严格符合需求。但如果用户正看着面板，磁贴上的 "On" 不会变成 "Paused"，直到下次重新打开。
建议在两个分支里各加一行（包在同一个 try/catch 里）：

<<<CODE
        ACTION_PAUSE -> {
            val h = handler
            if (h != null) {
                h.pause(PAUSE_DURATION_MS)
                updateNotification()
                scheduleNotifRefresh(PAUSE_DURATION_MS)
                try {
                    android.service.quicksettings.TileService.requestListeningState(
                        applicationContext,
                        android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                    )
                } catch (_: Throwable) {}
            }
            return START_STICKY
        }
        ACTION_RESUME -> {
            val h = handler
            if (h != null) {
                h.resume()
                updateNotification()
                cancelNotifRefresh()
                try {
                    android.service.quicksettings.TileService.requestListeningState(
                        applicationContext,
                        android.content.ComponentName(applicationContext, GuardianTileService::class.java)
                    )
                } catch (_: Throwable) {}
            }
            return START_STICKY
        }
