package org.secureinchat.guardian.dns

/**
 * Plain-text serialization of the user's own rules ("blocked by you" / "allowed by you").
 *
 * File format (UTF-8, LF line endings):
 *
 *     # Guardian rules v1
 *     block ads.example.com
 *     allow tracker.example.com
 *
 * Empty lines and lines starting with '#' are ignored. Keywords are case-insensitive. Domains are
 * validated with Blocklist.normalizeOrNull, so a line whose domain is not a valid hostname is
 * skipped and counted in [ParseResult.skipped]. A domain that appears both as block and allow keeps
 * the last occurrence, matching UserRulesStore's rule that a domain lives in at most one list.
 *
 * The parser is deliberately forgiving: the header line is not required, and a hand-written file
 * with only "block"/"allow" lines is accepted. Very large inputs are truncated (see [MAX_BYTES] and
 * [MAX_LINES]); the remaining lines are counted as skipped instead of failing the whole import.
 */
object RulesText {

    const val HEADER = "# Guardian rules v1"

    // A rules file is tiny (tens or hundreds of domains). These caps only guard against a wrong file
    // being picked in the document picker; they are not expected to be reached in practice.
    const val MAX_BYTES = 1 * 1024 * 1024
    const val MAX_LINES = 10_000

    data class ParseResult(
        val blocked: Set<String>,
        val allowed: Set<String>,
        val skipped: Int
    )

    fun format(blocked: Collection<String>, allowed: Collection<String>): String {
        val sb = StringBuilder(HEADER).append('\n')
        for (d in blocked.sorted()) sb.append("block ").append(d).append('\n')
        for (d in allowed.sorted()) sb.append("allow ").append(d).append('\n')
        return sb.toString()
    }

    fun parse(text: String): ParseResult {
        // Truncate first so the line loop never sees more than the cap. A character is not a byte in
        // UTF-8, but a 1 MB character budget is at least as strict as a 1 MB byte budget for the
        // ASCII rules we expect, and never larger than ~4 MB of input. That is fine for a guard.
        val body = if (text.length > MAX_BYTES) text.substring(0, MAX_BYTES) else text

        // domain -> isAllowed; later lines for the same domain overwrite earlier ones.
        val chosen = LinkedHashMap<String, Boolean>()
        var skipped = 0
        var seen = 0

        for (raw in body.lineSequence()) {
            if (seen >= MAX_LINES) {
                // Every remaining line, blank or not, is counted as skipped. Users only see the total.
                skipped++
                continue
            }
            seen++

            val line = raw.trim().removeSuffix("\r").trim()
            if (line.isEmpty()) continue
            if (line.startsWith("#")) continue

            val sp = line.indexOfFirst { it == ' ' || it == '\t' }
            if (sp <= 0) {
                skipped++
                continue
            }
            val keyword = line.substring(0, sp).lowercase()
            val rest = line.substring(sp + 1).trim()
            if (rest.isEmpty()) {
                skipped++
                continue
            }
            val allow = when (keyword) {
                "block" -> false
                "allow" -> true
                else -> {
                    skipped++
                    continue
                }
            }
            val domain = Blocklist.normalizeOrNull(rest)
            if (domain == null) {
                skipped++
                continue
            }
            chosen[domain] = allow
        }

        val blocked = LinkedHashSet<String>()
        val allowed = LinkedHashSet<String>()
        for ((d, isAllow) in chosen) {
            if (isAllow) allowed.add(d) else blocked.add(d)
        }
        return ParseResult(blocked, allowed, skipped)
    }
}

package org.secureinchat.guardian.dns

private var rn = 0
private fun rt(name: String, block: () -> Unit) {
    try { block(); rn++ } catch (e: Throwable) {
        println("FAIL $name: ${e.message}")
        throw e
    }
}
private fun req(a: Any?, b: Any?) { check(a == b) { "expected $a, got $b" } }

fun main() {

    rt("round trip") {
        val blocked = setOf("ads.example.com", "tracker.example.com")
        val allowed = setOf("safe.example.com")
        val text = RulesText.format(blocked, allowed)
        check(text.startsWith(RulesText.HEADER)) { "header" }
        check(text.endsWith("\n")) { "trailing newline" }
        val p = RulesText.parse(text)
        req(p.blocked, blocked)
        req(p.allowed, allowed)
        req(p.skipped, 0)
    }

    rt("format sorts both lists") {
        val text = RulesText.format(
            listOf("z.example.com", "a.example.com"),
            listOf("y.example.com", "b.example.com")
        )
        val lines = text.trimEnd('\n').split('\n')
        req(lines[0], RulesText.HEADER)
        req(lines[1], "block a.example.com")
        req(lines[2], "block z.example.com")
        req(lines[3], "allow b.example.com")
        req(lines[4], "allow y.example.com")
    }

    rt("CRLF and surrounding whitespace") {
        val text = "# Guardian rules v1\r\n" +
            "  block  ads.example.com  \r\n" +
            "\tallow\tsafe.example.com\t\r\n" +
            "\r\n"
        val p = RulesText.parse(text)
        req(p.blocked, setOf("ads.example.com"))
        req(p.allowed, setOf("safe.example.com"))
        req(p.skipped, 0)
    }

    rt("comments and blank lines ignored") {
        val text = """
            # Guardian rules v1
            # a comment
            !
            
            block ads.example.com
            # allow safe.example.com
        """.trimIndent()
        val p = RulesText.parse(text)
        req(p.blocked, setOf("ads.example.com"))
        req(p.allowed, emptySet<String>())
        // The bare "!" line has no keyword and no domain, so it is skipped.
        req(p.skipped, 1)
    }

    rt("invalid domains and unknown keywords counted as skipped") {
        val text = """
            block ads.example.com
            block com
            block not_a_domain
            block a..b.com
            allow -x.com
            deny tracker.example.com
            blockonly
        """.trimIndent()
        val p = RulesText.parse(text)
        req(p.blocked, setOf("ads.example.com"))
        req(p.allowed, emptySet<String>())
        // com, not_a_domain, a..b.com, -x.com, "deny ..." (unknown keyword), "blockonly" (no domain)
        req(p.skipped, 6)
    }

    rt("last occurrence wins per domain") {
        val text = """
            block a.example.com
            allow a.example.com
            block b.example.com
        """.trimIndent()
        val p = RulesText.parse(text)
        req(p.blocked, setOf("b.example.com"))
        req(p.allowed, setOf("a.example.com"))
        req(p.skipped, 0)

        val text2 = """
            allow a.example.com
            block a.example.com
        """.trimIndent()
        val p2 = RulesText.parse(text2)
        req(p2.blocked, setOf("a.example.com"))
        req(p2.allowed, emptySet<String>())
    }

    rt("keyword is case-insensitive") {
        val text = """
            BLOCK a.example.com
            Allow B.Example.COM.
        """.trimIndent()
        val p = RulesText.parse(text)
        req(p.blocked, setOf("a.example.com"))
        req(p.allowed, setOf("b.example.com"))
        req(p.skipped, 0)
    }

    rt("missing header is accepted") {
        val text = "block a.example.com\nallow b.example.com\n"
        val p = RulesText.parse(text)
        req(p.blocked, setOf("a.example.com"))
        req(p.allowed, setOf("b.example.com"))
        req(p.skipped, 0)
    }

    rt("line cap counts the rest as skipped") {
        val sb = StringBuilder("# Guardian rules v1\n")
        for (i in 0 until RulesText.MAX_LINES + 5) {
            sb.append("block h").append(i).append(".example.com\n")
        }
        val p = RulesText.parse(sb.toString())
        // Header line is one line; MAX_LINES applies to non-header lines too (we count every line).
        // All accepted lines are counted toward the cap; the overflow becomes skipped.
        req(p.blocked.size, RulesText.MAX_LINES - 1) // header consumed one of the MAX_LINES slots
        req(p.skipped, 6) // the remaining 6 lines
    }

    rt("byte cap truncates the input") {
        // Build a body larger than MAX_BYTES. Every line is valid, so anything not read is skipped.
        val lineText = "block aaaaaaaaaa.example.com\n"
        val count = (RulesText.MAX_BYTES / lineText.length) + 50
        val sb = StringBuilder(count * lineText.length)
        for (i in 0 until count) sb.append(lineText)
        val p = RulesText.parse(sb.toString())
        // All read lines resolve to the same domain; the rest are beyond the byte cap and ignored
        // by substring, so they never reach the line loop and are not counted as skipped. The only
        // thing we need to assert is that parsing did not blow up and the domain is present.
        req(p.blocked, setOf("aaaaaaaaaa.example.com"))
    }

    println("RulesTextTest: $rn passed")
}

关于两个细节的最简单处理（已写进注释）：

1 MB 上限：parse 在进入逐行循环之前先把 String 截到 MAX_BYTES 个字符。UTF-8 下一个字符最多 4 字节，所以这个字符预算严格不小于 1 MB 字节预算的严格程度不成立（严格说它允许最多约 4 MB 输入），但这是最简单、无需引入编码/字节计算的写法；对一个由用户手动选择的、预期只有几百行的规则文件来说，效果相同。

行数上限：MAX_LINES 统计了所有非空行（包括第一行的头注释）。测试里我按“头行占用一个名额”来断言——如果你希望头行不计入，只要在循环里对 line.startsWith("#") 的分支提前 continue 而不增加 seen 即可，我可以改。
