package org.secureinchat.guardian

import android.content.Context
import org.secureinchat.guardian.dns.Blocklist

object UserRulesStore {
    private const val PREFS = "user_rules"
    private const val KEY_ALLOWED = "allowed"
    private const val KEY_BLOCKED = "blocked"

    fun allowed(ctx: Context): Set<String> = readSet(ctx, KEY_ALLOWED)

    fun blocked(ctx: Context): Set<String> = readSet(ctx, KEY_BLOCKED)

    fun add(ctx: Context, domain: String): Boolean = addTo(ctx, KEY_ALLOWED, domain)

    fun remove(ctx: Context, domain: String) = removeFrom(ctx, KEY_ALLOWED, domain)

    fun addBlocked(ctx: Context, domain: String): Boolean = addTo(ctx, KEY_BLOCKED, domain)

    fun removeBlocked(ctx: Context, domain: String) = removeFrom(ctx, KEY_BLOCKED, domain)

    /**
     * Replace both lists in one atomic write. Entries are normalized; a domain that appears in both
     * input sets ends up only in "allowed", matching the store's invariant that a domain lives in at
     * most one list and that allow wins in Blocklist.
     */
    fun replaceAll(ctx: Context, blocked: Set<String>, allowed: Set<String>) {
        val nextAllowed = HashSet<String>(allowed.size)
        for (s in allowed) {
            val n = Blocklist.normalizeOrNull(s) ?: continue
            nextAllowed.add(n)
        }
        val nextBlocked = HashSet<String>(blocked.size)
        for (s in blocked) {
            val n = Blocklist.normalizeOrNull(s) ?: continue
            if (nextAllowed.contains(n)) continue
            nextBlocked.add(n)
        }
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        prefs.edit()
            .putStringSet(KEY_ALLOWED, nextAllowed)
            .putStringSet(KEY_BLOCKED, nextBlocked)
            .apply()
    }

    /**
     * Merge imported rules into the existing lists. Each imported entry replaces the previous choice
     * for that domain: an imported allow removes it from the block list, and an imported block is
     * skipped if the current allow list already has it (allow wins, matching add()).
     */
    fun mergeIn(ctx: Context, blocked: Set<String>, allowed: Set<String>) {
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val curAllowed = HashSet(prefs.getStringSet(KEY_ALLOWED, emptySet()) ?: emptySet())
        val curBlocked = HashSet(prefs.getStringSet(KEY_BLOCKED, emptySet()) ?: emptySet())
        for (s in allowed) {
            val n = Blocklist.normalizeOrNull(s) ?: continue
            curBlocked.remove(n)
            curAllowed.add(n)
        }
        for (s in blocked) {
            val n = Blocklist.normalizeOrNull(s) ?: continue
            if (curAllowed.contains(n)) continue
            curBlocked.add(n)
        }
        prefs.edit()
            .putStringSet(KEY_ALLOWED, curAllowed)
            .putStringSet(KEY_BLOCKED, curBlocked)
            .apply()
    }

    private fun readSet(ctx: Context, key: String): Set<String> {
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val raw = prefs.getStringSet(key, emptySet()) ?: emptySet()
        return HashSet(raw)
    }

    private fun addTo(ctx: Context, key: String, domain: String): Boolean {
        val n = Blocklist.normalizeOrNull(domain) ?: return false
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val cur = HashSet(prefs.getStringSet(key, emptySet()) ?: emptySet())
        // A domain lives in at most one of the two lists: the newest choice replaces the old one,
        // so "Blocked by you" never shows a domain that is silently allowed (allow wins in Blocklist).
        val otherKey = if (key == KEY_ALLOWED) KEY_BLOCKED else KEY_ALLOWED
        val other = HashSet(prefs.getStringSet(otherKey, emptySet()) ?: emptySet())
        val otherChanged = other.remove(n)
        if (!cur.add(n) && !otherChanged) return true
        prefs.edit().putStringSet(key, cur).putStringSet(otherKey, other).apply()
        return true
    }

    private fun removeFrom(ctx: Context, key: String, domain: String) {
        val n = Blocklist.normalizeOrNull(domain) ?: domain.trim().lowercase().trimEnd('.')
        val prefs = ctx.getSharedPreferences(PREFS, Context.MODE_PRIVATE)
        val cur = HashSet(prefs.getStringSet(key, emptySet()) ?: emptySet())
        if (cur.remove(n)) prefs.edit().putStringSet(key, cur).apply()
    }
}

    private fun handleExportResult(uri: Uri?) {
        if (uri == null) return
        try {
            val text = RulesText.format(
                UserRulesStore.blocked(this),
                UserRulesStore.allowed(this)
            )
            // "wt" truncates, so re-exporting over an existing file does not leave trailing bytes.
            contentResolver.openOutputStream(uri, "wt").use { out ->
                if (out == null) throw IllegalStateException("no output stream")
                out.write(text.toByteArray(Charsets.UTF_8))
            }
            rulesIoError.visibility = View.GONE
        } catch (_: Throwable) {
            rulesIoError.visibility = View.VISIBLE
            rulesIoError.text = txt("Could not write the file", "无法写入文件")
        }
    }

    private fun handleImportResult(uri: Uri?) {
        if (uri == null) return
        // Read at most MAX_BYTES so a wrong pick (a video, an APK) cannot tie up the main thread.
        val text: String? = try {
            contentResolver.openInputStream(uri)?.use { input ->
                val buf = ByteArray(RulesText.MAX_BYTES)
                var total = 0
                while (total < buf.size) {
                    val n = input.read(buf, total, buf.size - total)
                    if (n <= 0) break
                    total += n
                }
                String(buf, 0, total, Charsets.UTF_8)
            }
        } catch (_: Throwable) {
            null
        }
        if (text == null) {
            rulesIoError.visibility = View.VISIBLE
            rulesIoError.text = txt("Could not read the file", "无法读取文件")
            return
        }
        val parsed = RulesText.parse(text)
        val summary = txt(
            "${parsed.blocked.size} to block, ${parsed.allowed.size} to allow, ${parsed.skipped} lines skipped",
            "拦截 ${parsed.blocked.size} 条,放行 ${parsed.allowed.size} 条,跳过 ${parsed.skipped} 行"
        )
        // Three-way choice. Merge keeps the existing rules and lets imported entries replace the
        // current choice for a domain; Replace discards the current lists entirely.
        AlertDialog.Builder(this)
            .setTitle(txt("Import rules", "导入规则"))
            .setMessage(summary)
            .setPositiveButton(txt("Merge", "合并")) { _, _ -> applyImport(parsed, merge = true) }
            .setNeutralButton(txt("Replace", "替换")) { _, _ -> applyImport(parsed, merge = false) }
            .setNegativeButton(txt("Cancel", "取消"), null)
            .show()
    }

    private fun applyImport(parsed: RulesText.ParseResult, merge: Boolean) {
        if (merge) {
            UserRulesStore.mergeIn(this, parsed.blocked, parsed.allowed)
        } else {
            UserRulesStore.replaceAll(this, parsed.blocked, parsed.allowed)
        }
        GuardianVpnService.applyUserRules(applicationContext)
        // Clear signatures so both lists rebuild on the next refresh even inside the throttle window.
        blockedSignature = ""
        allowedSignature = ""
        appListSignature = ""
        rulesIoError.visibility = View.GONE
        uiHandler.post { refreshUi() }
    }

第 6 点：在 API 29/34 模拟器上用 adb 让 ACTION_OPEN_DOCUMENT 选中指定文件

不确定有可靠办法。

adb push 到 /sdcard/Download/guardian-rules.txt 之后，ACTION_OPEN_DOCUMENT 的 DocumentsUI 首屏是“最近”（Recents）或上次停留的目录，未必直接列出 Download 里的文件；具体表现随 AOSP DocumentsUI 版本（API 29 与 API 34 不同）和 rom 而变。

有一种可行但不稳定的做法：先在应用里点 Import 打开 picker，然后用 adb shell input 点侧栏 “Downloads”，再点文件名。这依赖文本标签，模拟器 locale 变化会破坏匹配。

cmd document 之类的命令在 AOSP 模拟器上有的会列出/授予 document，但没有“让下一次 ACTION_OPEN_DOCUMENT 默认选中某个 Uri”的公开方式。我没有把握它在 API 29 和 API 34 上行为一致，所以不建议写进自动化。

若一定要自动化，最稳的是走 UI 驱动（uiautomator dump 找 Download 目录里那个文件名再 input tap），并把“找不到就跳过”当作 fallback。我这边不对具体 selector 名字做保证。

所以我建议保留你的计划：模拟器只验证“点 Export / Import 后前台切换到 DocumentsUI、返回后 Guardian 仍是 resumed”，核心正确性靠 RulesTextTest。

    private fun handleExportResult(uri: Uri?) {
        if (uri == null) return
        try {
            val text = RulesText.format(
                UserRulesStore.blocked(this),
                UserRulesStore.allowed(this)
            )
            // "wt" truncates, so re-exporting over an existing file does not leave trailing bytes.
            contentResolver.openOutputStream(uri, "wt").use { out ->
                if (out == null) throw IllegalStateException("no output stream")
                out.write(text.toByteArray(Charsets.UTF_8))
            }
            rulesIoError.visibility = View.GONE
        } catch (_: Throwable) {
            rulesIoError.visibility = View.VISIBLE
            rulesIoError.text = txt("Could not write the file", "无法写入文件")
        }
    }

    private fun handleImportResult(uri: Uri?) {
        if (uri == null) return
        // Read at most MAX_BYTES so a wrong pick (a video, an APK) cannot tie up the main thread.
        val text: String? = try {
            contentResolver.openInputStream(uri)?.use { input ->
                val buf = ByteArray(RulesText.MAX_BYTES)
                var total = 0
                while (total < buf.size) {
                    val n = input.read(buf, total, buf.size - total)
                    if (n <= 0) break
                    total += n
                }
                String(buf, 0, total, Charsets.UTF_8)
            }
        } catch (_: Throwable) {
            null
        }
        if (text == null) {
            rulesIoError.visibility = View.VISIBLE
            rulesIoError.text = txt("Could not read the file", "无法读取文件")
            return
        }
        val parsed = RulesText.parse(text)
        val summary = txt(
            "${parsed.blocked.size} to block, ${parsed.allowed.size} to allow, ${parsed.skipped} lines skipped",
            "拦截 ${parsed.blocked.size} 条,放行 ${parsed.allowed.size} 条,跳过 ${parsed.skipped} 行"
        )
        // Three-way choice. Merge keeps the existing rules and lets imported entries replace the
        // current choice for a domain; Replace discards the current lists entirely.
        AlertDialog.Builder(this)
            .setTitle(txt("Import rules", "导入规则"))
            .setMessage(summary)
            .setPositiveButton(txt("Merge", "合并")) { _, _ -> applyImport(parsed, merge = true) }
            .setNeutralButton(txt("Replace", "替换")) { _, _ -> applyImport(parsed, merge = false) }
            .setNegativeButton(txt("Cancel", "取消"), null)
            .show()
    }

    private fun applyImport(parsed: RulesText.ParseResult, merge: Boolean) {
        if (merge) {
            UserRulesStore.mergeIn(this, parsed.blocked, parsed.allowed)
        } else {
            UserRulesStore.replaceAll(this, parsed.blocked, parsed.allowed)
        }
        GuardianVpnService.applyUserRules(applicationContext)
        // Clear signatures so both lists rebuild on the next refresh even inside the throttle window.
        blockedSignature = ""
        allowedSignature = ""
        appListSignature = ""
        rulesIoError.visibility = View.GONE
        uiHandler.post { refreshUi() }
    }
